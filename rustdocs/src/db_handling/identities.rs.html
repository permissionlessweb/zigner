1//! Creating, collecting, and removing addresses in cold database
2//!
3//! In Vault and Vault-related ecosystem the address is a key pair within a
4//! certain network. Vault Rust database stores only **non-secret** data
5//! associated with each address generated by user.
6//!
7//! Required data to generate address:
8//!
9//! - seed phrase (secret words set, must be always zeroized after use and is
10//!   never stored in Rust database)
11//! - derivation path (`/soft//hard///password`); password part, i.e. whatever
12//!   follows `///` is not stored in Rust database and must be zeroized after use
13//! - [`NetworkSpecsKey`]
14//!
15//! The combination of seed phrase and derivation path allows to generate key
16//! pair using the encryption algorithm supported by the network, for which the
17//! address is generated.
18//!
19//! Vault stores only the public key, it is used together with encryption
20//! algorithm [`Encryption`] to generate
21//! [`AddressKey`].
22//!
23//! Same `AddressKey` could correspond to more than one network if the addresses
24//! were produced for same seed phrase and derivation path, and networks use
25//! same encryption algorithm. The networks specs keys of associated networks
26//! are stored in `network_id` field of `AddressDetails`.
27//!
28//! Vault interface shows those addresses as separate entities.
29//!
30//! Non-secret data associated with `AddressKey` is stored in
31//! [`ADDRTREE`] tree of the cold database as SCALE-encoded
32//! [`AddressDetails`].

93/// Get display address for a multisigner - checks for stored addresses first
94#[cfg(feature = "active")]
95fn get_display_address(
96    database: &sled::Db,
97    multisigner: &MultiSigner,
98    optional_prefix: Option<u16>,
99    encryption: Encryption,
100    network_name: Option<&str>,
101) -> String {
102    // For Penumbra, try to get the stored bech32m address
103    if encryption == Encryption::Penumbra {
104        let ak_hex = hex::encode(multisigner_to_public(multisigner));
105        if let Ok(Some(address)) = crate::penumbra::get_penumbra_address(database, &ak_hex) {
106            return address;
107        }
108    }
109    // For Zcash, try to get the stored unified address
110    #[cfg(feature = "zcash")]
111    if encryption == Encryption::Zcash {
112        let pubkey_hex = hex::encode(multisigner_to_public(multisigner));
113        if let Ok(Some(address)) = crate::zcash::get_zcash_address(database, &pubkey_hex) {
114            return address;
115        }
116    }
117    // For Cosmos, compute bech32 address on-the-fly from pubkey + chain prefix
118    // Same key for all cosmos chains (coin type 118), different bech32 prefix per chain
119    #[cfg(feature = "cosmos")]
120    if encryption == Encryption::Cosmos {
121        let pubkey_bytes = multisigner_to_public(multisigner);
122        if let Ok(address) =
123            crate::cosmos::pubkey_to_bech32_address(&pubkey_bytes, network_name.unwrap_or("cosmos"))
124        {
125            return address;
126        }
127    }
128    // Fall back to default display
129    print_multisigner_as_base58_or_eth_address(multisigner, optional_prefix, encryption)
130}

256/// Export info about keys and their addresses known to Vault
257pub fn export_key_set_addrs(
258    database: &sled::Db,
259    seed_name: &str,
260    exported_set: ExportedSet,
261) -> Result<ExportAddrs> {
262    let keys = get_addresses_by_seed_name(database, seed_name)?;
263    let root_multisigner = keys
264        .iter()
265        .find(|(_, a)| a.is_root())
266        .map(|(m, _)| m.to_owned())
267        .ok_or(Error::NoRootKeyForSeed(seed_name.to_owned()))?;
268
269    let mut derived_keys = vec![];
270
271    for key in keys {
272        if key.1.is_root() {
273            continue;
274        }
275
276        let mut selected = false;
277
278        match &exported_set {
279            ExportedSet::All => selected = true,
280            ExportedSet::Selected {
281                s: selected_derivations,
282            } => {
283                for selected_derivation in selected_derivations {
284                    if let Some(id) = &key.1.network_id {
285                        if selected_derivation.derivation == key.1.path
286                            && selected_derivation.network_specs_key == hex::encode(id.key())
287                        {
288                            selected = true;
289                            break;
290                        }
291                    }
292                }
293            }
294        }
295
296        if !selected {
297            continue;
298        }
299
300        if let Some(id) = &key.1.network_id {
301            let specs = get_network_specs(database, id)?;
302            let address_or_pub_key = print_multisigner_as_base58_or_eth_public_key(
303                &key.0,
304                Some(specs.specs.base58prefix),
305            );
306            derived_keys.push(AddrInfo {
307                address_or_pubkey: address_or_pub_key.clone(),
308                derivation_path: if key.1.path.is_empty() {
309                    None
310                } else {
311                    Some(key.1.path.to_owned())
312                },
313                encryption: key.1.encryption,
314                genesis_hash: specs.specs.genesis_hash,
315            });
316        }
317    }
318
319    Ok(ExportAddrs::V2(ExportAddrsV2::new(SeedInfo {
320        name: seed_name.to_owned(),
321        multisigner: root_multisigner,
322        derived_keys,
323    })))
324}

326pub fn import_all_addrs(
327    database: &sled::Db,
328    seed_derived_keys: Vec<SeedKeysPreview>,
329) -> Result<()> {
330    // Address preparation set, to be modified and used as `create_address`
331    // input.
332    let mut adds: Vec<(AddressKey, AddressDetails)> = vec![];
333
334    // Associated `Event` set
335    let mut events: Vec<Event> = vec![];
336
337    for addr in &seed_derived_keys {
338        for derived_key in &addr.derived_keys {
339            let path = derived_key.derivation_path.clone().unwrap_or_default();
340            let network_specs_key =
341                NetworkSpecsKey::from_parts(&derived_key.genesis_hash, &derived_key.encryption);
342            let network_specs = get_network_specs(database, &network_specs_key)?;
343            match create_derivation_address(
344                database,
345                &adds, // a single address is created, no data to check against here
346                &path,
347                &network_specs.specs,
348                &addr.name,
349                &derived_key.address,
350                derived_key
351                    .has_pwd
352                    .ok_or_else(|| Error::MissingPasswordInfo(path.to_owned()))?,
353            ) {
354                // success, updating address preparation set and `Event` set
355                Ok(prep_data) => {
356                    adds = prep_data.address_prep;
357                    events.extend_from_slice(&prep_data.history_prep);
358                }
359                // exactly same address already exists, ignoring it
360                Err(Error::DerivationExists { .. }) => (),
361
362                // some other error, processed as a real error
363                Err(e) => return Err(e),
364            }
365        }
366    }
367    TrDbCold::new()
368        .set_addresses(upd_id_batch(Batch::default(), adds)) // modify addresses data
369        .set_history(events_to_batch(database, events)?) // add corresponding history
370        .apply(database)
371}

373/// Derive Penumbra MultiSigner and bech32m address from seed phrase and path
374/// Path format: m/44'/6532'/account'/... (BIP44 style)
375/// Returns (MultiSigner, bech32m_address)
376#[cfg(all(feature = "active", feature = "penumbra"))]
377fn penumbra_derive_multisigner_and_address(
378    seed_phrase: &str,
379    path: &str,
380) -> Result<(MultiSigner, String)> {
381    use crate::penumbra;
382
383    // Parse account number from path (m/44'/6532'/account'/...)
384    // Default to account 0 if parsing fails
385    let account = parse_penumbra_account(path).unwrap_or(0);
386
387    // Derive spend key and full viewing key
388    let spend_key_bytes = penumbra::derive_spend_key_bytes(seed_phrase, account)?;
389    let fvk = spend_key_bytes.full_viewing_key()?;
390
391    // Derive the default address (address index 0 within this account)
392    let address = fvk.get_address(0)?;
393    let bech32m_address = address.to_bech32m()?;
394
395    // Get the verification key (ak) - 32 bytes
396    let ak_bytes = fvk.ak.to_bytes();
397
398    // Store as Ed25519 since both are 32-byte keys
399    let multisigner = MultiSigner::Ed25519(ed25519::Public::from_raw(ak_bytes));
400
401    Ok((multisigner, bech32m_address))
402}

404/// Parse account number from Penumbra BIP44 path
405/// Path format: m/44'/6532'/account'/...
406#[cfg(all(feature = "active", feature = "penumbra"))]
407fn parse_penumbra_account(path: &str) -> Option<u32> {
408    // Handle paths like "m/44'/6532'/0'" or just the path_id from network specs
409    let parts: Vec<&str> = path.trim_start_matches("m/").split('/').collect();
410
411    // Look for the account part (third element, after 44' and 6532')
412    if parts.len() >= 3 {
413        let account_part = parts[2].trim_end_matches('\'');
414        account_part.parse().ok()
415    } else {
416        None
417    }
418}

420/// Derive Ledger Ed25519 MultiSigner from seed phrase and path
421/// Path format: m/44'/354'/account'/0'/0' (BIP44 style for Polkadot)
422/// or //ledger//account_number for simplified UI input
423#[cfg(all(feature = "active", feature = "ledger"))]
424fn ledger_derive_multisigner(seed_phrase: &str, path: &str) -> Result<MultiSigner> {
425    use crate::ledger_ed25519;
426
427    // Parse the path to extract SLIP-0044 coin type and account number
428    // Support both full BIP44 path (m/44'/354'/0'/0'/0') and simplified (//ledger//0)
429    let (slip0044, account) = parse_ledger_path(path)?;

433
434    Ok(MultiSigner::Ed25519(ed25519::Public::from_raw(
435        key_pair.public_key,
436    )))
437}

439/// Parse Ledger BIP44 path to extract SLIP-0044 coin type and account number
440///
441/// Supports two modes:
442///
443/// **Universal mode** (recommended, works on all Substrate networks):
444/// - `//ledger//0` → coin 354 (Polkadot), account 0 - same key for all networks
445/// - `//ledger//1` → coin 354, account 1
446///
447/// **Legacy per-network mode** (different key per network):
448/// - `//polkadot_ledger//0` → coin 354, account 0
449/// - `//kusama_ledger//0` → coin 434, account 0 (different key!)
450/// - `//astar_ledger//0` → coin 810, account 0
451///
452/// **Full BIP44 path**:
453/// - `m/44'/354'/0'/0'/0'` → coin 354, account 0
454#[cfg(all(feature = "active", feature = "ledger"))]
455fn parse_ledger_path(path: &str) -> Result<(u32, u32)> {
456    use crate::ledger_ed25519::{SLIP0044_KUSAMA, SLIP0044_POLKADOT};
457
458    // SLIP-0044 coin types for supported networks (legacy per-network mode)
459    // See: https://github.com/satoshilabs/slips/blob/master/slip-0044.md
460    const SLIP0044_ASTAR: u32 = 810;
461    const SLIP0044_ACALA: u32 = 787;
462
463    // Try parsing as full BIP44 path: m/44'/slip0044'/account'/change'/address'
464    if path.starts_with("m/44'/") || path.starts_with("44'/") {
465        let parts: Vec<&str> = path.trim_start_matches("m/").split('/').collect();
466        if parts.len() >= 3 {
467            // parts[0] = "44'" (purpose)
468            // parts[1] = "354'" or "434'" (slip0044 coin type)
469            // parts[2] = "0'" (account)
470            let slip0044_str = parts[1].trim_end_matches('\'');
471            let account_str = parts[2].trim_end_matches('\'');
472
473            let slip0044 = slip0044_str
474                .parse::<u32>()
475                .map_err(|_| Error::InvalidDerivation(path.to_string()))?;
476            let account = account_str
477                .parse::<u32>()
478                .map_err(|_| Error::InvalidDerivation(path.to_string()))?;
479
480            return Ok((slip0044, account));
481        }
482    }
483
484    let path_lower = path.to_lowercase();
485
486    // Extract account number - find the last numeric part after //
487    let parts: Vec<&str> = path.split("//").filter(|s| !s.is_empty()).collect();
488    let account = parts
489        .iter()
490        .filter_map(|p| p.parse::<u32>().ok())
491        .next_back()
492        .unwrap_or(0);
493
494    // Universal mode: //ledger//0 (no network prefix)
495    // Uses Polkadot coin type (354) for all networks - same key everywhere
496    if path_lower.contains("ledger")
497        && !path_lower.contains("polkadot")
498        && !path_lower.contains("kusama")
499        && !path_lower.contains("astar")
500        && !path_lower.contains("acala")
501    {
502        return Ok((SLIP0044_POLKADOT, account));
503    }
504
505    // Legacy per-network mode: //<network>_ledger//0
506    // Different coin type per network - different keys
507    let slip0044 = if path_lower.contains("kusama") {
508        SLIP0044_KUSAMA
509    } else if path_lower.contains("astar") {
510        SLIP0044_ASTAR
511    } else if path_lower.contains("acala") {
512        SLIP0044_ACALA
513    } else {
514        // Default to Polkadot
515        SLIP0044_POLKADOT
516    };
517
518    Ok((slip0044, account))
519}

521/// Derive Cosmos MultiSigner and bech32 address from seed phrase and path
522/// Path format: m/44'/118'/account'/0/index (BIP44 style)
523/// or //cosmos//account_number for simplified UI input
524#[cfg(all(feature = "active", feature = "cosmos"))]
525fn cosmos_derive_multisigner_and_address(
526    seed_phrase: &str,
527    path: &str,
528    network_specs: Option<&NetworkSpecs>,
529) -> Result<(MultiSigner, String)> {
530    use crate::cosmos;
531
532    // Parse the path to extract coin type, account, and address index
533    let (coin_type, account, address_index) = cosmos::parse_cosmos_path(path)?;
534
535    // Derive the key
536    let key_pair = cosmos::derive_cosmos_key(seed_phrase, coin_type, account, address_index)
537        .map_err(|e| Error::Other(anyhow::anyhow!("Cosmos key derivation failed: {}", e)))?;
538
539    // Get bech32 prefix from network name
540    let prefix = network_specs
541        .map(|ns| match ns.name.as_str() {
542            "osmosis" => cosmos::PREFIX_OSMOSIS,
543            "noble" => cosmos::PREFIX_NOBLE,
544            "celestia" => cosmos::PREFIX_CELESTIA,
545            _ => cosmos::PREFIX_COSMOS,
546        })
547        .unwrap_or(cosmos::PREFIX_COSMOS);
548
549    // Generate bech32 address
550    let bech32_address = key_pair
551        .bech32_address(prefix)
552        .map_err(|e| Error::Other(anyhow::anyhow!("Bech32 encoding failed: {}", e)))?;
553
554    // Store as Ecdsa since Cosmos uses secp256k1
555    let multisigner = MultiSigner::Ecdsa(ecdsa::Public::from_raw(key_pair.public_key));
556
557    Ok((multisigner, bech32_address))
558}

585/// Get public key from seed phrase and derivation path
586fn full_address_to_multisigner(
587    mut full_address: String,
588    encryption: Encryption,
589) -> Result<MultiSigner> {
590    let multisigner_result = match encryption {
591        Encryption::Ed25519 => match ed25519::Pair::from_string(&full_address, None) {
592            Ok(a) => Ok(MultiSigner::Ed25519(a.public())),
593            Err(e) => Err(Error::SecretStringError(e)),
594        },
595        Encryption::Sr25519 => match sr25519::Pair::from_string(&full_address, None) {
596            Ok(a) => Ok(MultiSigner::Sr25519(a.public())),
597            Err(e) => Err(Error::SecretStringError(e)),
598        },
599        Encryption::Ecdsa | Encryption::Ethereum => {
600            match ecdsa::Pair::from_string(&full_address, None) {
601                Ok(a) => Ok(MultiSigner::Ecdsa(a.public())),
602                Err(e) => Err(Error::SecretStringError(e)),
603            }
604        }
605        Encryption::Penumbra => {
606            // penumbra uses bip44 derivation (m/44'/6532'/0'), not substrate-style paths
607            // this function is for substrate key derivation, penumbra keys are handled separately
608            Err(Error::PenumbraNotSubstrate)
609        }
610        Encryption::Zcash => {
611            // zcash uses ZIP-32 derivation (m/32'/133'/account'), not substrate-style paths
612            // this function is for substrate key derivation, zcash keys are handled separately
613            Err(Error::ZcashNotSubstrate)
614        }
615        Encryption::LedgerEd25519 => {
616            // Ledger uses SLIP-10/BIP32-Ed25519 derivation (m/44'/354'/account'/0'/0')
617            // not substrate-style paths - handled separately in create_address
618            Err(Error::LedgerNotSubstrate)
619        }
620        Encryption::Cosmos => {
621            // Cosmos uses BIP44 secp256k1 derivation (m/44'/118'/account'/0/0)
622            // not substrate-style paths - handled separately in create_address
623            Err(Error::CosmosNotSubstrate)
624        }
625        Encryption::Bitcoin => {
626            // Bitcoin uses BIP-84/86 derivation (m/84'/0'/account'/change/index)
627            // not substrate-style paths - handled separately in create_address
628            Err(Error::BitcoinNotSubstrate)
629        }
630        Encryption::Nostr => {
631            // Nostr uses NIP-06 derivation (m/44'/1237'/account'/0/0)
632            // not substrate-style paths - handled separately in create_address
633            Err(Error::NostrNotSubstrate)
634        }
635        Encryption::AtProtocol => {
636            // AT Protocol uses BIP-44 secp256k1 derivation (m/44'/29'/account'/0/0)
637            // not substrate-style paths - handled separately in create_address
638            Err(Error::AtProtoNotSubstrate)
639        }
640    };
641    full_address.zeroize();
642    multisigner_result
643}

657/// Return address details that corresponds to a public key. The root public keys are prioritized.
658/// Then network hashes passed as last param allows to choose one address
659/// in case multiple addresses available in different networks
660pub fn find_address_details_for_multisigner(
661    database: &sled::Db,
662    multisigner: &MultiSigner,
663    prioritizing_network_hashes: Vec<H256>,
664) -> Result<Option<AddressDetails>> {
665    let matching_addresses: Vec<AddressDetails> = get_all_addresses(database)?
666        .into_iter()
667        .filter(|(m, _)| m == multisigner)
668        .map(|(_, details)| details)
669        .collect();
670
671    let maybe_root_address = matching_addresses
672        .iter()
673        .find(|details| details.network_id.is_none());
674
675    if let Some(root_address) = maybe_root_address {
676        return Ok(Some(root_address.clone()));
677    }
678
679    if prioritizing_network_hashes.is_empty() {
680        return Ok(matching_addresses.first().cloned());
681    }
682
683    let mut indexed_addresses: HashMap<H256, &AddressDetails> = HashMap::new();
684
685    for addr in matching_addresses.iter() {
686        if let Some(Ok((hash, _))) = addr
687            .network_id
688            .as_ref()
689            .map(|id| id.genesis_hash_encryption())
690        {
691            _ = indexed_addresses.insert(hash, addr)
692        }
693    }
694
695    for genesis_hash in prioritizing_network_hashes {
696        if let Some(addr) = indexed_addresses.get(&genesis_hash) {
697            return Ok(Some((*addr).clone()));
698        }
699    }
700
701    Ok(matching_addresses.first().cloned())
702}

715pub fn process_dynamic_derivations_v1(
716    database: &sled::Db,
717    seeds: HashMap<String, String>,
718    request: DynamicDerivationsAddressRequestV1,
719) -> Result<DDPreview> {
720    let mut is_some_already_imported = false;
721    let mut is_some_network_missing = false;
722
723    let mut new_addrs: Vec<(AddressKey, AddressDetails)> = vec![];
724    let seed_request = &request.addr;
725    let seed_name = find_seed_name_for_multisigner(database, &seed_request.multisigner)?
726        .ok_or_else(|| Error::NoSeedForKeyPair {
727            multisigner: seed_request.multisigner.clone(),
728        })?;
729    let seed_phrase = seeds
730        .get(&seed_name)
731        .ok_or_else(|| Error::NoSeedForKeyPair {
732            multisigner: seed_request.multisigner.clone(),
733        })?;
734    let mut derivations = vec![];
735    for derivation_request in &seed_request.dynamic_derivations {
736        let network_specs_key = NetworkSpecsKey::from_parts(
737            &derivation_request.genesis_hash,
738            &derivation_request.encryption,
739        );
740        let network_specs = match get_network_specs(database, &network_specs_key) {
741            Ok(s) => s,
742            Err(_) => {
743                is_some_network_missing = true;
744                continue;
745            }
746        };
747        let multisigner = match create_address(
748            database,
749            &new_addrs,
750            &derivation_request.derivation_path,
751            Some(&network_specs.specs),
752            &seed_name,
753            seed_phrase,
754        ) {
755            // success
756            Ok(prep_data) => {
757                new_addrs = prep_data.address_prep;
758                new_addrs.last()
759                    .expect("new_addrs is never empty")
760                    .0
761                    .multi_signer()}
762            ,
763            // exactly same address already exists, ignoring it
764            Err(Error::DerivationExists { .. }) => {
765                is_some_already_imported = true;
766                continue;
767            },
768
769            // some other error, processed as a real error
770            Err(e) => return Err(e),
771        };
772        let encryption = derivation_request.encryption;
773        derivations.push(DDDetail {
774            base58: get_display_address(
775                database,
776                multisigner,
777                Some(network_specs.specs.base58prefix),
778                encryption,
779                Some(&network_specs.specs.name),
780            ),
781            path: derivation_request.derivation_path.clone(),
782            network_logo: network_specs.specs.logo,
783            network_specs_key: hex::encode(network_specs_key.key()),
784            identicon: make_identicon_from_multisigner(multisigner, encryption.identicon_style()),
785        })
786    }
787
788    let key_set = DDKeySet {
789        seed_name: seed_name.to_string(),
790        derivations,
791    };
792    let qr = dynamic_derivations_response_qr(&request, seed_phrase)?;
793    Ok(DDPreview {
794        qr,
795        key_set,
796        is_some_already_imported,
797        is_some_network_missing,
798    })
799}

811/// Prepare `DynamicDerivationsAddressResponse` for the given request
812pub fn dynamic_derivations_response(
813    request: &DynamicDerivationsAddressRequestV1,
814    seed_phrase: &str,
815) -> Result<DynamicDerivationsAddressResponse> {
816    let seed_request = &request.addr;
817    let mut derivations = vec![];
818    for derivation_request in &seed_request.dynamic_derivations {
819        let path = derivation_request.derivation_path.as_str();
820
821        // Skip non-substrate encryption types - they use their own derivation schemes
822        match derivation_request.encryption {
823            Encryption::Penumbra
824            | Encryption::Zcash
825            | Encryption::LedgerEd25519
826            | Encryption::Cosmos => {
827                // These encryption types don't support dynamic derivations via substrate paths
828                // Return an error for these - they should use their specific derivation endpoints
829                return Err(Error::Other(anyhow::anyhow!(
830                    "Dynamic derivations not supported for {:?} encryption. Use specific key derivation for this chain.",
831                    derivation_request.encryption
832                )));
833            }
834            _ => {}
835        }
836
837        // create fixed-length string to avoid reallocations
838        let mut full_address = String::with_capacity(seed_phrase.len() + path.len());
839        full_address.push_str(seed_phrase);
840        full_address.push_str(path);
841
842        derivations.push(DynamicDerivationResponseInfo {
843            derivation_path: derivation_request.derivation_path.clone(),
844            encryption: derivation_request.encryption,
845            public_key: full_address_to_multisigner(full_address, derivation_request.encryption)?,
846        });
847    }
848    let addr = DynamicDerivationsResponseInfo {
849        multisigner: seed_request.multisigner.clone(),
850        dynamic_derivations: derivations,
851    };
852
853    Ok(DynamicDerivationsAddressResponse::V1(
854        DynamicDerivationsAddressResponseV1 { addr },
855    ))
856}

859pub fn derive_single_key(
860    database: &sled::Db,
861    seeds: &HashMap<String, String>,
862    derivation_path: &str,
863    root_key_id: &[u8; 32],
864    network_key: NetworkSpecsKey,

896pub fn inject_derivations_has_pwd(
897    mut seed_derived_keys: Vec<SeedKeysPreview>,
898    seeds: HashMap<String, String>,
899) -> Result<Vec<SeedKeysPreview>> {
900    let mut sr25519_signers = HashMap::new();
901    let mut ed25519_signers = HashMap::new();
902    let mut ecdsa_signers = HashMap::new();
903
904    for (k, v) in &seeds {
905        let sr25519_public = sr25519::Pair::from_phrase(v, None).unwrap().0.public();
906        let ed25519_public = ed25519::Pair::from_phrase(v, None).unwrap().0.public();
907        let ecdsa_public = ecdsa::Pair::from_phrase(v, None).unwrap().0.public();
908        sr25519_signers.insert(sr25519_public, k);
909        ed25519_signers.insert(ed25519_public, k);
910        ecdsa_signers.insert(ecdsa_public, k);
911    }
912
913    for addr in seed_derived_keys.iter_mut() {
914        for derived_key in addr.derived_keys.iter_mut() {
915            let path = derived_key.derivation_path.clone().unwrap_or_default();
916            let seed_name = match addr.multisigner {
917                MultiSigner::Sr25519(p) => sr25519_signers.get(&p),
918                MultiSigner::Ed25519(p) => ed25519_signers.get(&p),
919                MultiSigner::Ecdsa(p) => ecdsa_signers.get(&p),
920            };
921
922            let seed_name = if let Some(seed_name) = seed_name {
923                seed_name
924            } else {
925                continue;
926            };
927
928            let seed_phrase = if let Some(seed_phrase) = seeds.get(seed_name.as_str()) {
929                seed_phrase
930            } else {
931                continue;
932            };
933
934            // Skip non-substrate encryption types - they don't support substrate-style password derivation
935            // Penumbra, Zcash, LedgerEd25519, and Cosmos use their own derivation schemes
936            match derived_key.encryption {
937                Encryption::Penumbra
938                | Encryption::Zcash
939                | Encryption::LedgerEd25519
940                | Encryption::Cosmos => {
941                    derived_key.has_pwd = Some(false);
942                    continue;
943                }
944                _ => {}
945            }
946
947            // create fixed-length string to avoid reallocation
948            let mut full_address = String::with_capacity(seed_phrase.len() + path.len());
949            full_address.push_str(seed_phrase);
950            full_address.push_str(&path);
951
952            let multisigner_pwdless =
953                full_address_to_multisigner(full_address, derived_key.encryption)?;
954            let multisigner =
955                base58_or_eth_pubkey_to_multisigner(&derived_key.address, &derived_key.encryption)?;
956            derived_key.has_pwd = Some(multisigner_pwdless != multisigner);
957        }
958    }
959    Ok(seed_derived_keys)
960}

1122/// Prepare data for address generation using seed phrase, derivation path, and
1123/// network specs.
1124///
1125/// Function generates addresses individually, and checks the database to
1126/// avoid collisions and duplicates. However, sometimes within a single database
1127/// transaction multiple addresses are generated, and each new address must be
1128/// checked with already semi-prepared new addresses set, to avoid collisions,
1129/// duplicates and overwriting.
1130///
1131/// Therefore, function inputs a set of `(AddressKey, AddressDetails)` values
1132/// already processed in preparation for the current database transaction and if
1133/// corresponding `AddressKey` is already there, adds new `NetworkSpecsKey` into
1134/// `AddressDetails` field `network_id` instead of creating a new entry.
1135///
1136/// The output [`PrepData`] contains:
1137///
1138/// - `address_prep` set of `(AddressKey, AddressDetails)` values that
1139///   eventually will be transformed into [`Batch`](sled::Batch) to update the
1140///   `ADDRTREE` of the cold database
1141/// - output `Event` set that will be eventually added into `HISTORY` as a part
1142///   of single `Entry`(definitions::history::Entry)
1143///
1144/// Function creates only **one** address, and output `PrepData` **updates**
1145/// address preparation set received as input.
1146///
1147/// Function gets used both on Active side (when generating test cold database
1148/// with well-known addresses) and on Vault side (when real addresses are
1149/// actually created by the user).
1150///
1151/// Secrets passed into this function are:
1152///
1153/// - seed phrase
1154/// - path, possibly with secret password part (`///<...>`).
1155///
1156/// Both are input as `&str`. Seed phrase is combined with derivation path into
1157/// a string, a reference to which is sent into inner logic of
1158/// [`sp_core::crypto`]. Combined secret string is then zeroized here regardless
1159/// of the address generation success.
1160#[cfg(feature = "active")]
1161pub(crate) fn create_address(
1162    database: &sled::Db,
1163    input_batch_prep: &[(AddressKey, AddressDetails)],
1164    path: &str,
1165    network_specs: Option<&NetworkSpecs>,
1166    seed_name: &str,
1167    seed_phrase: &str,
1168) -> Result<PrepData> {
1169    // Check that the seed phrase is not empty.
1170    // In upstream, empty seed phrase means default Alice seed phrase.
1171    if seed_phrase.is_empty() {
1172        return Err(Error::EmptySeed);
1173    }
1174
1175    let encryption = network_specs
1176        .map(|ns| ns.encryption)
1177        .unwrap_or(Encryption::Sr25519);
1178
1179    // Handle Penumbra separately - it uses BIP44 derivation, not Substrate-style paths
1180    let multisigner = if encryption == Encryption::Penumbra {
1181        #[cfg(feature = "penumbra")]
1182        {
1183            let (multisigner, bech32m_address) =
1184                penumbra_derive_multisigner_and_address(seed_phrase, path)?;
1185            // Store the bech32m address for later retrieval
1186            let ak_hex = hex::encode(multisigner_to_public(&multisigner));
1187            crate::penumbra::store_penumbra_address(database, &ak_hex, &bech32m_address)?;
1188            multisigner
1189        }
1190        #[cfg(not(feature = "penumbra"))]
1191        {
1192            return Err(Error::PenumbraNotSubstrate);
1193        }
1194    } else if encryption == Encryption::LedgerEd25519 {
1195        // Ledger Ed25519 uses SLIP-10/BIP32-Ed25519 derivation
1196        #[cfg(feature = "ledger")]
1197        {
1198            ledger_derive_multisigner(seed_phrase, path)?
1199        }
1200        #[cfg(not(feature = "ledger"))]
1201        {
1202            return Err(Error::LedgerNotSubstrate);
1203        }
1204    } else if encryption == Encryption::Cosmos {
1205        // Cosmos uses BIP44 secp256k1 derivation
1206        #[cfg(feature = "cosmos")]
1207        {
1208            let (multisigner, bech32_address) =
1209                cosmos_derive_multisigner_and_address(seed_phrase, path, network_specs)?;
1210            // Store the bech32 address for later retrieval, keyed by pubkey + genesis hash
1211            let pubkey_hex = hex::encode(multisigner_to_public(&multisigner));
1212            let genesis_hash_hex = network_specs
1213                .map(|ns| hex::encode(ns.genesis_hash))
1214                .unwrap_or_default();
1215            crate::cosmos::store_cosmos_address(
1216                database,
1217                &pubkey_hex,
1218                &genesis_hash_hex,
1219                &bech32_address,
1220            )?;
1221            multisigner
1222        }
1223        #[cfg(not(feature = "cosmos"))]
1224        {
1225            return Err(Error::CosmosNotSubstrate);
1226        }
1227    } else if encryption == Encryption::Zcash {
1228        // Zcash uses ZIP-32 Orchard derivation
1229        #[cfg(feature = "zcash")]
1230        {
1231            let (multisigner, unified_address) =
1232                zcash_derive_multisigner_and_address(seed_phrase, path, network_specs)?;
1233            // Store the unified address for later retrieval
1234            let pubkey_hex = hex::encode(multisigner_to_public(&multisigner));
1235            crate::zcash::store_zcash_address(database, &pubkey_hex, &unified_address)?;
1236            multisigner
1237        }
1238        #[cfg(not(feature = "zcash"))]
1239        {
1240            return Err(Error::ZcashNotSubstrate);
1241        }
1242    } else {
1243        // create fixed-length string to avoid reallocations
1244        let mut full_address = String::with_capacity(seed_phrase.len() + path.len());
1245        full_address.push_str(seed_phrase);
1246        full_address.push_str(path);
1247        full_address_to_multisigner(full_address, encryption)?
1248    };
1249
1250    // TODO regex elements may keep the line with password somewhere, how to
1251    // zeroize then? checked regex crate and it appears that only references are
1252    // moved around; need to double-check later;
1253    // BIP44-style paths (m/44'/...) don't match the Substrate regex, use them directly
1254    let (cropped_path, has_pwd) = if path.starts_with("m/") {
1255        (path, false)
1256    } else {
1257        match REG_PATH.captures(path) {
1258            Some(caps) => match caps.name("path") {
1259                Some(a) => (a.as_str(), caps.name("password").is_some()),
1260                None => ("", caps.name("password").is_some()),
1261            },
1262            None => ("", false),
1263        }
1264    };
1265
1266    do_create_address(
1267        database,
1268        input_batch_prep,
1269        cropped_path,
1270        network_specs,
1271        seed_name,
1272        multisigner,
1273        has_pwd,
1274    )
1275}

1300#[allow(clippy::too_many_arguments)]
1301fn do_create_address(
1302    database: &sled::Db,
1303    input_batch_prep: &[(AddressKey, AddressDetails)],
1304    cropped_path: &str,
1305    network_specs: Option<&NetworkSpecs>,
1306    seed_name: &str,
1307    multisigner: MultiSigner,
1308    has_pwd: bool,
1309) -> Result<PrepData> {
1310    // Check that the seed name is not empty.
1311    if seed_name.is_empty() {
1312        return Err(Error::EmptySeedName);
1313    }
1314    let mut address_prep = input_batch_prep.to_vec();
1315    let mut history_prep = vec![];
1316    let mut address_key: Option<AddressKey> = None;
1317    let mut network_specs_key: Option<NetworkSpecsKey> = None;
1318    if let Some(network_specs) = network_specs {
1319        network_specs_key = Some(NetworkSpecsKey::from_parts(
1320            &network_specs.genesis_hash,
1321            &network_specs.encryption,
1322        ));
1323
1324        let public_key = multisigner_to_public(&multisigner);
1325        address_key = Some(AddressKey::new(
1326            multisigner.clone(),
1327            Some(network_specs.genesis_hash),
1328        ));
1329
1330        // prepare history log here
1331        let identity_history = IdentityHistory::get(
1332            seed_name,
1333            &network_specs.encryption,
1334            &public_key,
1335            cropped_path,
1336            network_specs.genesis_hash,
1337        );
1338        history_prep.push(Event::IdentityAdded { identity_history });
1339    }
1340    if address_key.is_none() {
1341        address_key = Some(AddressKey::new(multisigner.clone(), None))
1342    }
1343
1344    // check if the same address key already participates in current database
1345    // transaction
1346
1347    // number in `(AddressKey, AddressDetails)` set with same `AddressKey`, if
1348    // found
1349    let number_in_current = None;
1350
1351    match number_in_current {
1352        // `AddressKey` already participates in transaction, just add
1353        // `NetworkSpecsKey` into corresponding `AddressDetails` `network_id`
1354        // set
1355        Some(i) => {
1356            let mod_entry = address_prep.remove(i);
1357            address_prep.push(mod_entry);
1358            Ok(PrepData {
1359                address_prep,
1360                history_prep,
1361            })
1362        }
1363
1364        // `AddressKey` is not yet in the transaction
1365        None => {
1366            // The only way to create address with `secret_exposed` marker is if
1367            // the parent with exposed secret is in the database.
1368            //
1369            // If the address key already exists in the database, the
1370            // `secret_exposed` flag will not be changed by extending key to
1371            // another network.
1372            let secret_exposed =
1373                has_parent_with_exposed_secret(cropped_path, has_pwd, seed_name, database)?;
1374
1375            // check if the `AddressKey` is already in the database
1376            let identities = open_tree(database, ADDRTREE)?;
1377            if let Some(address_key) = address_key {
1378                match identities.get(address_key.key()) {
1379                    // `AddressKey` is in the database
1380                    Ok(Some(address_entry)) => {
1381                        if let Some(network_specs_key) = network_specs_key {
1382                            let address_details = AddressDetails::from_entry_with_key_checked(
1383                                &address_key,
1384                                address_entry,
1385                            )?;
1386
1387                            Err(Error::DerivationExists {
1388                                multisigner,
1389                                address_details: Box::new(address_details),
1390                                network_specs_key,
1391                            })
1392                        } else {
1393                            Ok(PrepData {
1394                                address_prep,
1395                                history_prep,
1396                            })
1397                        }
1398                    }
1399
1400                    // `AddressKey` is not in the database either.
1401                    // Make altogether new entry.
1402                    Ok(None) => {
1403                        let address_details = AddressDetails {
1404                            seed_name: seed_name.to_string(),
1405                            path: cropped_path.to_string(),
1406                            has_pwd,
1407                            network_id: network_specs_key,
1408                            encryption: network_specs
1409                                .map(|ns| ns.encryption)
1410                                .unwrap_or(Encryption::Sr25519),
1411                            secret_exposed,
1412                        };
1413                        address_prep.push((address_key, address_details));
1414                        Ok(PrepData {
1415                            address_prep,
1416                            history_prep,
1417                        })
1418                    }
1419
1420                    // database error
1421                    Err(e) => Err(e.into()),
1422                }
1423            } else {
1424                Ok(PrepData {
1425                    address_prep,
1426                    history_prep,
1427                })
1428            }
1429        }
1430    }
1431}

1433/// Prepare data for default addresses generation for a seed.
1434///
1435/// Each network has a default derivation path, recorded in [`NetworkSpecs`]
1436/// field `path_id`, normally `path_id` is `//<network_name>`.
1437///
1438/// This function prepares the generation of addresses with default derivation
1439/// paths for each of the available networks. Additionally, if the
1440/// `make_seed_keys` flag is set to `true` the function also prepares seed key
1441/// generation.
1442///
1443/// The output [`PrepData`] contains:
1444///
1445/// - `address_prep` set of `(AddressKey, AddressDetails)` values that
1446///   eventually will be transformed into [`Batch`](sled::Batch) to update the
1447///   `ADDRTREE` of the cold database
1448/// - output `Event` set that will be eventually added into `HISTORY` as a part
1449///   of single `Entry`(definitions::history::Entry)
1450///
1451/// All output `PrepData` elements are generated within this function.
1452/// It does not input addresses previously processed for the same transaction
1453/// and therefore does not check for collisions, so this function always goes
1454/// first, when there are no address additions planned yet.
1455///
1456/// This function inputs secret seed phrase as `&str`. It is passed as `&str`
1457/// into `create_address` and used there.
1458#[cfg(feature = "active")]
1459fn populate_addresses(
1460    database: &sled::Db,
1461    seed_name: &str,
1462    seed_phrase: &str,
1463    make_seed_keys: bool,
1464) -> Result<PrepData> {
1465    // Set of `(AddressKey, AddressDetails)` to be added into the database.
1466    // Set is updated and is used on each iteration of `create_address` to check
1467    // for collisions.
1468    let mut address_prep: Vec<(AddressKey, AddressDetails)> = Vec::new();
1469
1470    // Set of associated `Event::IdentityAdded(_)` entries.
1471    // Gets updated after each successful `create_address` run.
1472    let mut history_prep: Vec<Event> = Vec::new();
1473
1474    // Make seed keys if requested.
1475    // Seed keys **must** be possible to generate,
1476    // if a seed key has a collision with some other key, it is an error
1477    if make_seed_keys {
1478        let prep_data = create_address(database, &address_prep, "", None, seed_name, seed_phrase)?;
1479        address_prep = prep_data.address_prep;
1480        history_prep.extend_from_slice(&prep_data.history_prep);
1481    }
1482    Ok(PrepData {
1483        address_prep,
1484        history_prep,
1485    })
1486}

1626/// Add a set of new derived addresses: N+1, N+2, etc into Vault database.
1627///
1628/// The number of the added addresses is set by `increment` input.
1629///
1630/// Input [`MultiSigner`] and [`NetworkSpecsKey`] determine the address, and the
1631/// associated [`AddressDetails`] are found in the database. The derivations are
1632/// based on the `path` from the `AddressDetails`.
1633///
1634/// If the database has addressed in given network with path that could be
1635/// interpreted as `<path>//<number>`, the first derivation of the increment set
1636/// is `<path>//<number>+1`. If no such addresses are existing in the database,
1637/// the first derivation of the increment set is `<path>//0`.
1638///
1639/// Further derivations of the increment set have number part increased by 1
1640/// until all requested derivations are generated.
1641///
1642/// Note: this is intended only for addresses not protected by the password
1643/// (i.e. having derivation path **without** `///<password>` part). If the
1644/// increment set is created on a passworded address, only public derivation
1645/// path part will be used for increment set generation.
1646///
1647/// This function inputs secret seed phrase as `&str`. It is passed as `&str`
1648/// into `create_address` and used there.
1649///
1650/// ## Example
1651///
1652/// User creates increment set of 3 on derivation `//user`. Resulting
1653/// derivations are:
1654///
1655/// - `//user`
1656/// - `//user//0`
1657/// - `//user//1`
1658/// - `//user//2`
1659///
1660/// User creates increment set of 2 on `//user` again, resulting derivations
1661/// are:
1662///
1663/// - `//user`
1664/// - `//user//0`
1665/// - `//user//1`
1666/// - `//user//2`
1667/// - `//user//3`
1668/// - `//user//4`
1669///
1670/// If the increment set of 2 is created on `//user//0`, resulting derivations
1671/// would be:
1672///
1673/// - `//user`
1674/// - `//user//0`
1675/// - `//user//0//0`
1676/// - `//user//0//1`
1677/// - `//user//1`
1678/// - `//user//2`
1679/// - `//user//3`
1680/// - `//user//4`
1681pub fn create_increment_set(
1682    database: &sled::Db,
1683    increment: u32,
1684    multisigner: &MultiSigner,
1685    network_specs_key: &NetworkSpecsKey,
1686    seed_phrase: &str,
1687) -> Result<()> {
1688    let network_specs = get_network_specs(database, network_specs_key)?;
1689    let address_details = get_address_details(
1690        database,
1691        &AddressKey::new(multisigner.clone(), Some(network_specs.specs.genesis_hash)),
1692    )?;
1693
1694    let existing_identities =
1695        addresses_set_seed_name_network(database, &address_details.seed_name, network_specs_key)?;
1696    let mut last_index = 0;
1697    for (_, details) in existing_identities.iter() {
1698        if let Some(("", suffix)) = details.path.split_once(&address_details.path) {
1699            if let Some(could_be_number) = suffix.get(2..) {
1700                if let Ok(index) = could_be_number.parse::<u32>() {
1701                    last_index = std::cmp::max(index + 1, last_index);
1702                }
1703            }
1704        }
1705    }
1706    let mut identity_adds: Vec<(AddressKey, AddressDetails)> = Vec::new();
1707    let mut current_events: Vec<Event> = Vec::new();
1708    for i in 0..increment {
1709        let path = address_details.path.to_string() + "//" + &(last_index + i).to_string();
1710        let prep_data = create_address(
1711            database,
1712            &identity_adds,
1713            &path,
1714            Some(&network_specs.specs),
1715            &address_details.seed_name,
1716            seed_phrase,
1717        )?;
1718        identity_adds = prep_data.address_prep;
1719        current_events.extend_from_slice(&prep_data.history_prep);
1720    }
1721    let id_batch = upd_id_batch(Batch::default(), identity_adds);
1722    TrDbCold::new()
1723        .set_addresses(id_batch) // add created addresses
1724        .set_history(events_to_batch(database, current_events)?) // add corresponding history
1725        .apply(database)
1726}

1729///
1730/// Invalid derivation produces an error. Valid derivation is checked for
1731/// password with regex.
1732///
1733// TODO regex and secrets, see `create_address` comments.

1743/// Proposed derivation status.
1744pub enum DerivationCheck {
1745    /// Derivation has bad format, UI disables proceeding with address
1746    /// generation.
1747    BadFormat,
1748
1749    /// Derivation has password (`///<password>` part).
1750    ///
1751    /// Dynamic check in the database for already existing address is disabled.
1752    Password,
1753
1754    /// Derivation has no password.
1755    ///
1756    /// Dynamic check in the database for already existing address is enabled.
1757    ///
1758    /// Inner content reflects if the already existing address is found and its
1759    /// details.
1760    NoPassword(Option<(MultiSigner, AddressDetails)>),
1761}

1763/// Preliminary checks if the user-proposed derivation could be used to create
1764/// the address.
1765///
1766/// This function **does not** use secret phrase, and only searches the
1767/// [`AddressDetails`] from the database for already existing addresses with
1768/// same derivation path, in case there is no password in the proposed
1769/// derivation.
1770///
1771/// If the proposed derivation is passworded, no checking could be done without
1772/// using the seed phrase, and UI allows to proceed to try creating new address.
1773/// If address already exists, an error would appear on the next stage.
1774///
1775/// No checking for collisions is done here (i.e. no checking if the same
1776/// [`AddressKey`] will be produced for <secret phrase 1> + <derivation path 1>
1777/// as for already known <secret phrase 2> + <derivation path 2>), because seed
1778/// phrase is not known. If collision occurs, an error would appear on the next
1779/// stage.
1780///
1781/// Note that the only errors in this function could appear from database
1782/// malfunction when the database is being searched for the exact match, in case
1783/// of password-free valid derivation. Bad format of the derivation is **not**
1784/// an error, UI just does not allow to proceed.
1785pub fn derivation_check(
1786    database: &sled::Db,
1787    seed_name: &str,
1788    path: &str,
1789    network_specs_key: &NetworkSpecsKey,
1790) -> Result<DerivationCheck> {
1791    // BIP44/ZIP-32 paths (zcash `m/32'/133'/account'`, penumbra
1792    // `m/44'/6532'/0'`, cosmos, ...) are not substrate junctions and carry no
1793    // password. The substrate-only `is_passworded`/`REG_PATH` rejects them,
1794    // which used to surface as `Error::InvalidDerivation` from
1795    // `try_create_address`. Mirror `create_address`'s `path.starts_with("m/")`
1796    // handling: treat them as valid, unpassworded paths and only run the
1797    // exact-collision check.
1798    let passworded = if path.starts_with("m/") {
1799        false
1800    } else {
1801        match is_passworded(path) {
1802            Ok(p) => p,
1803            // Proposed derivation is not suitable, UI would not allow to
1804            // proceed. Note that this is **not** an error.
1805            Err(_) => return Ok(DerivationCheck::BadFormat),
1806        }
1807    };
1808    match passworded {
1809        // Proposed derivation has password, no checks could be made, proceed.
1810        true => Ok(DerivationCheck::Password),
1811
1812        // Proposed derivation has no password, checking the database for exact
1813        // coincidence.
1814        false => {
1815            let mut found_exact = None;
1816            for (multisigner, address_details) in get_all_addresses(database)?.into_iter() {
1817                if (address_details.seed_name == seed_name) // seed name
1818                    && (address_details.path == path) // derivation path, cropped part without password
1819                    && (address_details.network_id.as_ref() == Some(network_specs_key)) // in this network
1820                    && (!address_details.has_pwd)
1821                // has no password to begin with
1822                {
1823                    found_exact = Some((multisigner, address_details));
1824                    break;
1825                }
1826            }
1827            Ok(DerivationCheck::NoPassword(found_exact))
1828        }
1829    }
1830}

1832/// Separate derivation path with password into non-secret part and password.
1833///
1834/// Function is expected to deal **only** with derivations having a password
1835/// (i.e. with `///<password>` part of the derivation path). Password-free
1836/// derivations do not get to the point in `navigator` where this function is
1837/// called. Calling this function on password-free derivation is an error.
1838///
1839/// Function receives derivation path with secret password part, and outputs
1840/// hard and soft derivations part and secret password part of the derivation
1841/// path separately.
1842///
1843// TODO regex and secrets, see `create_address` comments.
1844pub fn cut_path(path: &str) -> Result<(String, String)> {
1845    let caps = REG_PATH
1846        .captures(path)
1847        .ok_or_else(|| Error::InvalidDerivation(path.to_string()))?;
1848    let cropped_path = caps
1849        .name("path")
1850        .map(|a| a.as_str().to_string())
1851        .ok_or_else(|| Error::InvalidDerivation(path.to_string()))?;
1852    match caps.name("password") {
1853        Some(pwd) => Ok((cropped_path, pwd.as_str().to_string())),
1854        None => Err(Error::LostPwd),
1855    }
1856}

1858/// Create a new address in the Vault database.
1859///
1860/// If the associated [`AddressKey`] already exists in the [`ADDRTREE`] tree of
1861/// the Vault database, a new network [`NetworkSpecsKey`] is added into
1862/// `network_id` set of the `AddressDetails`. If the `AddressKey` is new,
1863/// a new `ADDRTREE` entry is made.
1864///
1865/// Secrets passed into this function are:
1866///
1867/// - seed phrase
1868/// - path, possibly with secret password part (`///<...>`).
1869///
1870/// Both are input as `&str` and go directly in `create_address` (both) and
1871/// `derivation_check` (path only).
1872pub fn try_create_address(
1873    database: &sled::Db,
1874    seed_name: &str,
1875    seed_phrase: &str,
1876    path: &str,
1877    network_specs_key: &NetworkSpecsKey,
1878) -> Result<()> {
1879    match derivation_check(database, seed_name, path, network_specs_key)? {
1880        // UI should prevent user from getting into `try_create_address` if
1881        // derivation has a bad format
1882        DerivationCheck::BadFormat => Err(Error::InvalidDerivation(path.to_string())),
1883
1884        // UI should prevent user from getting into `try_create_address` if
1885        // derivation already exists
1886        DerivationCheck::NoPassword(Some((multisigner, address_details))) => {
1887            Err(Error::DerivationExists {
1888                multisigner,
1889                address_details: Box::new(address_details),
1890                network_specs_key: network_specs_key.to_owned(),
1891            })
1892        }
1893
1894        // Valid derivation:
1895        // - without a password and not in the database
1896        // - with password
1897        // Vault can try adding either. Collisions (for both cases) and exact
1898        // coincidence (for passworded derivation) could not have been checked
1899        // preliminarily and would result in `create_address` errors here.
1900        _ => {
1901            let network_specs = get_network_specs(database, network_specs_key)?;
1902            let prep_data = create_address(
1903                database,
1904                &Vec::new(), // a single address is created, no data to check against here
1905                path,
1906                Some(&network_specs.specs),
1907                seed_name,
1908                seed_phrase,
1909            )?;
1910            let id_batch = upd_id_batch(Batch::default(), prep_data.address_prep);
1911            TrDbCold::new()
1912                .set_addresses(id_batch) // add created address
1913                .set_history(events_to_batch(database, prep_data.history_prep)?) // add corresponding history
1914                .apply(database)
1915        }
1916    }
1917}

1971/// Remove all addresses associated with given seed name from the Vault
1972/// database.
1973///
1974/// Complementary action in frontend is removal of the seed data from the device
1975/// key management system.
1976pub fn remove_seed(database: &sled::Db, seed_name: &str) -> Result<()> {
1977    // `Batch` to use
1978    let mut identity_batch = Batch::default();
1979
1980    // All addresses with given seed name from the database
1981    let id_set = get_addresses_by_seed_name(database, seed_name)?;
1982    if id_set.is_empty() {
1983        return Err(Error::NoKnownSeeds);
1984    }
1985
1986    // Associated `Event` set
1987    let mut events = vec![Event::SeedRemoved {
1988        seed_name: seed_name.to_owned(),
1989    }];
1990
1991    for (multisigner, address_details) in id_set.iter() {
1992        if let Some(id) = &address_details.network_id {
1993            let network_specs = get_network_specs(database, id)?;
1994            let address_key =
1995                AddressKey::new(multisigner.clone(), Some(network_specs.specs.genesis_hash));
1996
1997            // removal of all addresses corresponging to `AddressKey`
1998            identity_batch.remove(address_key.key());
1999
2000            let public_key = multisigner_to_public(multisigner);
2001            let (genesis_hash_vec, _) = id.genesis_hash_encryption()?;
2002            let identity_history = IdentityHistory::get(
2003                seed_name,
2004                &address_details.encryption,
2005                &public_key,
2006                &address_details.path,
2007                genesis_hash_vec,
2008            );
2009            // separate `Event` for each `NetworkSpecsKey` from `network_id` set
2010            events.push(Event::IdentityRemoved { identity_history });
2011        } else {
2012            let address_key = AddressKey::new(multisigner.clone(), None);
2013            identity_batch.remove(address_key.key());
2014        }
2015    }
2016    TrDbCold::new()
2017        .set_addresses(identity_batch) // modify addresses
2018        .set_history(events_to_batch(database, events)?) // add corresponding history
2019        .apply(database)
2020}

2022/// Check derivations before offering user to import them.
2023///
2024/// Vault allows bulk import only for valid derivations, with or without
2025/// password. For passworded derivations no password is ever confirmed, as the
2026/// user presumably has the derivations already recorded somewhere.
2027///
2028/// This function is used in `transaction_parsing::derivations` to check the
2029/// incoming derivations once they are received in `derivations` qr code
2030/// payload.
2031///
2032/// Note that once the derivations are checked, they are moved around as
2033/// "checked derivations" and not checked anymore until they are used in
2034/// `import_derivations`. Only checked derivations are written into the
2035/// database temporary storage in `TRANSACTION` tree, same data is recovered as
2036/// ensured by the checksum matching.
2037pub fn check_derivation_set(derivations: &[String]) -> Result<()> {
2038    for path in derivations.iter() {
2039        if REG_PATH.captures(path).is_none() {
2040            return Err(Error::InvalidDerivation(path.to_string()));
2041        }
2042    }
2043    Ok(())
2044}

2046fn prepare_secret_key_for_export(
2047    multisigner: &MultiSigner,
2048    full_address: &str,
2049    pwd: Option<&str>,
2050) -> Result<[u8; 32]> {
2051    match multisigner {
2052        MultiSigner::Ed25519(public) => {
2053            let ed25519_pair =
2054                ed25519::Pair::from_string(full_address, pwd).map_err(Error::SecretStringError)?;
2055            if public != &ed25519_pair.public() {
2056                return Err(Error::WrongPassword);
2057            }
2058            Ok(ed25519_pair.seed().to_owned())
2059        }
2060        MultiSigner::Sr25519(public) => {
2061            let (sr25519_pair, seed) = sr25519::Pair::from_string_with_seed(full_address, pwd)
2062                .map_err(Error::SecretStringError)?;
2063            if public != &sr25519_pair.public() {
2064                return Err(Error::WrongPassword);
2065            }
2066            Ok(seed.ok_or_else(|| Error::NoSeedForKeyPair {
2067                multisigner: multisigner.clone(),
2068            })?)
2069        }
2070        MultiSigner::Ecdsa(public) => {
2071            let ecdsa_pair =
2072                ecdsa::Pair::from_string(full_address, pwd).map_err(Error::SecretStringError)?;
2073            if public != &ecdsa_pair.public() {
2074                return Err(Error::WrongPassword);
2075            }
2076            Ok(ecdsa_pair.seed())
2077        }
2078    }
2079}

2207fn generate_secret_qr(
2208    multisigner: &MultiSigner,
2209    address_details: &AddressDetails,
2210    genesis_hash: &H256,
2211    seed_phrase: &str,
2212    pwd: Option<&str>,
2213) -> Result<QrData> {
2214    // create fixed-length string to avoid reallocations
2215
2216    let mut full_address = String::with_capacity(seed_phrase.len() + address_details.path.len());
2217    full_address.push_str(seed_phrase);
2218    full_address.push_str(&address_details.path);
2219
2220    let mut secret = match prepare_secret_key_for_export(multisigner, &full_address, pwd) {
2221        Ok(a) => {
2222            full_address.zeroize();
2223            a
2224        }
2225        Err(e) => {
2226            full_address.zeroize();
2227            return Err(e);
2228        }
2229    };
2230
2231    let qr = QrData::Sensitive {
2232        data: format!(
2233            "secret:0x{}:{}",
2234            hex::encode(secret),
2235            hex::encode(genesis_hash)
2236        )
2237        .as_bytes()
2238        .to_vec(),
2239    };
2240    secret.zeroize();
2241    Ok(qr)
2242}
