1//! Cosmos SDK key derivation using BIP44 secp256k1
2//!
3//! This implements key derivation for Cosmos SDK chains (Osmosis, Noble, Celestia, etc.)
4//!
5//! Key derivation path: m/44'/coin_type'/account'/0/address_index
6//! For Cosmos Hub (ATOM): m/44'/118'/0'/0/0
7//!
8//! Address derivation:
9//! 1. secp256k1 public key (33 bytes compressed)
10//! 2. SHA256(pubkey) → 32 bytes
11//! 3. RIPEMD160(sha256_hash) → 20 bytes
12//! 4. bech32 encode with chain-specific prefix (cosmos, osmo, noble, celestia, etc.)
13
14use crate::error::{Error, Result};
15
16/// Sled tree name for storing Cosmos bech32 addresses
17const COSMOS_ADDRS: &str = "cosmos_addresses";
18
19/// Store a Cosmos bech32 address for a given public key hex and genesis hash
20pub fn store_cosmos_address(
21    database: &sled::Db,
22    pubkey_hex: &str,
23    genesis_hash: &str,
24    bech32_address: &str,
25) -> Result<()> {
26    let tree = database.open_tree(COSMOS_ADDRS)?;
27    let key = format!("{}_{}", pubkey_hex, genesis_hash);
28    tree.insert(key.as_bytes(), bech32_address.as_bytes())?;
29    Ok(())
30}

32/// Retrieve a Cosmos bech32 address for a given public key hex and genesis hash
33pub fn get_cosmos_address(
34    database: &sled::Db,
35    pubkey_hex: &str,
36    genesis_hash: &str,
37) -> Result<Option<String>> {
38    let tree = database.open_tree(COSMOS_ADDRS)?;
39    let key = format!("{}_{}", pubkey_hex, genesis_hash);
40    match tree.get(key.as_bytes())? {
41        Some(bytes) => {
42            let address = String::from_utf8(bytes.to_vec())
43                .map_err(|e| Error::Other(anyhow::anyhow!("Invalid UTF-8: {}", e)))?;
44            Ok(Some(address))
45        }
46        None => Ok(None),
47    }
48}

54/// Compute bech32 cosmos address from compressed pubkey bytes and network name.
55/// Same key for all cosmos chains (coin type 118), just different bech32 prefix.
56/// This follows how zafu differentiates addresses: `from_bech32` → `to_bech32` with chain prefix.
57pub fn pubkey_to_bech32_address(pubkey_bytes: &[u8], network_name: &str) -> Result<String> {
58    use bech32::{Bech32, Hrp};
59
60    if pubkey_bytes.len() != 33 {
61        return Err(Error::Other(anyhow::anyhow!(
62            "expected 33-byte compressed secp256k1 pubkey, got {} bytes",
63            pubkey_bytes.len()
64        )));
65    }
66
67    // Map network name to bech32 prefix
68    let prefix = match network_name {
69        "osmosis" => PREFIX_OSMOSIS,
70        "noble" => PREFIX_NOBLE,
71        "celestia" => PREFIX_CELESTIA,
72        "terra" => PREFIX_TERRA,
73        "kava" => PREFIX_KAVA,
74        "secret" => PREFIX_SECRET,
75        "injective" => PREFIX_INJECTIVE,
76        _ => PREFIX_COSMOS,
77    };
78
79    // RIPEMD160(SHA256(pubkey)) → 20 bytes address
80    let sha256_hash = Sha256::digest(pubkey_bytes);
81    let ripemd160_hash = Ripemd160::digest(sha256_hash);
82
83    let hrp = Hrp::parse(prefix)
84        .map_err(|e| Error::Other(anyhow::anyhow!("Invalid bech32 prefix: {}", e)))?;
85    let address = bech32::encode::<Bech32>(hrp, &ripemd160_hash)
86        .map_err(|e| Error::Other(anyhow::anyhow!("Bech32 encoding error: {}", e)))?;
87
88    Ok(address)
89}

111/// Cosmos key pair with secp256k1 keys
112#[derive(Zeroize, ZeroizeOnDrop)]
113pub struct CosmosKeyPair {
114    /// 32-byte secp256k1 secret key
115    pub secret_key: [u8; 32],
116    /// 33-byte compressed secp256k1 public key
117    pub public_key: [u8; 33],
118    /// 20-byte address (RIPEMD160(SHA256(pubkey)))
119    pub address_bytes: [u8; 20],
120}
121
122impl CosmosKeyPair {
123    /// Get bech32 address with the specified prefix
124    pub fn bech32_address(&self, prefix: &str) -> Result<String> {
125        use bech32::{Bech32, Hrp};
126
127        let hrp = Hrp::parse(prefix)
128            .map_err(|e| Error::Other(anyhow::anyhow!("Invalid bech32 prefix: {}", e)))?;
129
130        let address = bech32::encode::<Bech32>(hrp, &self.address_bytes)
131            .map_err(|e| Error::Other(anyhow::anyhow!("Bech32 encoding error: {}", e)))?;
132
133        Ok(address)
134    }
135
136    // NOTE: signing is done in transaction_signing::cosmos::sign_cosmos_amino()
137    // which correctly uses SHA256 pre-hash. Do NOT add a sign() method here
138    // using sp_core::ecdsa — it uses blake2b-256, which is wrong for Cosmos.
139}

141/// Derive Cosmos key from seed phrase
142///
143/// # Arguments
144/// * `seed_phrase` - BIP39 mnemonic phrase
145/// * `coin_type` - SLIP-0044 coin type (118 for Cosmos, 330 for Terra, etc.)
146/// * `account` - Account index (typically 0)
147/// * `address_index` - Address index within the account (typically 0)
148///
149/// # Returns
150/// A CosmosKeyPair containing the secret key, public key, and address bytes
151pub fn derive_cosmos_key(
152    seed_phrase: &str,
153    coin_type: u32,
154    account: u32,
155    address_index: u32,
156) -> Result<CosmosKeyPair> {
157    use bip39::{Language, Mnemonic, Seed};
158
159    // Parse mnemonic and generate seed
160    let mnemonic = Mnemonic::from_phrase(seed_phrase, Language::English)
161        .map_err(|e| Error::Other(anyhow::anyhow!("Invalid mnemonic: {}", e)))?;
162    let seed = Seed::new(&mnemonic, ""); // No passphrase
163
164    // Build derivation path: m/44'/coin_type'/account'/0/address_index
165    let path_str = format!("m/44'/{}'/{}'/0/{}", coin_type, account, address_index);
166    let path: DerivationPath = path_str
167        .parse()
168        .map_err(|e| Error::Other(anyhow::anyhow!("Invalid derivation path: {}", e)))?;
169
170    // Derive extended private key
171    let xprv = XPrv::derive_from_path(seed.as_bytes(), &path)
172        .map_err(|e| Error::Other(anyhow::anyhow!("BIP32 derivation error: {}", e)))?;
173
174    // Get the private key bytes
175    let mut secret_key = [0u8; 32];
176    secret_key.copy_from_slice(&xprv.private_key().to_bytes());
177
178    // Derive public key using the bip32 crate's key
179    let verifying_key = xprv.public_key();
180    let public_key_bytes = verifying_key.to_bytes();
181
182    let mut public_key = [0u8; 33];
183    public_key.copy_from_slice(&public_key_bytes);
184
185    // Derive address: RIPEMD160(SHA256(pubkey))
186    let sha256_hash = Sha256::digest(public_key);
187    let ripemd160_hash = Ripemd160::digest(sha256_hash);
188
189    let mut address_bytes = [0u8; 20];
190    address_bytes.copy_from_slice(&ripemd160_hash);
191
192    Ok(CosmosKeyPair {
193        secret_key,
194        public_key,
195        address_bytes,
196    })
197}

199/// Derive the account's change-level extended public key (xpub) at
200/// `m/44'/coin_type'/account'/0`.
201///
202/// This is the parent of the address-index leaves, so a watch-only holder can
203/// derive `m/44'/coin_type'/account'/0/i` for any `i` via non-hardened CKD
204/// WITHOUT the seed. That is the whole point: one cold export lets the hot
205/// wallet mint unlimited burner receive addresses, each still signable on the
206/// device at the matching `address_index`, and each recoverable from the seed.
207///
208/// Serialized as a standard base58 `xpub...` string (mainnet public prefix), so
209/// any BIP32 library can consume it.
210pub fn derive_cosmos_account_xpub(
211    seed_phrase: &str,
212    coin_type: u32,
213    account: u32,
214) -> Result<String> {
215    use bip32::Prefix;
216    use bip39::{Language, Mnemonic, Seed};
217
218    let mnemonic = Mnemonic::from_phrase(seed_phrase, Language::English)
219        .map_err(|e| Error::Other(anyhow::anyhow!("Invalid mnemonic: {}", e)))?;
220    let seed = Seed::new(&mnemonic, "");
221
222    // Change level: m/44'/coin_type'/account'/0. Stop above address_index so the
223    // returned xpub can derive every leaf non-hardened.
224    let path_str = format!("m/44'/{}'/{}'/0", coin_type, account);
225    let path: DerivationPath = path_str
226        .parse()
227        .map_err(|e| Error::Other(anyhow::anyhow!("Invalid derivation path: {}", e)))?;
228
229    let xprv = XPrv::derive_from_path(seed.as_bytes(), &path)
230        .map_err(|e| Error::Other(anyhow::anyhow!("BIP32 derivation error: {}", e)))?;
231
232    Ok(xprv.public_key().to_string(Prefix::XPUB))
233}

255/// Parse Cosmos BIP44 path to extract coin type, account, and address index
256///
257/// Supports paths like:
258/// - `m/44'/118'/0'/0/0` - standard full path
259/// - `//cosmos//0` - simplified UI input (uses coin type 118)
260/// - `//osmosis//0` - simplified UI input for Osmosis
261/// - `//0` - just account number (uses coin type 118, index 0)
262pub fn parse_cosmos_path(path: &str) -> Result<(u32, u32, u32)> {
263    // Try parsing as full BIP44 path: m/44'/coin_type'/account'/0/index
264    if path.starts_with("m/44'/") || path.starts_with("44'/") {
265        let parts: Vec<&str> = path.trim_start_matches("m/").split('/').collect();
266        if parts.len() >= 5 {
267            // parts[0] = "44'" (purpose)
268            // parts[1] = "118'" (coin type)
269            // parts[2] = "0'" (account)
270            // parts[3] = "0" (change, always 0 for Cosmos)
271            // parts[4] = "0" (address index)
272            let coin_type_str = parts[1].trim_end_matches('\'');
273            let account_str = parts[2].trim_end_matches('\'');
274            let index_str = parts[4].trim_end_matches('\'');
275
276            let coin_type = coin_type_str
277                .parse::<u32>()
278                .map_err(|_| Error::InvalidDerivation(path.to_string()))?;
279            let account = account_str
280                .parse::<u32>()
281                .map_err(|_| Error::InvalidDerivation(path.to_string()))?;
282            let index = index_str
283                .parse::<u32>()
284                .map_err(|_| Error::InvalidDerivation(path.to_string()))?;
285
286            return Ok((coin_type, account, index));
287        }
288    }
289
290    let path_lower = path.to_lowercase();
291    let parts: Vec<&str> = path.split("//").filter(|s| !s.is_empty()).collect();
292
293    // Extract numeric values from path
294    let account = parts
295        .iter()
296        .filter_map(|p| p.parse::<u32>().ok())
297        .next()
298        .unwrap_or(0);
299
300    // Determine coin type from chain name in path
301    let coin_type = if path_lower.contains("terra") {
302        SLIP0044_TERRA
303    } else if path_lower.contains("kava") {
304        SLIP0044_KAVA
305    } else if path_lower.contains("secret") {
306        SLIP0044_SECRET
307    } else if path_lower.contains("injective") || path_lower.contains("inj") {
308        SLIP0044_INJECTIVE
309    } else {
310        // Default to Cosmos coin type (118) for cosmos, osmo, noble, celestia, etc.
311        SLIP0044_COSMOS
312    };
313
314    Ok((coin_type, account, 0))
315}

421    #[test]
422    fn test_xpub_derives_same_children_as_seed() {
423        // The watch-only contract: a burner derived from the exported xpub at
424        // .../0/i must be byte-identical to what the device signs with at the
425        // same i. If this fails, the hot wallet shows an address the device can
426        // never spend from - and post-store-release that is unfixable.
427        use bip32::{ChildNumber, XPub};
428
429        let mnemonic = "abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon about";
430        let xpub_str = derive_cosmos_account_xpub(mnemonic, SLIP0044_COSMOS, 0).unwrap();
431        assert!(
432            xpub_str.starts_with("xpub"),
433            "expected base58 xpub, got {xpub_str}"
434        );
435
436        let xpub: XPub = xpub_str.parse().expect("re-parse exported xpub");
437        for i in 0..5u32 {
438            let child = xpub
439                .derive_child(ChildNumber::new(i, false).unwrap())
440                .unwrap();
441            let seed_key = derive_cosmos_key(mnemonic, SLIP0044_COSMOS, 0, i).unwrap();
442            assert_eq!(
443                child.to_bytes().as_slice(),
444                &seed_key.public_key[..],
445                "xpub child {i} must equal the seed-derived leaf",
446            );
447        }
448    }
