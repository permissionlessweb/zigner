1//! PRF expansion functions for Penumbra key derivation
2//! Based on ledger-penumbra implementation
3
4use crate::error::{Error, Result};
5
6/// Blake2b PRF expansion
7pub fn expand(label: &[u8; 16], key: &[u8], input: &[u8]) -> Result<[u8; 64]> {
8    use blake2b_simd::Params;
9
10    if key.len() > blake2b_simd::KEYBYTES {
11        return Err(Error::Other(anyhow::anyhow!("Invalid key length for PRF")));
12    }
13
14    let mut params = Params::new();
15    params.personal(label);
16
17    // Add key only if not empty
18    if !key.is_empty() {
19        params.key(key);
20    }
21
22    let hash = params.hash(input);
23    let mut output = [0u8; 64];
24    output.copy_from_slice(hash.as_bytes());
25
26    Ok(output)
27}
28
29/// Expand to Fr (scalar field element)
30pub fn expand_fr(label: &[u8; 16], key: &[u8], input: &[u8]) -> Result<decaf377::Fr> {
31    let bytes = expand(label, key, input)?;
32    Ok(decaf377::Fr::from_le_bytes_mod_order(&bytes))
33}
34
35/// Expand to Fq (base field element)
36pub fn expand_fq(label: &[u8; 16], key: &[u8], input: &[u8]) -> Result<decaf377::Fq> {
37    let bytes = expand(label, key, input)?;
38    Ok(decaf377::Fq::from_le_bytes_mod_order(&bytes))
