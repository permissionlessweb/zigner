1//! Ledger Ed25519 key derivation using Cardano-style BIP32-Ed25519
2//!
3//! This implements the same key derivation used by Ledger hardware wallets
4//! for Polkadot and other Substrate-based chains.
5//!
6//! Key derivation path: m/44'/slip0044'/account'/change'/address'
7//! For Polkadot: m/44'/354'/0'/0'/0'
8//!
9//! Note: Ledger uses Cardano-style BIP32-Ed25519, NOT SLIP-10.
10//! The difference is in the master key generation which iterates
11//! until a valid key is found.
12
13use crate::error::{Error, Result};
14use hmac::{Hmac, Mac};
15use sha2::{Sha256, Sha512};
16use sp_core::Pair;
17use zeroize::{Zeroize, ZeroizeOnDrop};

19/// SLIP-0044 coin types
20pub const SLIP0044_POLKADOT: u32 = 354;
21pub const SLIP0044_KUSAMA: u32 = 434;

23/// Hardened derivation flag
24const HARDENED: u32 = 0x80000000;

26/// Generate Ledger master key from seed using Cardano-style derivation
27/// Returns `(extended_secret[64], chain_code[32])` = 96 bytes total
28/// This matches the polkadot-js ledgerMaster implementation
29fn ledger_master(seed: &[u8]) -> ([u8; 64], [u8; 32]) {
30    // Chain code: HMAC-SHA256 with key "ed25519 seed", data = [1, ...seed]
31    // Note: polkadot-js uses 256 bits (SHA256), not 512!
32    let mut cc_hmac =
33        Hmac::<Sha256>::new_from_slice(b"ed25519 seed").expect("HMAC can take key of any size");
34    let mut cc_data = vec![1u8];
35    cc_data.extend_from_slice(seed);
36    cc_hmac.update(&cc_data);
37    let cc_result = cc_hmac.finalize().into_bytes();
38
39    let mut chain_code = [0u8; 32];
40    chain_code.copy_from_slice(&cc_result);
41
42    // Private key: iterate HMAC-SHA512 until (priv[31] & 0x20) == 0
43    let mut priv_data: Option<Vec<u8>> = None;
44    let mut priv_key: [u8; 64];
45
46    loop {
47        let mut hmac =
48            Hmac::<Sha512>::new_from_slice(b"ed25519 seed").expect("HMAC can take key of any size");
49
50        match &priv_data {
51            Some(data) => hmac.update(data),
52            None => hmac.update(seed),
53        }
54
55        let result = hmac.finalize().into_bytes();
56        priv_key = result.into();
57
58        // Cardano validity check: third highest bit of byte 31 must be zero
59        if (priv_key[31] & 0b0010_0000) == 0 {
60            // Clamp the private key (Cardano-style)
61            priv_key[0] &= 0b1111_1000; // Clear lowest 3 bits
62            priv_key[31] &= 0b0111_1111; // Clear bit 255
63            priv_key[31] |= 0b0100_0000; // Set bit 254
64
65            return (priv_key, chain_code);
66        }
67
68        // If not valid, hash again with the result
69        priv_data = Some(result.to_vec());
70    }
71}

73/// Full Ledger key derivation from mnemonic
74/// Path: m/44'/slip0044'/account'/change'/address'
75pub fn derive_ledger_key(
76    seed_phrase: &str,
77    slip0044: u32,
78    account: u32,
79    change: u32,
80    address_index: u32,
81) -> Result<LedgerKeyPair> {
82    use bip39::{Language, Mnemonic, Seed};
83
84    // Parse mnemonic and generate seed
85    let mnemonic = Mnemonic::from_phrase(seed_phrase, Language::English)
86        .map_err(|e| Error::Other(anyhow::anyhow!("Invalid mnemonic: {}", e)))?;
87    let seed = Seed::new(&mnemonic, ""); // No passphrase
88
89    // Generate master key using Cardano-style derivation (96 bytes: kL[32] || kR[32] || cc[32])
90    let (extended_secret, chain_code) = ledger_master(seed.as_bytes());
91
92    // Derive the full path: m/44'/slip0044'/account'/change'/address'
93    let path = [
94        44 | HARDENED,
95        slip0044 | HARDENED,
96        account | HARDENED,
97        change | HARDENED,
98        address_index | HARDENED,
99    ];
100
101    let (final_extended, _final_cc) = ledger_derive_path(&extended_secret, &chain_code, &path);
102
103    // For Ed25519, the secret key is kL (first 32 bytes)
104    // This is used as an ed25519 SEED, which gets hashed internally
105    let mut secret_key = [0u8; 32];
106    secret_key.copy_from_slice(&final_extended[0..32]);
107
108    // Derive public key using sp_core::ed25519 (already a dependency)
109    // This matches polkadot-js ed25519PairFromSeed behavior
110    let pair = sp_core::ed25519::Pair::from_seed_slice(&secret_key)
111        .map_err(|e| Error::Other(anyhow::anyhow!("Ed25519 seed error: {:?}", e)))?;
112    let public_key = pair.public().0;
113
114    Ok(LedgerKeyPair {
115        extended_key: final_extended,
116        secret_key,
117        public_key,
118    })
119}

121/// Derive a child key along the path (hardened only)
122/// This implements the polkadot-js ledgerDerivePrivate
123fn ledger_derive_path(
124    extended_secret: &[u8; 64],
125    chain_code: &[u8; 32],
126    path: &[u32],
127) -> ([u8; 64], [u8; 32]) {
128    let mut current_key = *extended_secret;
129    let mut current_cc = *chain_code;
130
131    for &index in path {
132        (current_key, current_cc) = ledger_derive_private(&current_key, &current_cc, index);
133    }
134
135    (current_key, current_cc)
136}

138/// Derive a single child key (hard derivation only)
139/// Based on polkadot-js ledgerDerivePrivate implementation
140fn ledger_derive_private(
141    extended_secret: &[u8; 64],
142    chain_code: &[u8; 32],
143    index: u32,
144) -> ([u8; 64], [u8; 32]) {
145    let kl = &extended_secret[0..32];
146    let kr = &extended_secret[32..64];
147
148    // First HMAC: data = [0x00, kL, kR, index_le]
149    let mut data1 = vec![0x00];
150    data1.extend_from_slice(kl);
151    data1.extend_from_slice(kr);
152    data1.extend_from_slice(&index.to_le_bytes());
153
154    let mut hmac1 =
155        Hmac::<Sha512>::new_from_slice(chain_code).expect("HMAC can take key of any size");
156    hmac1.update(&data1);
157    let z = hmac1.finalize().into_bytes();
158
159    // Second HMAC for new chain code: data = [0x01, kL, kR, index_le]
160    let mut data2 = vec![0x01];
161    data2.extend_from_slice(kl);
162    data2.extend_from_slice(kr);
163    data2.extend_from_slice(&index.to_le_bytes());
164
165    let mut hmac2 =
166        Hmac::<Sha512>::new_from_slice(chain_code).expect("HMAC can take key of any size");
167    hmac2.update(&data2);
168    let cc_full = hmac2.finalize().into_bytes();
169
170    // New chain code is second 32 bytes of cc_full
171    let mut new_cc = [0u8; 32];
172    new_cc.copy_from_slice(&cc_full[32..64]);
173
174    // Calculate new kL: kL + 8 * trunc28(zL) (little-endian addition)
175    let zl = &z[0..32];
176    let zr = &z[32..64];
177
178    // trunc28(zL) = zL with last 4 bytes zeroed, then multiply by 8
179    let mut zl_trunc = [0u8; 32];
180    zl_trunc[0..28].copy_from_slice(&zl[0..28]);
181    // Multiply by 8 (shift left by 3)
182    let mut carry = 0u16;
183    for byte in zl_trunc.iter_mut() {
184        let val = (*byte as u16) * 8 + carry;
185        *byte = val as u8;
186        carry = val >> 8;
187    }
188
189    // Add to kL (little-endian)
190    let mut new_kl = [0u8; 32];
191    let mut c = 0u16;
192    for i in 0..32 {
193        let sum = (kl[i] as u16) + (zl_trunc[i] as u16) + c;
194        new_kl[i] = sum as u8;
195        c = sum >> 8;
196    }
197
198    // Calculate new kR: kR + zR (little-endian addition mod 2^256)
199    let mut new_kr = [0u8; 32];
200    let mut c = 0u16;
201    for i in 0..32 {
202        let sum = (kr[i] as u16) + (zr[i] as u16) + c;
203        new_kr[i] = sum as u8;
204        c = sum >> 8;
205    }
206
207    // Combine into new extended secret
208    let mut new_extended = [0u8; 64];
209    new_extended[0..32].copy_from_slice(&new_kl);
210    new_extended[32..64].copy_from_slice(&new_kr);
211
212    (new_extended, new_cc)
213}

215/// Ledger key pair with extended key for signing
216#[derive(Clone, Zeroize, ZeroizeOnDrop)]
217pub struct LedgerKeyPair {
218    /// Full 64-byte extended secret key (for BIP32-Ed25519 signing)
219    pub extended_key: [u8; 64],
220    /// 32-byte Ed25519 secret key (clamped scalar)
221    pub secret_key: [u8; 32],
222    /// 32-byte Ed25519 public key
223    pub public_key: [u8; 32],
224}
225
226impl LedgerKeyPair {
227    /// Sign a message using Ed25519
228    pub fn sign(&self, message: &[u8]) -> [u8; 64] {
229        let pair =
230            sp_core::ed25519::Pair::from_seed_slice(&self.secret_key).expect("Valid 32-byte seed");
231        pair.sign(message).0
232    }
233
234    /// Get SS58 address for the public key
235    pub fn ss58_address(&self, prefix: u16) -> String {
236        use sp_core::crypto::{Ss58AddressFormat, Ss58Codec};
237        use sp_core::ed25519::Public;
238
239        let public = Public::from_raw(self.public_key);
240        let format = Ss58AddressFormat::custom(prefix);
241        public.to_ss58check_with_version(format)
242    }
243}

245/// Derive Polkadot Ledger key (m/44'/354'/account'/0'/0')
246pub fn derive_polkadot_ledger_key(seed_phrase: &str, account: u32) -> Result<LedgerKeyPair> {
247    derive_ledger_key(seed_phrase, SLIP0044_POLKADOT, account, 0, 0)
248}

250/// Derive Kusama Ledger key (m/44'/434'/account'/0'/0')
251pub fn derive_kusama_ledger_key(seed_phrase: &str, account: u32) -> Result<LedgerKeyPair> {
252    derive_ledger_key(seed_phrase, SLIP0044_KUSAMA, account, 0, 0)
253}
