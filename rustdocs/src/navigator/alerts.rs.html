3#[derive(PartialEq, Eq, Debug, Copy, Clone)]
4pub enum Alert {

6    Error,

9}
10
11impl Alert {
12    pub fn get_name(&self) -> String {
13        match self {
14            Alert::Empty => String::from("Empty"),
15            Alert::Error => String::from("Error"),
16            Alert::ErrorDisplay => String::from("ErrorDisplay"),
17            Alert::Shield => String::from("Shield"),
18        }
19    }
