17lazy_static::lazy_static! {
18    /// Track nonce hashes that have been consumed by signing.
19    /// A nonce that appears here has been used and MUST NOT be reused.
20    /// This prevents catastrophic key recovery from nonce reuse.
21    /// Fingerprint is SHA-256(nonces_hex) — collision-resistant; the
22    /// full 32 bytes are stored so a malicious peer cannot engineer a
23    /// fingerprint collision to bypass reuse detection.
24    static ref USED_NONCES: Mutex<HashSet<[u8; 32]>> = Mutex::new(HashSet::new());
25}
26
27/// Cryptographic fingerprint of a nonce for reuse tracking.
28/// Uses SHA-256 — engineered collisions are infeasible.
29fn nonce_fingerprint(nonces_hex: &str) -> [u8; 32] {
30    use sha2::{Digest, Sha256};
31    Sha256::digest(nonces_hex.as_bytes()).into()
32}
33
34/// Mark a nonce as consumed. Returns Err if already used.
35///
36/// Reuse is checked against BOTH the in-memory HashSet and the persistent
37/// sled tree. The persistent layer defeats a force-quit attack where an
38/// adversary could otherwise restart the app, drop the in-memory state,
39/// and replay an already-used nonce against a different message —
40/// which would catastrophically leak the FROST private key.
41fn consume_nonce(nonces_hex: &str) -> Result<(), String> {
42    let fp = nonce_fingerprint(nonces_hex);
43
44    // Persistent check first — survives app restart.
45    let db_guard = crate::DB
46        .read()
47        .map_err(|_| "DB lock poisoned".to_string())?;
48    if let Some(database) = db_guard.as_ref() {
49        db_handling::frost::mark_nonce_used(database, &fp).map_err(|e| {
50            format!(
51                "CRITICAL: {}. Reusing a FROST nonce with a different message would \
52                 leak the private key. Generate fresh nonces with sign_round1.",
53                e
54            )
55        })?;
56    }
57    drop(db_guard);
58
59    // In-memory write-through cache (fast-path for repeated reuse within
60    // the same session — sled would catch it too but at a sled round trip).
61    let mut used = USED_NONCES
62        .lock()
63        .map_err(|_| "nonce tracker poisoned".to_string())?;
64    if !used.insert(fp) {
65        return Err(
66            "CRITICAL: Nonce reuse detected. This nonce was already used for signing. \
67             Reusing a FROST nonce with a different message would leak the private key. \
68             Generate fresh nonces with sign_round1."
69                .to_string(),
70        );
71    }
72    // Bound the in-memory cache; the persistent tracker remains authoritative.
73    if used.len() > 1000 {
74        used.clear();
75        used.insert(fp);
76    }
77    Ok(())
78}

80/// DKG round 1: generate ephemeral identity + signed commitment.
81/// Returns JSON: { "secret": hex, "broadcast": hex }
82///
83/// The returned JSON contains secret material in the "secret" field. The
84/// FFI caller is responsible for clearing the returned String once it
85/// has been persisted (encrypted) by the native layer. We zero our own
86/// in-memory copies of the secret hex before returning.
87pub fn frost_dkg_part1(max_signers: u16, min_signers: u16) -> Result<String, String> {
88    let mut result = orchestrate::dkg_part1(max_signers, min_signers).map_err(|e| e.to_string())?;
89    let json = serde_json::to_string(&serde_json::json!({
90        "secret": result.secret_hex,
91        "broadcast": result.broadcast_hex,
92    }))
93    .map_err(|e| e.to_string())?;
94    result.secret_hex.zeroize();
95    Ok(json)
96}
97
98/// DKG round 2: process signed round1 broadcasts.
99/// peer_broadcasts_json: JSON array of hex strings.
100/// Returns JSON: { "secret": hex, "peer_packages": [hex, ...] }
101///
102/// As with round 1, the returned JSON contains secret material; the FFI
103/// caller must clear it after persistence. Round 1's `secret_hex` was
104/// already zeroized when its caller dropped the JSON string; this round
105/// zeroes its own intermediate copy.
106pub fn frost_dkg_part2(secret_hex: &str, peer_broadcasts_json: &str) -> Result<String, String> {
107    let broadcasts: Vec<String> = serde_json::from_str(peer_broadcasts_json)
108        .map_err(|e| format!("bad broadcasts JSON: {}", e))?;
109    let mut result = orchestrate::dkg_part2(secret_hex, &broadcasts).map_err(|e| e.to_string())?;
110    let json = serde_json::to_string(&serde_json::json!({
111        "secret": result.secret_hex,
112        "peer_packages": result.peer_packages,
113    }))
114    .map_err(|e| e.to_string())?;
115    result.secret_hex.zeroize();
116    Ok(json)
117}

119/// DKG round 3: finalize — returns key package + public key package.
120/// Returns JSON: { "key_package": hex, "public_key_package": hex, "ephemeral_seed": hex }
121///
122/// `key_package` and `ephemeral_seed` are long-lived secrets — the FFI
123/// caller persists them encrypted as the FROST wallet record and zeroes
124/// the JSON after that. We zero our local copies of those fields once
125/// they have been serialized into the JSON return value.
126pub fn frost_dkg_part3(
127    secret_hex: &str,
128    round1_broadcasts_json: &str,
129    round2_packages_json: &str,
130) -> Result<String, String> {
131    let r1: Vec<String> = serde_json::from_str(round1_broadcasts_json)
132        .map_err(|e| format!("bad round1 JSON: {}", e))?;
133    let r2: Vec<String> = serde_json::from_str(round2_packages_json)
134        .map_err(|e| format!("bad round2 JSON: {}", e))?;
135    let mut result = orchestrate::dkg_part3(secret_hex, &r1, &r2).map_err(|e| e.to_string())?;
136    let json = serde_json::to_string(&serde_json::json!({
137        "key_package": result.key_package_hex,
138        "public_key_package": result.public_key_package_hex,
139        "ephemeral_seed": result.ephemeral_seed_hex,
140    }))
141    .map_err(|e| e.to_string())?;
142    result.key_package_hex.zeroize();
143    result.ephemeral_seed_hex.zeroize();
144    Ok(json)
145}

147/// signing round 1: generate nonces + signed commitments.
148/// Returns JSON: { "nonces": hex, "commitments": hex }
149pub fn frost_sign_round1(
150    ephemeral_seed_hex: &str,
151    key_package_hex: &str,
152) -> Result<String, String> {
153    let seed = parse_seed(ephemeral_seed_hex)?;
154    let (nonces, commitments) =
155        orchestrate::sign_round1(&seed, key_package_hex).map_err(|e| e.to_string())?;
156    serde_json::to_string(&serde_json::json!({
157        "nonces": nonces,
158        "commitments": commitments,
159    }))
160    .map_err(|e| e.to_string())
161}

163/// spend-authorize round 2: produce FROST share bound to sighash + alpha.
164/// This is what Zigner does when participating in a multisig spend.
165/// commitments_json: JSON array of hex-encoded signed commitments from all signers.
166/// Returns: hex-encoded signature share.
167pub fn frost_spend_sign_round2(
168    key_package_hex: &str,
169    nonces_hex: &str,
170    sighash_hex: &str,
171    alpha_hex: &str,
172    commitments_json: &str,
173) -> Result<String, String> {
174    // Consume nonce BEFORE signing — prevents reuse even if signing fails
175    consume_nonce(nonces_hex)?;
176
177    let sighash = parse_32(sighash_hex, "sighash")?;
178    let alpha = parse_32(alpha_hex, "alpha")?;
179    let commitments: Vec<String> = serde_json::from_str(commitments_json)
180        .map_err(|e| format!("bad commitments JSON: {}", e))?;
181    orchestrate::spend_sign_round2(key_package_hex, nonces_hex, &sighash, &alpha, &commitments)
182        .map_err(|e| e.to_string())
183}

186/// can map it to the correct FROST identifier by VK, regardless of BTreeMap order.
187/// Use this instead of frost_spend_sign_round2 whenever sharing over the relay.
188/// Returns: hex-encoded SignedMessage containing the signature share.
189pub fn frost_spend_sign_round2_signed(

202    let commitments: Vec<String> = serde_json::from_str(commitments_json)

204    orchestrate::spend_sign_round2_signed(

207        nonces_hex,

215/// sign multiple actions at once (one share per alpha).
216/// alphas_json: JSON array of hex-encoded 32-byte alphas (one per Orchard action).
217/// commitments_json: JSON array of hex-encoded signed commitments.
218/// Returns JSON: { "shares": [hex, ...] }
219pub fn frost_spend_sign_actions(
220    key_package_hex: &str,
221    nonces_hex: &str,
222    sighash_hex: &str,
223    alphas_json: &str,
224    commitments_json: &str,
225) -> Result<String, String> {
226    // Consume nonce BEFORE any signing — prevents reuse even if one action fails
227    consume_nonce(nonces_hex)?;
228
229    let sighash = parse_32(sighash_hex, "sighash")?;
230    let alphas: Vec<String> =
231        serde_json::from_str(alphas_json).map_err(|e| format!("bad alphas JSON: {}", e))?;
232    let commitments: Vec<String> = serde_json::from_str(commitments_json)
233        .map_err(|e| format!("bad commitments JSON: {}", e))?;
234
235    let mut shares = Vec::new();
236    for alpha_hex in &alphas {
237        let alpha = parse_32(alpha_hex, "alpha")?;
238        let share = orchestrate::spend_sign_round2(
239            key_package_hex,
240            nonces_hex,
241            &sighash,
242            &alpha,
243            &commitments,
244        )
245        .map_err(|e| e.to_string())?;
246        shares.push(share);
247    }
248
249    serde_json::to_string(&serde_json::json!({ "shares": shares })).map_err(|e| e.to_string())
250}

263/// Derive both the Orchard-only UFVK (`uview1…`) and unified address (`u1…`)
264/// from a FROST public_key_package + the host-broadcast `sk`. Every participant
265/// who calls this with identical inputs lands on byte-identical outputs.
266/// Returns JSON `{ "orchard_fvk_uview": "...", "address": "..." }`.
267pub fn frost_derive_metadata(
268    public_key_package_hex: &str,
269    sk_hex: &str,
270    mainnet: bool,
271    diversifier_index: u32,
272) -> Result<String, String> {
273    use frost_spend::frost_keys::PublicKeyPackage;
274    use frost_spend::keys::{derive_address as fs_derive_address, derive_fvk_from_sk};
275    use frost_spend::orchestrate::from_hex;
276    use zcash_address::unified::{Address as UnifiedAddress, Encoding, Fvk, Receiver, Ufvk};
277    use zcash_protocol::consensus::NetworkType as Network;
278
279    let sk = parse_32(sk_hex, "fvk sk")?;
280    let pubkeys: PublicKeyPackage = from_hex(public_key_package_hex).map_err(|e| e.to_string())?;
281    let fvk = derive_fvk_from_sk(sk, &pubkeys)
282        .ok_or_else(|| "failed to derive FVK from group key + sk".to_string())?;
283    let addr = fs_derive_address(&fvk, diversifier_index);
284    let raw = addr.to_raw_address_bytes();
285
286    let network = if mainnet {
287        Network::Main
288    } else {
289        Network::Test
290    };
291
292    let ufvk_str = Ufvk::try_from_items(vec![Fvk::Orchard(fvk.to_bytes())])
293        .map_err(|e| format!("build UFVK: {e}"))?
294        .encode(&network);
295
296    let addr_str = UnifiedAddress::try_from_items(vec![Receiver::Orchard(raw)])
297        .map_err(|e| format!("build address: {e}"))?
298        .encode(&network);
299
300    serde_json::to_string(&serde_json::json!({
301        "orchard_fvk_uview": ufvk_str,
302        "address": addr_str,
303    }))
304    .map_err(|e| e.to_string())
305}

307/// Verify a FROST payout PCZT on-device before signing (gh #17). Recomputes the
308/// canonical ZIP-244 sighash from the PCZT and OVK-decodes its outputs with the
309/// group UFVK (which zigner stored at DKG — NOT trusting the host), so the cold
310/// signer authorizes the ACTUAL recipient/amount rather than a host-supplied
311/// claim. The caller compares `recomputed_sighash` against the sighash it's
312/// being asked to sign, and the derived outputs against the displayed summary;
313/// any mismatch must block the signature.
314///
315/// Returns JSON:
316/// `{ recomputed_sighash, sighash_match, outputs:[{recipient,amount_zat,is_change}],
317///    total_send_zat, total_change_zat }`
318pub fn frost_verify_pczt(
319    pczt_hex: &str,
320    claimed_sighash_hex: &str,
321    orchard_fvk_uview: &str,
322) -> Result<String, String> {
323    use orchard::keys::Scope;
324    use orchard::note_encryption::{IronwoodVersion, OrchardVersion};
325    use zcash_address::unified::{Container, Encoding, Fvk, Ufvk};
326
327    let bytes = hex::decode(pczt_hex.trim()).map_err(|e| format!("bad pczt hex: {e}"))?;
328    let pczt = pczt::Pczt::parse(&bytes).map_err(|e| format!("pczt parse: {e:?}"))?;
329
330    // The sighash the FROST share will commit to, from pczt's own Signer: it
331    // dispatches on the transaction version (V5, or V6 for an ironwood spend)
332    // and the PCZT's branch id (NU6.3 or NU7). zcli's builder and joiner take
333    // it from the same place, so all three agree by construction.
334    let shielded_sighash = pczt::roles::signer::Signer::new(pczt.clone())
335        .map_err(|e| format!("pczt signer init: {e:?}"))?
336        .shielded_sighash();
337    let recomputed = hex::encode(shielded_sighash);
338    let sighash_match = recomputed.eq_ignore_ascii_case(claimed_sighash_hex.trim());
339
340    // group orchard FVK from the stored UFVK string (uview1…)
341    let (network, ufvk) =
342        Ufvk::decode(orchard_fvk_uview.trim()).map_err(|e| format!("ufvk decode: {e}"))?;
343    let fvk_bytes = ufvk
344        .items()
345        .into_iter()
346        .find_map(|item| match item {
347            Fvk::Orchard(b) => Some(b),
348            _ => None,
349        })
350        .ok_or_else(|| "UFVK has no orchard component".to_string())?;
351    let fvk = orchard::keys::FullViewingKey::from_bytes(&fvk_bytes)
352        .ok_or_else(|| "invalid orchard FVK in UFVK".to_string())?;
353    let ovks = [fvk.to_ovk(Scope::External), fvk.to_ovk(Scope::Internal)];
354
355    // Same byte stream the sighash above was derived from.
356    let tx_data = pczt
357        .into_effects()
358        .map_err(|e| format!("pczt into_effects: {e:?}"))?;
359
360    let mut out = Recovered::default();
361    // Both pools: since NU6.3 a spend's outputs land in the ironwood bundle,
362    // whose note plaintexts the orchard domain refuses by design.
363    if let Some(b) = tx_data.orchard_bundle() {
364        recover_outputs::<OrchardVersion, _>(b, &ovks, network, &mut out);
365    }
366    if let Some(b) = tx_data.ironwood_bundle() {
367        recover_outputs::<IronwoodVersion, _>(b, &ovks, network, &mut out);
368    }
369
370    serde_json::to_string(&serde_json::json!({
371        "recomputed_sighash": recomputed,
372        "sighash_match": sighash_match,
373        "outputs": out.outputs,
374        "total_send_zat": out.total_send,
375        "total_change_zat": out.total_change,
376    }))
377    .map_err(|e| e.to_string())
378}

380#[derive(Default)]
381struct Recovered {
382    outputs: Vec<serde_json::Value>,
383    total_send: u64,
384    total_change: u64,
385}
386
387/// Decrypt one bundle's outputs with the outgoing viewing keys: the external
388/// key opens a payment, the internal key opens change back to the group.
389fn recover_outputs<V, A>(
390    bundle: &orchard::Bundle<A, zcash_protocol::value::ZatBalance>,
391    ovks: &[orchard::keys::OutgoingViewingKey; 2],
392    network: zcash_protocol::consensus::NetworkType,
393    out: &mut Recovered,
394) where
395    V: orchard::note_encryption::DomainVersion,
396    A: orchard::bundle::Authorization,
397{
398    use orchard::note_encryption::NoteEncryptionDomain;
399    use zcash_address::unified::{Address as UnifiedAddress, Encoding, Receiver};
400    use zcash_note_encryption::try_output_recovery_with_ovk;
401
402    for action in bundle.actions().iter() {
403        let domain = NoteEncryptionDomain::<V>::for_action(action);
404        let cv = action.cv_net();
405        let out_ct = action.encrypted_note().out_ciphertext;
406        let recovered = ovks.iter().enumerate().find_map(|(i, ovk)| {
407            try_output_recovery_with_ovk(&domain, ovk, action, cv, &out_ct).map(|r| (r, i == 1))
408        });
409        if let Some(((note, addr, _memo), is_change)) = recovered {
410            let amount = note.value().inner();
411            if is_change {
412                out.total_change = out.total_change.saturating_add(amount);
413            } else {
414                out.total_send = out.total_send.saturating_add(amount);
415            }
416            let ua = UnifiedAddress::try_from_items(vec![Receiver::Orchard(
417                addr.to_raw_address_bytes(),
418            )])
419            .map(|a| a.encode(&network))
420            .unwrap_or_default();
421            out.outputs.push(serde_json::json!({
422                "recipient": ua,
423                "amount_zat": amount,
424                "is_change": is_change,
425            }));
426        }
427    }
428}

432fn parse_seed(hex_str: &str) -> Result<[u8; 32], String> {
433    let bytes = hex::decode(hex_str).map_err(|e| format!("bad seed hex: {}", e))?;
434    bytes
435        .try_into()
436        .map_err(|_| "seed must be 32 bytes".to_string())
437}

441    bytes
