1// age_backup.rs — encrypt a backup to public keys instead of to a passphrase.
2//
3// The FROST share envelope (frost_backup.rs) is already encrypted, under a
4// PBKDF2 passphrase. This is a second option for the same bytes, and it
5// changes what you must not lose rather than how strong the cipher is.
6//
7// A passphrase is a SEPARATE secret. Forgetting it is one of the commonest
8// ways a backup turns out to be worthless, and it fails silently - you find
9// out at restore time, which is the worst moment to find anything out.
10//
11// Encrypting to this device's derived SSH key instead makes the ability to
12// open the backup seed-derived, like every other key here. The share itself
13// still cannot be regenerated from the seed - FROST shares come out of the DKG
14// with their own entropy, and that is the property that makes a threshold
15// worth having - but the 24 words you already back up now open the envelope,
16// so there is one fewer secret to lose.
17//
18// MULTIPLE RECIPIENTS, deliberately. Encrypting only to the device key ties
19// the backup's fate to the seed's: lose the seed and you lose both, which is
20// the correlated failure a threshold exists to avoid. Add a co-signer's key,
21// or a second device's, and the two can fail independently.
22//
23// Recipients are ordinary `ssh-ed25519 AAAA...` lines - the same thing the
24// device shows on its Release/SSH key screen, and the same thing anyone
25// publishes on GitHub - or native `age1...` recipients.
26

32/// Own domain, NOT the zigner-ssh or ZID key.
33///
34/// Both of those exist to SIGN, and the ZID key signs challenges chosen by
35/// whatever site asked for one. age's ssh recipients work by mapping an
36/// ed25519 key onto X25519, so pointing one at this would make a single
37/// keypair both a signing oracle for attacker-chosen input and a decryption
38/// key - the sort of cross-protocol reuse that is hard to reason about and
39/// unnecessary here.
40///
41/// Unnecessary because this recipient is self-addressed: no counterparty has
42/// to know it in advance, so a dedicated key costs nothing. Co-signers still
43/// paste their ordinary ssh public keys, which is their call to make and the
44/// normal way age is used.
45pub const AGE_BACKUP_DOMAIN: &str = "zigner-backup-age";
46
47/// Parse one recipient string, accepting either flavour.
48///
49/// Both are tried rather than dispatched on a prefix: "starts with ssh-" would
50/// quietly reject ssh-rsa, and guessing wrong produces a confusing error about
51/// the wrong format instead of about the actual input.
52fn parse_recipient(s: &str) -> Result<Box<dyn age::Recipient + Send>, String> {
53    let t = s.trim();
54    if t.is_empty() {
55        return Err("empty recipient".into());
56    }
57    if let Ok(r) = age::ssh::Recipient::from_str(t) {
58        return Ok(Box::new(r));
59    }
60    if let Ok(r) = age::x25519::Recipient::from_str(t) {
61        return Ok(Box::new(r));
62    }
63    Err(format!(
64        "not a usable recipient: expected an ssh public key line or an age1... recipient, got \"{}\"",
65        t.chars().take(24).collect::<String>()
66    ))
67}

69/// Encrypt `plaintext` to every recipient. ASCII-armored, so it survives being
70/// pasted into a message or a text file.
71pub fn encrypt_to_recipients(plaintext: &[u8], recipients: &[String]) -> Result<String, String> {
72    if recipients.is_empty() {
73        // Encrypting to nobody produces a file nobody can open. Better to
74        // refuse than to hand back something that looks like a backup.
75        return Err("no recipients: the result would be undecryptable by anyone".into());
76    }
77    let parsed: Vec<Box<dyn age::Recipient + Send>> = recipients
78        .iter()
79        .map(|r| parse_recipient(r))
80        .collect::<Result<_, _>>()?;
81
82    let encryptor =
83        age::Encryptor::with_recipients(parsed.iter().map(|r| r.as_ref() as &dyn age::Recipient))
84            .map_err(|e| format!("age encryptor: {e}"))?;
85
86    let mut armored = Vec::new();
87    let writer =
88        age::armor::ArmoredWriter::wrap_output(&mut armored, age::armor::Format::AsciiArmor)
89            .map_err(|e| format!("armor: {e}"))?;
90    let mut w = encryptor
91        .wrap_output(writer)
92        .map_err(|e| format!("age encrypt: {e}"))?;
93    w.write_all(plaintext)
94        .map_err(|e| format!("age write: {e}"))?;
95    w.finish()
96        .map_err(|e| format!("age finish: {e}"))?
97        .finish()
98        .map_err(|e| format!("armor finish: {e}"))?;
99
100    String::from_utf8(armored).map_err(|e| format!("armor utf8: {e}"))
101}

103/// The device's backup keypair, in the shape ssh-key uses.
104fn device_keypair(
105    seed_phrase: &str,
106    index: u32,
107) -> Result<ssh_key::private::Ed25519Keypair, String> {
108    let seed = derive_domain_seed(seed_phrase, AGE_BACKUP_DOMAIN, index)?;
109    Ok(ssh_key::private::Ed25519Keypair::from_seed(&seed))
110}
111
112/// This device's recipient as an `ssh-ed25519 ...` line — what to hand out so
113/// a backup can be addressed back to this device.
114pub fn device_recipient(seed_phrase: &str, index: u32) -> Result<String, String> {
115    let keypair = device_keypair(seed_phrase, index)?;
116    ssh_key::PublicKey::from(keypair.public)
117        .to_openssh()
118        .map_err(|e| format!("openssh public encode: {e}"))
119}

121/// Rebuild this device's SSH key as an OpenSSH private key file.
122///
123/// age's ssh identity reads a key file, and the device holds a derived seed.
124/// The conversion is a formatting step and nothing more - the same key either
125/// way - but it means the private half exists in a second representation for
126/// the life of this call, so it is built here and dropped here rather than
127/// handed to a caller.
128fn device_identity(seed_phrase: &str, index: u32) -> Result<age::ssh::Identity, String> {
129    let keypair = device_keypair(seed_phrase, index)?;
130    let pem = ssh_key::PrivateKey::from(keypair)
131        .to_openssh(ssh_key::LineEnding::LF)
132        .map_err(|e| format!("openssh encode: {e}"))?;
133    age::ssh::Identity::from_buffer(pem.as_bytes(), None)
134        .map_err(|e| format!("age ssh identity: {e}"))
135}

137/// Decrypt an armored age file addressed to this device.
138pub fn decrypt_with_device_key(
139    seed_phrase: &str,
140    index: u32,
141    armored: &str,
142) -> Result<Vec<u8>, String> {
143    let identity = device_identity(seed_phrase, index)?;
144    let decryptor = age::Decryptor::new(age::armor::ArmoredReader::new(armored.as_bytes()))
145        .map_err(|e| format!("age decryptor: {e}"))?;
146    let mut reader = decryptor
147        .decrypt(std::iter::once(&identity as &dyn age::Identity))
148        .map_err(|e| {
149            format!("age decrypt failed - is this file addressed to this device's key? ({e})")
150        })?;
151    let mut out = Vec::new();
152    reader
153        .read_to_end(&mut out)
154        .map_err(|e| format!("age read: {e}"))?;
155    Ok(out)
156}

158#[cfg(test)]
159mod tests {
160    use super::*;
161
162    const SEED_A: &str = "bottom drive obey lake curtain smoke basket hold race lonely fit walk";
163    const SEED_B: &str =
164        "legal winner thank year wave sausage worth useful legal winner thank yellow";
165
166    #[test]
167    fn round_trips_to_this_device() {
168        let recipient = device_recipient(SEED_A, 0).unwrap();
169        assert!(recipient.starts_with("ssh-ed25519 "));
170        let armored = encrypt_to_recipients(b"share envelope bytes", &[recipient]).unwrap();
171        assert!(armored.starts_with("-----BEGIN AGE ENCRYPTED FILE-----"));
172        let out = decrypt_with_device_key(SEED_A, 0, &armored).unwrap();
173        assert_eq!(out, b"share envelope bytes");
174    }
175
176    /// The point of multiple recipients: either holder can open it, alone.
177    #[test]
178    fn either_recipient_can_open_it() {
179        let a = device_recipient(SEED_A, 0).unwrap();
180        let b = device_recipient(SEED_B, 0).unwrap();
181        let armored = encrypt_to_recipients(b"two holders", &[a, b]).unwrap();
182        assert_eq!(
183            decrypt_with_device_key(SEED_A, 0, &armored).unwrap(),
184            b"two holders"
185        );
186        assert_eq!(
187            decrypt_with_device_key(SEED_B, 0, &armored).unwrap(),
188            b"two holders"
189        );
190    }
191
192    /// A device that is not a recipient must not be able to read it, which is
193    /// the whole claim being made.
194    #[test]
195    fn a_device_not_addressed_cannot_open_it() {
196        let a = device_recipient(SEED_A, 0).unwrap();
197        let armored = encrypt_to_recipients(b"only for A", &[a]).unwrap();
198        assert!(decrypt_with_device_key(SEED_B, 0, &armored).is_err());
199    }
200
201    /// Different derivation index is a different key, so it is a different
202    /// recipient - worth pinning, since the index is easy to pass wrongly.
203    #[test]
204    fn a_different_index_is_a_different_recipient() {
205        let a0 = device_recipient(SEED_A, 0).unwrap();
206        let a1 = device_recipient(SEED_A, 1).unwrap();
207        assert_ne!(a0, a1);
208        let armored = encrypt_to_recipients(b"index zero", &[a0]).unwrap();
209        assert!(decrypt_with_device_key(SEED_A, 1, &armored).is_err());
210    }
211
212    #[test]
213    fn refuses_to_encrypt_to_nobody() {
214        assert!(encrypt_to_recipients(b"x", &[]).is_err());
215    }
216
217    #[test]
218    fn refuses_a_recipient_it_cannot_parse() {
219        let err = encrypt_to_recipients(b"x", &["not-a-key".to_string()]).unwrap_err();
220        assert!(
221            err.contains("not a usable recipient"),
222            "unexpected: {}",
223            err
224        );
225    }
