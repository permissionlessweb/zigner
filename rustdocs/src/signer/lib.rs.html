71pub enum ErrorDisplayed {
72    /// String description of error
73    Str {
74        /// Error description
75        s: String,
76    },
77    MutexPoisoned,
78    DbNotInitialized,
79    /// Tried to load metadata for unknown network.
80    LoadMetaUnknownNetwork {
81        /// Name of the network not known to the Vault.
82        name: String,
83    },
84    /// Tried to add specs already present in Vault.
85    SpecsKnown {
86        name: String,
87        encryption: Encryption,
88    },
89    /// The metadata with this network version already in db.
90    MetadataKnown {
91        name: String,
92        version: u32,
93    },
94    /// Do not have an up-to-date version of metadata in db
95    MetadataOutdated {
96        name: String,
97        have: u32,
98        want: u32,
99    },
100    /// Tried to sign transaction with an unknown network
101    UnknownNetwork {
102        genesis_hash: H256,
103        encryption: Encryption,
104    },
105    /// No metadata for a known network found in store
106    NoMetadata {
107        name: String,
108    },
109    /// Provided password is incorrect
110    WrongPassword,
111    /// Database schema mismatch
112    DbSchemaMismatch,
113}

115impl From<NavigatorError> for ErrorDisplayed {
116    fn from(e: NavigatorError) -> Self {
117        match &e {
118            NavigatorError::MutexPoisoned => Self::MutexPoisoned,
119            NavigatorError::DbNotInitialized => Self::DbNotInitialized,
120            NavigatorError::TransactionParsing(t) => match t {
121                TxParsingError::LoadMetaUnknownNetwork { name } => {
122                    Self::LoadMetaUnknownNetwork { name: name.clone() }
123                }
124                TxParsingError::SpecsKnown { name, encryption } => Self::SpecsKnown {
125                    name: name.clone(),
126                    encryption: *encryption,
127                },
128                TxParsingError::MetadataKnown { name, version } => Self::MetadataKnown {
129                    name: name.clone(),
130                    version: *version,
131                },
132                TxParsingError::AllExtensionsParsingFailed {
133                    ref network_name,
134                    ref errors,
135                } => {
136                    if let Some((want, parser::Error::WrongNetworkVersion { in_metadata, .. })) =
137                        errors.first()
138                    {
139                        Self::MetadataOutdated {
140                            name: network_name.to_string(),
141                            have: *in_metadata,
142                            want: *want,
143                        }
144                    } else {
145                        Self::Str { s: format!("{e}") }
146                    }
147                }
148                TxParsingError::UnknownNetwork {
149                    genesis_hash,
150                    encryption,
151                } => Self::UnknownNetwork {
152                    genesis_hash: *genesis_hash,
153                    encryption: *encryption,
154                },
155                TxParsingError::NoMetadata { name } => Self::NoMetadata {
156                    name: name.to_string(),
157                },
158                _ => Self::Str { s: format!("{e}") },
159            },
160            NavigatorError::TransactionSigning(transaction_signing::Error::WrongPassword) => {
161                Self::WrongPassword
162            }
163            _ => Self::Str { s: format!("{e}") },
164        }
165    }
166}

263/// Should be called once at start of the app and could be called on app reset
264///
265/// Accepts list of seed names to avoid calling [`update_seed_names`] every time
266fn init_navigation(dbname: &str, seed_names: Vec<String>) -> Result<(), ErrorDisplayed> {
267    let val = Some(sled::open(dbname).map_err(|e| ErrorDisplayed::from(e.to_string()))?);
268
269    // Each DB lock MUST be released before the next is acquired. `DB` is a
270    // std RwLock: a thread that holds the write guard and then calls read()
271    // on the same lock deadlocks forever. 0.10.0 released the write lock
272    // implicitly (`*DB.write().unwrap() = val;` drops the temporary at the
273    // `;`); the poisoned-recovery refactor bound the write guard to a
274    // variable that stayed alive across the reads below, freezing
275    // init_navigation on EVERY launch (universal startup ANR). Keep the
276    // poisoned recovery, but scope each guard so it drops before the next.
277    // (A poisoned lock means an earlier call panicked while holding it;
278    // unwrap() would then make every subsequent call panic too.)
279    {
280        let mut guard = match DB.write() {
281            Ok(g) => g,
282            Err(poisoned) => poisoned.into_inner(),
283        };
284        *guard = val;
285    }
286    init_logging("Vault".to_string());
287
288    // Bootstrap the anchor-verifier registry with the built-in default
289    // on first run. Idempotent: skipped if the tree already has the
290    // default key, and skipped entirely if the user has populated their
291    // own verifiers (we don't override their explicit configuration).
292    {
293        let guard = match DB.read() {
294            Ok(g) => g,
295            Err(poisoned) => poisoned.into_inner(),
296        };
297        if let Some(database) = guard.as_ref() {
298            if constants::ROTKO_ZCASH_VERIFIER != [0u8; 32] {
299                let _ = db_handling::anchor_verifiers::bootstrap_default(
300                    database,
301                    &constants::ROTKO_ZCASH_VERIFIER,
302                    "Rotko Networks (built-in)",
303                );
304            }
305        }
306    }
307
308    // Clone the sled handle out under a short-lived read lock, then release it
309    // before calling navigator::init_navigation (which may lock DB itself).
310    let database = {
311        let guard = match DB.read() {
312            Ok(g) => g,
313            Err(poisoned) => poisoned.into_inner(),
314        };
315        guard.as_ref().unwrap().clone()
316    };
317    Ok(navigator::init_navigation(database, seed_names)?)
318}

425/// Create address by looking up network via genesis hash
426/// genesis_hash_hex: hex-encoded 32-byte genesis hash
427fn try_create_address_by_genesis(
428    seed_name: &str,
429    seed_phrase: &str,
430    path: &str,
431    genesis_hash_hex: &str,
432) -> anyhow::Result<(), ErrorDisplayed> {
433    use db_handling::helpers::genesis_hash_in_specs;
434    use sp_core::H256;
435
436    let db = get_db()?;
437    let genesis_hash_bytes = hex::decode(genesis_hash_hex).map_err(|e| ErrorDisplayed::Str {
438        s: format!("Invalid genesis hash hex: {e}"),
439    })?;
440    if genesis_hash_bytes.len() != 32 {
441        return Err(ErrorDisplayed::Str {
442            s: "Genesis hash must be 32 bytes".to_string(),
443        });
444    }
445    let genesis_hash = H256::from_slice(&genesis_hash_bytes);
446
447    let specs_invariants = genesis_hash_in_specs(&db, genesis_hash)
448        .map_err(|e| ErrorDisplayed::Str { s: format!("{e}") })?
449        .ok_or_else(|| ErrorDisplayed::Str {
450            s: format!("Network with genesis hash {} not found", genesis_hash_hex),
451        })?;
452
453    db_handling::identities::try_create_address(
454        &db,
455        seed_name,
456        seed_phrase,
457        path,
458        &specs_invariants.first_network_specs_key,
459    )
460    .map_err(|e| e.to_string().into())
461}

727/// Export seed backup data as JSON including account metadata (NO seed phrase)
728/// Returns JSON: {"v":2,"name":"...","accounts":[{"path":"...","genesis_hash":"...","network":"...","encryption":"..."}]}
729/// The seed phrase must be restored separately via banana split or manual entry
730fn bs_export_backup_data(seed_name: &str, _seed_phrase: &str) -> Result<String, ErrorDisplayed> {
731    use db_handling::helpers::try_get_network_specs;
732    use db_handling::identities::get_addresses_by_seed_name;
733    use serde_json::{json, Value};
734
735    let db = get_db()?;
736    let addresses = get_addresses_by_seed_name(&db, seed_name)
737        .map_err(|e| ErrorDisplayed::from(format!("{e}")))?;
738
739    let accounts: Vec<Value> = addresses
740        .into_iter()
741        .filter(|(_, details)| !details.is_root()) // skip root key, only derived accounts
742        .map(|(_, details)| {
743            // Look up network specs to get genesis_hash, network name, and base58prefix
744            let (genesis_hash, network_name, base58prefix) = if let Some(ref network_id) =
745                details.network_id
746            {
747                if let Ok(Some(specs)) = try_get_network_specs(&db, network_id) {
748                    // Only include base58prefix for Substrate networks (sr25519/ed25519/ecdsa)
749                    // Zcash and Penumbra don't use SS58 addresses
750                    let prefix: Option<u16> = match details.encryption {
751                        definitions::crypto::Encryption::Sr25519
752                        | definitions::crypto::Encryption::Ed25519
753                        | definitions::crypto::Encryption::Ecdsa => Some(specs.specs.base58prefix),
754                        _ => None,
755                    };
756                    (
757                        Some(hex::encode(specs.specs.genesis_hash)),
758                        Some(specs.specs.name),
759                        prefix,
760                    )
761                } else {
762                    (None, None, None)
763                }
764            } else {
765                (None, None, None)
766            };
767
768            json!({
769                "path": details.path,
770                "genesis_hash": genesis_hash,
771                "network": network_name,
772                "encryption": details.encryption.show(),
773                "has_pwd": details.has_pwd,
774                "base58prefix": base58prefix,
775            })
776        })
777        .collect();
778
779    let backup = json!({
780        "v": 2,
781        "name": seed_name,
782        "accounts": accounts,
783    });
784
785    serde_json::to_string(&backup)
786        .map_err(|e| ErrorDisplayed::from(format!("JSON serialization error: {e}")))
787}

814/// Export seed backup as UR-encoded multipart string frames for device-to-device migration
815/// Uses fountain codes for reliable animated QR scanning
816/// max_fragment_len: max bytes per QR frame (0 = single QR, 200-500 typical for animated)
817fn export_backup_ur(
818    seed_name: &str,
819    seed_phrase: &str,
820    max_fragment_len: u32,
821) -> Result<Vec<String>, ErrorDisplayed> {
822    // Get JSON backup data
823    let json_data = bs_export_backup_data(seed_name, seed_phrase)?;
824    let data_bytes = json_data.as_bytes();
825
826    // Wrap in simple CBOR byte string
827    let mut cbor_data = Vec::with_capacity(data_bytes.len() + 10);
828    let len = data_bytes.len();
829    if len <= 23 {
830        cbor_data.push(0x40 | len as u8);
831    } else if len <= 255 {
832        cbor_data.push(0x58);
833        cbor_data.push(len as u8);
834    } else if len <= 65535 {
835        cbor_data.push(0x59);
836        cbor_data.push((len >> 8) as u8);
837        cbor_data.push(len as u8);
838    } else {
839        cbor_data.push(0x5a);
840        cbor_data.push((len >> 24) as u8);
841        cbor_data.push((len >> 16) as u8);
842        cbor_data.push((len >> 8) as u8);
843        cbor_data.push(len as u8);
844    }
845    cbor_data.extend_from_slice(data_bytes);
846
847    if max_fragment_len == 0 || cbor_data.len() <= max_fragment_len as usize {
848        // Single part UR
849        let ur_string = ur::ur::encode(&cbor_data, &ur::Type::Custom("zigner-backup"));
850        Ok(vec![ur_string])
851    } else {
852        // Multi-part (animated) UR using fountain codes
853        let mut encoder =
854            ur::ur::Encoder::new(&cbor_data, max_fragment_len as usize, "zigner-backup").map_err(
855                |e| ErrorDisplayed::Str {
856                    s: format!("Failed to create UR encoder: {:?}", e),
857                },
858            )?;
859
860        // Generate enough frames for reliable scanning (2x the minimum)
861        let frame_count = encoder.fragment_count() * 2;
862        let mut frames = Vec::with_capacity(frame_count);
863        for _ in 0..frame_count {
864            let part = encoder.next_part().map_err(|e| ErrorDisplayed::Str {
865                s: format!("Failed to encode UR part: {:?}", e),
866            })?;
867            frames.push(part);
868        }
869        Ok(frames)
870    }
871}

873/// Decode UR-encoded backup and return JSON string
874fn decode_backup_ur(ur_parts: Vec<String>) -> Result<String, ErrorDisplayed> {
875    if ur_parts.is_empty() {
876        return Err(ErrorDisplayed::Str {
877            s: "No UR parts provided".to_string(),
878        });
879    }
880
881    // Try single-part decode first
882    if ur_parts.len() == 1 {
883        let (_, cbor_data) = ur::ur::decode(&ur_parts[0]).map_err(|e| ErrorDisplayed::Str {
884            s: format!("UR decode error: {:?}", e),
885        })?;
886        return extract_backup_json_from_cbor(&cbor_data);
887    }
888
889    // Multi-part decode using fountain codes
890    let mut decoder = ur::ur::Decoder::default();
891
892    for part in &ur_parts {
893        decoder.receive(part).map_err(|e| ErrorDisplayed::Str {
894            s: format!("UR receive error: {:?}", e),
895        })?;
896
897        if decoder.complete() {
898            match decoder.message() {
899                Ok(Some(cbor_data)) => return extract_backup_json_from_cbor(&cbor_data),
900                Ok(None) => {
901                    return Err(ErrorDisplayed::Str {
902                        s: "UR decoder complete but no message".to_string(),
903                    })
904                }
905                Err(e) => {
906                    return Err(ErrorDisplayed::Str {
907                        s: format!("UR message error: {:?}", e),
908                    })
909                }
910            }
911        }
912    }
913
914    Err(ErrorDisplayed::Str {
915        s: "Incomplete UR data - need more frames".to_string(),
916    })
917}

970/// Export Penumbra Full Viewing Key for import into watch-only wallet (e.g., Prax)
971///
972/// Returns both:
973/// - FVK bech32m string (native Penumbra format) for direct import
974/// - UR-encoded string (penumbra-accounts format) for hardware wallet QR compatibility
975///
976/// ## UR (Uniform Resource) Format for Penumbra
977///
978/// We define these UR types (proposed by Zigner, following Blockchain Commons pattern):
979///
980/// | UR Type                    | CBOR Tag | Description                    |
981/// |----------------------------|----------|--------------------------------|
982/// | penumbra-accounts          | 49301    | Container for accounts         |
983/// | penumbra-full-viewing-key  | 49302    | Single FVK with metadata       |
984///
985/// These tags are in the "first-come-first-served" range (32768+), no registration needed.
986/// Adjacent to Keystone's Zcash tags (49201-49204) for consistency.
987///
988/// ## CBOR Structure for penumbra-accounts
989///
990/// ```text
991/// PenumbraAccounts = {
992///   1: bytes,                ; wallet_id (32 bytes)
993///   2: [+ #49302(FVK)]       ; accounts array
994/// }
995///
996/// PenumbraFullViewingKey (#49302) = {
997///   1: tstr,                 ; fvk - bech32m encoded ("penumbrafullviewingkey1...")
998///   2: uint,                 ; index - account index
999///   ? 3: tstr                ; name - optional label
1000/// }
1001/// ```
1002#[allow(clippy::vec_init_then_push)]
1003fn export_penumbra_fvk(
1004    seed_phrase: &str,
1005    account_index: u32,
1006    label: &str,
1007) -> Result<PenumbraFvkExport, ErrorDisplayed> {
1008    use transaction_signing::penumbra::{FullViewingKey, FvkExportData, SpendKeyBytes};
1009
1010    // Derive spend key from seed phrase
1011    let spend_key_bytes =
1012        SpendKeyBytes::from_seed_phrase(seed_phrase, account_index).map_err(|e| {
1013            ErrorDisplayed::Str {
1014                s: format!("Failed to derive spend key: {e}"),
1015            }
1016        })?;
1017
1018    // Create FVK export data
1019    let label_opt = if label.is_empty() {
1020        None
1021    } else {
1022        Some(label.to_string())
1023    };
1024    let mut export_data =
1025        FvkExportData::from_spend_key(&spend_key_bytes, account_index, label_opt.clone()).map_err(
1026            |e| ErrorDisplayed::Str {
1027                s: format!("Failed to create FVK export: {e}"),
1028            },
1029        )?;
1030
1031    // embed ZID pubkey so zafu can link this wallet to the same zigner device
1032    // across network imports. best-effort — if ZID derivation fails, the FVK
1033    // export still succeeds, just without the canonical device identity.
1034    export_data.zid_pubkey = auth::derive_zid_pubkey(seed_phrase)
1035        .ok()
1036        .and_then(|hex_str| hex::decode(&hex_str).ok())
1037        .and_then(|bytes| bytes.try_into().ok());
1038
1039    // Get bech32m encoded strings
1040    let fvk = FullViewingKey::derive_from(&spend_key_bytes).map_err(|e| ErrorDisplayed::Str {
1041        s: format!("Failed to derive FVK: {e}"),
1042    })?;
1043    let wallet_id = fvk.wallet_id().map_err(|e| ErrorDisplayed::Str {
1044        s: format!("Failed to compute wallet ID: {e}"),
1045    })?;
1046
1047    let fvk_bech32m = fvk.to_bech32m().map_err(|e| ErrorDisplayed::Str {
1048        s: format!("Failed to encode FVK: {e}"),
1049    })?;
1050    let wallet_id_bech32m = wallet_id.to_bech32m().map_err(|e| ErrorDisplayed::Str {
1051        s: format!("Failed to encode wallet ID: {e}"),
1052    })?;
1053
1054    // ========================================================================
1055    // Build UR-encoded "penumbra-accounts" for hardware wallet QR compatibility
1056    //
1057    // CBOR tag 49301 = penumbra-accounts
1058    // CBOR tag 49302 = penumbra-full-viewing-key
1059    // Tag 49301 in hex = 0xC095, encoded as: 0xd9 0xc0 0x95
1060    // Tag 49302 in hex = 0xC096, encoded as: 0xd9 0xc0 0x96
1061    // ========================================================================
1062    // Pre-compute ZID for the inner CBOR map (best-effort, optional).
1063    let ur_zid: Option<[u8; 32]> = export_data.zid_pubkey;
1064
1065    let ur_string = {
1066        let mut cbor_data = Vec::new();
1067
1068        // PenumbraAccounts: map with 2 entries
1069        // CBOR: 0xa2 = map(2)
1070        cbor_data.push(0xa2);
1071
1072        // Key 1: wallet_id (32 bytes)
1073        // CBOR: 0x01 = uint(1), 0x58 0x20 = bytes(32)
1074        cbor_data.push(0x01);
1075        cbor_data.push(0x58); // bytes with 1-byte length
1076        cbor_data.push(0x20); // 32 bytes
1077        cbor_data.extend_from_slice(&wallet_id.to_bytes());
1078
1079        // Key 2: accounts array
1080        // CBOR: 0x02 = uint(2), 0x81 = array(1)
1081        cbor_data.push(0x02);
1082        cbor_data.push(0x81); // array(1) - single account
1083
1084        // Tagged PenumbraFullViewingKey (tag 49302 = 0xC096)
1085        // Tag encoding: 0xd9 means "tag with 2-byte value follows"
1086        // 49302 = 0xC096 -> high byte 0xC0, low byte 0x96
1087        cbor_data.push(0xd9);
1088        cbor_data.push(0xc0); // high byte: 49302 >> 8 = 192 = 0xc0
1089        cbor_data.push(0x96); // low byte: 49302 & 0xff = 150 = 0x96
1090
1091        // PenumbraFullViewingKey: map with 2..4 entries
1092        // (key 1 fvk, key 2 index, key 3 name optional, key 4 zid optional)
1093        let has_name = !label.is_empty();
1094        let has_zid = ur_zid.is_some();
1095        let map_entries: u8 = 2 + (has_name as u8) + (has_zid as u8);
1096        cbor_data.push(0xa0 + map_entries);
1097
1098        // Key 1: fvk (text string - bech32m encoded)
1099        cbor_data.push(0x01); // key = 1
1100        let fvk_bytes = fvk_bech32m.as_bytes();
1101        if fvk_bytes.len() < 24 {
1102            cbor_data.push(0x60 + fvk_bytes.len() as u8);
1103        } else if fvk_bytes.len() < 256 {
1104            cbor_data.push(0x78); // text with 1-byte length
1105            cbor_data.push(fvk_bytes.len() as u8);
1106        } else {
1107            cbor_data.push(0x79); // text with 2-byte length (big-endian)
1108            cbor_data.extend_from_slice(&(fvk_bytes.len() as u16).to_be_bytes());
1109        }
1110        cbor_data.extend_from_slice(fvk_bytes);
1111
1112        // Key 2: index (unsigned int)
1113        cbor_data.push(0x02); // key = 2
1114        if account_index < 24 {
1115            cbor_data.push(account_index as u8);
1116        } else if account_index < 256 {
1117            cbor_data.push(0x18);
1118            cbor_data.push(account_index as u8);
1119        } else {
1120            cbor_data.push(0x19);
1121            cbor_data.extend_from_slice(&(account_index as u16).to_be_bytes());
1122        }
1123
1124        // Key 3: name (optional text string)
1125        if has_name {
1126            cbor_data.push(0x03); // key = 3
1127            let label_bytes = label.as_bytes();
1128            if label_bytes.len() < 24 {
1129                cbor_data.push(0x60 + label_bytes.len() as u8);
1130            } else {
1131                cbor_data.push(0x78);
1132                cbor_data.push(label_bytes.len() as u8);
1133            }
1134            cbor_data.extend_from_slice(label_bytes);
1135        }
1136
1137        // Key 4: zid_pubkey (optional 32-byte string) — canonical device identity
1138        // for zafu dedup across network imports. Unknown to legacy parsers.
1139        if let Some(zid) = ur_zid {
1140            cbor_data.push(0x04); // key = 4
1141            cbor_data.push(0x58); // bytes with 1-byte length
1142            cbor_data.push(0x20); // 32 bytes
1143            cbor_data.extend_from_slice(&zid);
1144        }
1145
1146        // Encode CBOR as UR string
1147        // Result format: "ur:penumbra-accounts/..."
1148        ur::ur::encode(&cbor_data, &ur::Type::Custom("penumbra-accounts"))
1149    };
1150
1151    Ok(PenumbraFvkExport {
1152        account_index,
1153        label: label_opt.unwrap_or_default(),
1154        fvk_bech32m,
1155        wallet_id_bech32m,
1156        qr_data: export_data.encode_qr(),
1157        ur_string,
1158    })
1159}

1177        derive_cosmos_account_xpub, derive_cosmos_key, PREFIX_CELESTIA, PREFIX_NOBLE,

1187    let pubkey_hex = hex::encode(&key.public_key);
1188
1189    // Change-level xpub (m/44'/118'/account'/0). Lets the hot wallet derive
1190    // burner receive addresses at .../0/i locally, watch-only, each still
1191    // signable on this device at the matching address_index and recoverable
1192    // from the seed. This is the rotation backbone - see the note in
1193    // db_handling::cosmos::derive_cosmos_account_xpub.
1194    let xpub =
1195        derive_cosmos_account_xpub(seed_phrase, SLIP0044_COSMOS, account_index).map_err(|e| {
1196            ErrorDisplayed::Str {
1197                s: format!("Failed to derive Cosmos xpub: {e}"),
1198            }
1199        })?;
1200
1201    // Generate addresses for supported chains (filter by network_name if specified)
1202    let all_chains: Vec<(&str, &str)> = vec![
1203        ("osmosis", PREFIX_OSMOSIS),
1204        ("noble", PREFIX_NOBLE),
1205        ("celestia", PREFIX_CELESTIA),
1206    ];
1207
1208    let chains: Vec<(&str, &str)> = if network_name.is_empty() {
1209        all_chains
1210    } else {
1211        let name = network_name.to_lowercase();
1212        all_chains
1213            .into_iter()
1214            .filter(|(id, _)| *id == name)
1215            .collect()
1216    };
1217
1218    let mut addresses = Vec::new();
1219    for (chain_id, prefix) in &chains {
1220        let addr = key
1221            .bech32_address(prefix)
1222            .map_err(|e| ErrorDisplayed::Str {
1223                s: format!("Failed to encode {chain_id} address: {e}"),
1224            })?;
1225        addresses.push(CosmosChainAddress {
1226            chain_id: chain_id.to_string(),
1227            address: addr,
1228            prefix: prefix.to_string(),
1229        });
1230    }
1231
1232    // Build QR data as JSON for Zafu import
1233    let label_str = if label.is_empty() { "Zigner" } else { label };
1234    let json = serde_json::json!({
1235        "type": "cosmos-accounts",
1236        "version": 1,
1237        "label": label_str,
1238        "account_index": account_index,
1239        "public_key": pubkey_hex,
1240        "xpub": xpub,
1241        "addresses": addresses.iter().map(|a| {
1242            serde_json::json!({
1243                "chain_id": a.chain_id,
1244                "address": a.address,
1245                "prefix": a.prefix,
1246            })
1247        }).collect::<Vec<_>>(),
1248    });
1249
1250    let json_bytes = json.to_string().into_bytes();
1251    let qr_data = encode_to_qr(&json_bytes, false).map_err(|e| ErrorDisplayed::Str {
1252        s: format!("Failed to generate QR: {e}"),
1253    })?;
1254
1255    Ok(CosmosAccountExport {
1256        account_index,
1257        label: label_str.to_string(),
1258        public_key_hex: pubkey_hex,
1259        xpub,
1260        addresses,
1261        qr_data,
1262    })
1263}

1306/// Sign a Cosmos transaction and return 64-byte compact signature.
1307///
1308/// IMPORTANT: re-derives display fields from raw_qr_hex and verifies they
1309/// match what the user approved. This prevents a compromised hot wallet from
1310/// displaying one transaction but signing another.
1311fn sign_cosmos_transaction(
1312    seed_phrase: &str,
1313    request: CosmosSignRequest,
1314) -> Result<Vec<u8>, ErrorDisplayed> {
1315    use db_handling::cosmos::{derive_cosmos_key, SLIP0044_COSMOS};
1316    use transaction_signing::cosmos::{
1317        extract_signers, sign_cosmos_amino, CosmosSignDocDisplay,
1318        CosmosSignRequest as InternalRequest,
1319    };
1320
1321    // re-parse the QR to get the sign doc bytes
1322    let req =
1323        InternalRequest::from_qr_hex(&request.raw_qr_hex).map_err(|e| ErrorDisplayed::Str {
1324            s: format!("Failed to re-parse QR: {e}"),
1325        })?;
1326
1327    // re-derive display fields from the raw QR and verify they match
1328    // what the user was shown. this is the cosmos equivalent of
1329    // penumbra's verify_effect_hash — it binds display to signing.
1330    let display =
1331        CosmosSignDocDisplay::from_json(&req.sign_doc_bytes).map_err(|e| ErrorDisplayed::Str {
1332            s: format!("Failed to re-parse sign doc: {e}"),
1333        })?;
1334
1335    if display.chain_id != request.chain_id {
1336        return Err(ErrorDisplayed::Str {
1337            s: format!(
1338                "Chain ID mismatch: display showed '{}' but QR contains '{}'",
1339                request.chain_id, display.chain_id
1340            ),
1341        });
1342    }
1343    if display.fee != request.fee {
1344        return Err(ErrorDisplayed::Str {
1345            s: format!(
1346                "Fee mismatch: display showed '{}' but QR contains '{}'",
1347                request.fee, display.fee
1348            ),
1349        });
1350    }
1351    if display.memo != request.memo {
1352        return Err(ErrorDisplayed::Str {
1353            s: format!(
1354                "Memo mismatch: display showed '{}' but QR contains '{}'",
1355                request.memo, display.memo
1356            ),
1357        });
1358    }
1359    if display.msgs.len() != request.msgs.len() {
1360        return Err(ErrorDisplayed::Str {
1361            s: format!(
1362                "Message count mismatch: display showed {} but QR contains {}",
1363                request.msgs.len(),
1364                display.msgs.len()
1365            ),
1366        });
1367    }
1368
1369    // derive the cosmos key from seed phrase. BOTH indices come from the
1370    // re-parsed QR (`req`), not the caller's struct, so the signing key is bound
1371    // to the exact bytes the device displayed. address_index selects a fresh,
1372    // never-reused receive address; absent in old QRs it is 0.
1373    let key = derive_cosmos_key(
1374        seed_phrase,
1375        SLIP0044_COSMOS,
1376        req.account_index,
1377        req.address_index,
1378    )
1379    .map_err(|e| ErrorDisplayed::Str {
1380        s: format!("Key derivation failed: {e}"),
1381    })?;
1382
1383    // Bind the signing key to the transaction source. Derive this device's
1384    // address for the chain and require it to equal every message's signer.
1385    // Without this, an app-chosen `address_index` could make the device produce
1386    // a valid signature spending from a source address the user never saw.
1387    // Blind messages (no recognizable signer field) are already warned in the
1388    // display and the chain still rejects a signature from the wrong key.
1389    let device_addr =
1390        db_handling::cosmos::pubkey_to_bech32_address(&key.public_key, &request.chain_name)
1391            .map_err(|e| ErrorDisplayed::Str {
1392                s: format!("Address derivation failed: {e}"),
1393            })?;
1394    for signer in extract_signers(&req.sign_doc_bytes).map_err(|e| ErrorDisplayed::Str {
1395        s: format!("Failed to read message signers: {e}"),
1396    })? {
1397        if signer != device_addr {
1398            return Err(ErrorDisplayed::Str {
1399                s: format!(
1400                    "Source mismatch: this key is {device_addr}, but the transaction spends from {signer}"
1401                ),
1402            });
1403        }
1404    }
1405
1406    // sign with SHA256 prehash (NOT blake2b)
1407    let signature = sign_cosmos_amino(&key.secret_key, &req.sign_doc_bytes).map_err(|e| {
1408        ErrorDisplayed::Str {
1409            s: format!("Signing failed: {e}"),
1410        }
1411    })?;
1412
1413    Ok(signature.to_vec())
1414}

1440/// Sign a Penumbra transaction and return encoded signature QR bytes
1441fn sign_penumbra_transaction(
1442    seed_phrase: &str,
1443    request: PenumbraSignRequest,
1444) -> Result<Vec<u8>, ErrorDisplayed> {
1445    use transaction_parsing::penumbra::{
1446        parse_penumbra_transaction, sign_transaction, SpendKeyBytes,
1447    };
1448    use transaction_signing::penumbra::verify_effect_hash;
1449
1450    // Re-parse to get the full plan data
1451    let plan =
1452        parse_penumbra_transaction(&request.raw_qr_hex).map_err(|e| ErrorDisplayed::Str {
1453            s: format!("Failed to parse Penumbra QR: {e}"),
1454        })?;
1455
1456    let effect_hash = plan.effect_hash.ok_or_else(|| ErrorDisplayed::Str {
1457        s: "No effect hash in QR".to_string(),
1458    })?;
1459
1460    // Derive spend key from seed phrase (account 0)
1461    let spend_key =
1462        SpendKeyBytes::from_seed_phrase(seed_phrase, 0).map_err(|e| ErrorDisplayed::Str {
1463            s: format!("Key derivation failed: {e}"),
1464        })?;
1465
1466    // SECURITY: Verify the effect hash from the QR matches what we compute
1467    // from the plan + our FVK. This prevents a compromised hot wallet from
1468    // tricking us into signing a different transaction.
1469    verify_effect_hash(&plan.plan_bytes, &effect_hash, &spend_key).map_err(|e| {
1470        ErrorDisplayed::Str {
1471            s: format!("Effect hash verification failed: {e}"),
1472        }
1473    })?;
1474
1475    // Sign the transaction
1476    let auth_data = sign_transaction(
1477        effect_hash,
1478        &plan.spend_randomizers,
1479        &plan.delegator_vote_randomizers,
1480        &plan.lqt_vote_randomizers,
1481        &spend_key,
1482    )
1483    .map_err(|e| ErrorDisplayed::Str {
1484        s: format!("Signing failed: {e}"),
1485    })?;
1486
1487    // Encode as QR response bytes
1488    auth_data.encode().map_err(|e| ErrorDisplayed::Str {
1489        s: format!("Encode failed: {e}"),
1490    })
1491}

1497/// Export Zcash Orchard full viewing key for import into watch-only wallet (e.g., Zafu/Prax)
1498///
1499/// Returns both:
1500/// - UFVK string (unified full viewing key) for direct import
1501/// - UR-encoded string (zcash-accounts format) for Zashi/Keystone QR compatibility
1502///
1503/// ## UR (Uniform Resource) Protocol
1504///
1505/// UR is a protocol by Blockchain Commons for encoding binary data in QR codes.
1506/// Format: `ur:<type>/<bytewords-encoded-cbor>`
1507///
1508/// Reference: https://github.com/BlockchainCommons/Research/blob/master/papers/bcr-2020-005-ur.md
1509///
1510/// ## Keystone SDK Registry Types for Zcash
1511///
1512/// The Keystone SDK defines these UR types for Zcash (see keystone-sdk-rust):
1513///
1514/// | Type Name                       | CBOR Tag | Description                    |
1515/// |---------------------------------|----------|--------------------------------|
1516/// | zcash-accounts                  | 49201    | Collection of accounts/UFVKs   |
1517/// | zcash-full-viewing-key          | 49202    | Single FVK (deprecated)        |
1518/// | zcash-unified-full-viewing-key  | 49203    | UFVK with metadata             |
1519/// | zcash-pczt                      | 49204    | Partially Created Zcash Tx     |
1520///
1521/// ## CBOR Structure for zcash-accounts
1522///
1523/// ```text
1524/// ZcashAccounts = {
1525///   1: bytes,              ; seed_fingerprint (32-byte ZIP-32 SeedFingerprint)
1526///   2: [+ #49203(UFVK)]    ; accounts array, each tagged with 49203
1527/// }
1528///
1529/// ZcashUnifiedFullViewingKey (#49203) = {
1530///   1: tstr,               ; ufvk - the unified full viewing key string
1531///   2: uint,               ; index - account index (0, 1, 2, ...)
1532///   ? 3: tstr              ; name - optional account label
1533/// }
1534/// ```
1535///
1536/// ## CBOR Encoding Reference
1537///
1538/// CBOR major types (high 3 bits of first byte):
1539/// - 0x00-0x17: unsigned int 0-23 (value in low 5 bits)
1540/// - 0x18: unsigned int, 1 byte follows
1541/// - 0x19: unsigned int, 2 bytes follow (big-endian)
1542/// - 0x40-0x57: bytes, length 0-23 in low 5 bits
1543/// - 0x58: bytes, 1-byte length follows
1544/// - 0x60-0x77: text string, length 0-23 in low 5 bits
1545/// - 0x78: text string, 1-byte length follows
1546/// - 0x79: text string, 2-byte length follows
1547/// - 0x80-0x97: array, length 0-23 in low 5 bits
1548/// - 0xa0-0xb7: map, length 0-23 in low 5 bits (pairs count)
1549/// - 0xd8: tag, 1-byte tag number follows
1550/// - 0xd9: tag, 2-byte tag number follows (big-endian)
1551///
1552/// Parse a Zcash simple sign request from QR hex data.
1553/// This is the non-PCZT signing flow used by Zafu wallet.
1554fn parse_zcash_sign_request(qr_hex: &str) -> Result<ZcashSimpleSignRequest, ErrorDisplayed> {
1555    let request = transaction_signing::ZcashSignRequest::from_qr_hex(qr_hex).map_err(|e| {
1556        ErrorDisplayed::Str {
1557            s: format!("Failed to parse Zcash sign request: {e}"),
1558        }
1559    })?;
1560
1561    Ok(ZcashSimpleSignRequest {
1562        account_index: request.account_index,
1563        sighash_hex: hex::encode(request.sighash),
1564        action_count: request.orchard_alphas.len() as u32,
1565        summary: request.summary,
1566        mainnet: request.mainnet,
1567        raw_qr_hex: qr_hex.to_string(),
1568    })
1569}

1591// hand-rolls ZcashAccounts CBOR with per-byte annotated pushes (clearer than a
1592// dense vec![...] literal); silence the resulting vec-init-then-push lint.
1593#[allow(clippy::vec_init_then_push)]
1594fn export_zcash_fvk(
1595    seed_phrase: &str,
1596    account_index: u32,
1597    label: &str,
1598    mainnet: bool,
1599) -> Result<ZcashFvkExport, ErrorDisplayed> {
1600    use transaction_signing::zcash::{OrchardSpendingKey, QR_TYPE_ZCASH_FVK_EXPORT};
1601
1602    // Derive Orchard spending key from seed phrase using ZIP-32 derivation
1603    let osk = OrchardSpendingKey::from_seed_phrase(seed_phrase, account_index).map_err(|e| {
1604        ErrorDisplayed::Str {
1605            s: format!("Failed to derive Orchard key: {e}"),
1606        }
1607    })?;
1608
1609    // Get FVK bytes (96 bytes raw orchard full viewing key)
1610    let fvk_bytes = osk.fvk_bytes();
1611
1612    // Get receiving address (unified address with orchard receiver)
1613    let address = osk.get_address(mainnet);
1614
1615    // Get UFVK string with orchard + transparent components (ZIP-316)
1616    // Format: "uview1..." for mainnet, "uviewtest1..." for testnet
1617    let ufvk = OrchardSpendingKey::get_ufvk_with_transparent(seed_phrase, account_index, mainnet)
1618        .map_err(|e| ErrorDisplayed::Str {
1619        s: format!("Failed to derive UFVK: {e}"),
1620    })?;
1621
1622    // ZIP-32 seed fingerprint (32 bytes): BLAKE2b-256 personalized
1623    // "Zcash_HD_Seed_FP" over the same 64-byte BIP39 seed used for key
1624    // derivation. Matches Keystone/Zashi/vizor so they recognize this account as
1625    // belonging to the same seed. (Previously SHA256(mnemonic)[..16] - wrong
1626    // length AND wrong preimage, which broke import into Keystone-compatible
1627    // wallets with "seed fingerprint must be 32 bytes".)
1628    let seed_fingerprint =
1629        OrchardSpendingKey::seed_fingerprint(seed_phrase).map_err(|e| ErrorDisplayed::Str {
1630            s: format!("Failed to derive seed fingerprint: {e}"),
1631        })?;
1632
1633    // ========================================================================
1634    // Build UR-encoded "zcash-accounts" for Zashi/Keystone QR code compatibility
1635    //
1636    // The UR string format is: ur:zcash-accounts/<bytewords-encoded-cbor>
1637    // where the CBOR payload matches the Keystone SDK zcash_accounts.rs structure
1638    // ========================================================================
1639    // Try to derive ZID early so we can include it as a CBOR field below.
1640    // Best-effort — if it fails, we just omit the field (backwards compatible).
1641    let ur_zid: Option<[u8; 32]> = auth::derive_zid_pubkey(seed_phrase)
1642        .ok()
1643        .and_then(|h| hex::decode(&h).ok())
1644        .and_then(|b| b.try_into().ok());
1645
1646    let ur_string = {
1647        // hand-rolled CBOR: the per-byte pushes below are annotated with their
1648        // CBOR meaning, which reads clearer than a dense vec![...] literal.
1649        let mut cbor_data = Vec::new();
1650
1651        // ZcashAccounts: map with 2 entries
1652        // CBOR: 0xa2 = map(2)
1653        cbor_data.push(0xa2);
1654
1655        // Key 1: seed_fingerprint (bytes)
1656        // CBOR: 0x01 = uint(1), then bytes(32): 0x58 (bytes, 1-byte length) 0x20 (=32)
1657        cbor_data.push(0x01);
1658        cbor_data.push(0x58); // bytes with 1-byte length follows
1659        cbor_data.push(0x20); // length = 32
1660        cbor_data.extend_from_slice(&seed_fingerprint);
1661
1662        // Key 2: accounts array
1663        // CBOR: 0x02 = uint(2), 0x81 = array(1)
1664        cbor_data.push(0x02);
1665        cbor_data.push(0x81); // array(1) - single account
1666
1667        // Each account is tagged with CBOR tag 49203 (ZCASH_UNIFIED_FULL_VIEWING_KEY)
1668        // Tag 49203 = 0xC033 in hex, encoded as: 0xd9 0xc0 0x33
1669        // 0xd9 means "tag with 2-byte value follows"
1670        cbor_data.push(0xd9);
1671        cbor_data.push(0xc0); // high byte: 49203 >> 8 = 192 = 0xc0
1672        cbor_data.push(0x33); // low byte: 49203 & 0xff = 51 = 0x33
1673
1674        // ZcashUnifiedFullViewingKey: map with 2..4 entries
1675        // (key 1 ufvk, key 2 index, key 3 name optional, key 4 zid optional)
1676        let has_name = !label.is_empty();
1677        let has_zid = ur_zid.is_some();
1678        let map_entries: u8 = 2 + (has_name as u8) + (has_zid as u8);
1679        cbor_data.push(0xa0 + map_entries); // 0xa2 / 0xa3 / 0xa4
1680
1681        // Key 1: ufvk (text string)
1682        // CBOR text string encoding: 0x60-0x77 for len 0-23, 0x78+len for len<256, 0x79+2bytes for len<65536
1683        cbor_data.push(0x01); // key = 1
1684        let ufvk_bytes = ufvk.as_bytes();
1685        if ufvk_bytes.len() < 24 {
1686            cbor_data.push(0x60 + ufvk_bytes.len() as u8);
1687        } else if ufvk_bytes.len() < 256 {
1688            cbor_data.push(0x78); // text with 1-byte length
1689            cbor_data.push(ufvk_bytes.len() as u8);
1690        } else {
1691            cbor_data.push(0x79); // text with 2-byte length (big-endian)
1692            cbor_data.extend_from_slice(&(ufvk_bytes.len() as u16).to_be_bytes());
1693        }
1694        cbor_data.extend_from_slice(ufvk_bytes);
1695
1696        // Key 2: index (unsigned int)
1697        // CBOR uint encoding: 0-23 inline, 0x18+byte for 24-255, 0x19+2bytes for 256-65535
1698        cbor_data.push(0x02); // key = 2
1699        if account_index < 24 {
1700            cbor_data.push(account_index as u8);
1701        } else if account_index < 256 {
1702            cbor_data.push(0x18);
1703            cbor_data.push(account_index as u8);
1704        } else {
1705            cbor_data.push(0x19);
1706            cbor_data.extend_from_slice(&(account_index as u16).to_be_bytes());
1707        }
1708
1709        // Key 3: name (optional text string)
1710        if has_name {
1711            cbor_data.push(0x03); // key = 3
1712            let label_bytes = label.as_bytes();
1713            if label_bytes.len() < 24 {
1714                cbor_data.push(0x60 + label_bytes.len() as u8);
1715            } else {
1716                cbor_data.push(0x78);
1717                cbor_data.push(label_bytes.len() as u8);
1718            }
1719            cbor_data.extend_from_slice(label_bytes);
1720        }
1721
1722        // Key 4: zid_pubkey (optional 32-byte string) — canonical device identity
1723        // for zafu dedup across network imports. Unknown to legacy parsers (Zashi,
1724        // Keystone) which simply ignore unrecognized keys.
1725        if let Some(zid) = ur_zid {
1726            cbor_data.push(0x04); // key = 4
1727            cbor_data.push(0x58); // bytes with 1-byte length
1728            cbor_data.push(0x20); // 32 bytes
1729            cbor_data.extend_from_slice(&zid);
1730        }
1731
1732        // Encode CBOR as UR string using bytewords encoding
1733        // Result format: "ur:zcash-accounts/..."
1734        ur::ur::encode(&cbor_data, &ur::Type::Custom("zcash-accounts"))
1735    };
1736
1737    // Build binary QR data for Zafu wallet
1738    // Format must match Zafu's parseZcashFvkQR():
1739    // [0x53][0x04][0x01][flags:1][account:4 LE][label_len:1][label][orchard_fvk:96][addr_len:2 LE][address]
1740    let label_bytes = label.as_bytes();
1741    let label_len = label_bytes.len().min(255) as u8;
1742    let address_bytes = address.as_bytes();
1743
1744    // Try to derive ZID pubkey (32 bytes) — lets zafu link this wallet to the
1745    // same zigner device across network imports for dedup. Best-effort: if it
1746    // fails for any reason, omit the ZID bytes rather than failing FVK export.
1747    let zid_bytes: Option<[u8; 32]> = auth::derive_zid_pubkey(seed_phrase)
1748        .ok()
1749        .and_then(|hex_str| hex::decode(&hex_str).ok())
1750        .and_then(|bytes| bytes.try_into().ok());
1751
1752    // flags: bit 0 = mainnet, bit 1 = has orchard, bit 2 = has transparent,
1753    //        bit 3 = has address, bit 4 = has ZID pubkey (32 bytes appended)
1754    let zid_flag: u8 = if zid_bytes.is_some() { 0x10 } else { 0x00 };
1755    let flags: u8 = (if mainnet { 0x01 } else { 0x00 }) | 0x02 | 0x08 | zid_flag;
1756
1757    let mut qr_data =
1758        Vec::with_capacity(3 + 1 + 4 + 1 + label_bytes.len() + 96 + 2 + address_bytes.len() + 32);
1759    qr_data.push(0x53); // 'S' for Signer (substrate compat)
1760    qr_data.push(0x04); // Zcash chain ID
1761    qr_data.push(QR_TYPE_ZCASH_FVK_EXPORT);
1762    qr_data.push(flags);
1763    qr_data.extend_from_slice(&account_index.to_le_bytes());
1764    qr_data.push(label_len);
1765    qr_data.extend_from_slice(&label_bytes[..label_len as usize]);
1766    qr_data.extend_from_slice(&fvk_bytes);
1767    // Add unified address (bit 3 flag)
1768    qr_data.extend_from_slice(&(address_bytes.len() as u16).to_le_bytes());
1769    qr_data.extend_from_slice(address_bytes);
1770    // Add ZID pubkey (bit 4 flag) — canonical device identity
1771    if let Some(zid) = zid_bytes {
1772        qr_data.extend_from_slice(&zid);
1773    }
1774
1775    Ok(ZcashFvkExport {
1776        account_index,
1777        label: label.to_string(),
1778        mainnet,
1779        address,
1780        fvk_hex: hex::encode(fvk_bytes),
1781        ufvk,
1782        qr_data,
1783        ur_string,
1784    })
1785}

1831// ============================================================================
1832// PCZT (Partially Created Zcash Transaction) functions for Zashi compatibility
1833// ============================================================================
1834//
1835// PCZT is the standard format for passing unsigned Zcash transactions between
1836// watch-only wallets (like Zashi) and hardware signers (like Zigner/Keystone).
1837//
1838// UR (Uniform Resource) encoding:
1839// - Type: zcash-pczt (CBOR tag 49204)
1840// - Format: ur:zcash-pczt/<bytewords-encoded-cbor>
1841// - For large PCZTs, UR supports animated QR sequences (fountain codes)
1842//
1843// Flow:
1844// 1. Zashi creates PCZT, encodes as ur:zcash-pczt, shows animated QR
1845// 2. Zigner scans QR sequence, decodes UR, parses PCZT
1846// 3. Zigner extracts sighash and rsk randomizers, signs with spending key
1847// 4. Zigner injects signatures into PCZT, encodes as ur:zcash-pczt
1848// 5. Zashi scans signed PCZT QR, extracts signatures, broadcasts tx
1849
1850/// Decode UR string to get PCZT bytes
1851/// Handles both single QR and animated QR sequences
1852fn decode_ur_zcash_pczt(ur_parts: Vec<String>) -> Result<Vec<u8>, ErrorDisplayed> {
1853    use ur::ur::decode;
1854
1855    if ur_parts.is_empty() {
1856        return Err(ErrorDisplayed::Str {
1857            s: "No UR parts provided".to_string(),
1858        });
1859    }
1860
1861    // Verify UR type from the URI string (format: ur:type/data)
1862    fn verify_ur_type(ur_string: &str) -> Result<(), ErrorDisplayed> {
1863        let lower = ur_string.to_lowercase();
1864        if !lower.starts_with("ur:zcash-pczt/") {
1865            return Err(ErrorDisplayed::Str {
1866                s: format!(
1867                    "Expected ur:zcash-pczt/... got: {}",
1868                    ur_string.chars().take(30).collect::<String>()
1869                ),
1870            });
1871        }
1872        Ok(())
1873    }
1874
1875    // For single-part UR, decode directly
1876    if ur_parts.len() == 1 {
1877        verify_ur_type(&ur_parts[0])?;
1878
1879        let (_kind, cbor_data) = decode(&ur_parts[0]).map_err(|e| ErrorDisplayed::Str {
1880            s: format!("Failed to decode UR: {:?}", e),
1881        })?;
1882
1883        // Extract PCZT bytes from CBOR
1884        // CBOR structure: { 1: bytes } (map with key 1 containing the PCZT bytes)
1885        extract_pczt_from_cbor(&cbor_data)
1886    } else {
1887        // For multi-part (animated) UR, use fountain decoder
1888        let mut decoder = ur::ur::Decoder::default();
1889
1890        // Verify type from first part
1891        verify_ur_type(&ur_parts[0])?;
1892
1893        for part in &ur_parts {
1894            decoder.receive(part).map_err(|e| ErrorDisplayed::Str {
1895                s: format!("Failed to receive UR part: {:?}", e),
1896            })?;
1897
1898            if decoder.complete() {
1899                break;
1900            }
1901        }
1902
1903        if !decoder.complete() {
1904            return Err(ErrorDisplayed::Str {
1905                s: format!(
1906                    "Incomplete UR sequence: received {} parts but not complete",
1907                    ur_parts.len()
1908                ),
1909            });
1910        }
1911
1912        let cbor_data = decoder
1913            .message()
1914            .map_err(|e| ErrorDisplayed::Str {
1915                s: format!("Failed to get UR message: {:?}", e),
1916            })?
1917            .ok_or_else(|| ErrorDisplayed::Str {
1918                s: "UR decoder returned None despite being complete".to_string(),
1919            })?;
1920
1921        extract_pczt_from_cbor(&cbor_data)
1922    }
1923}

1925/// Extract PCZT bytes from CBOR wrapper
1926/// CBOR structure: { 1: bytes } - map with key 1 containing raw PCZT bytes
1927fn extract_pczt_from_cbor(cbor_data: &[u8]) -> Result<Vec<u8>, ErrorDisplayed> {
1928    // Simple CBOR parsing for { 1: bytes }
1929    // 0xa1 = map(1), 0x01 = key 1, then bytes
1930    if cbor_data.len() < 3 {
1931        return Err(ErrorDisplayed::Str {
1932            s: "CBOR data too short".to_string(),
1933        });
1934    }
1935
1936    // Check for map(1)
1937    if cbor_data[0] != 0xa1 {
1938        return Err(ErrorDisplayed::Str {
1939            s: format!("Expected CBOR map(1), got: 0x{:02x}", cbor_data[0]),
1940        });
1941    }
1942
1943    // Check for key 1
1944    if cbor_data[1] != 0x01 {
1945        return Err(ErrorDisplayed::Str {
1946            s: format!("Expected CBOR key 1, got: 0x{:02x}", cbor_data[1]),
1947        });
1948    }
1949
1950    // Parse bytes at position 2
1951    let (bytes, _) = parse_cbor_bytes(&cbor_data[2..]).map_err(|e| ErrorDisplayed::Str {
1952        s: format!("Failed to parse CBOR bytes: {}", e),
1953    })?;
1954
1955    Ok(bytes)
1956}

1958/// Parse CBOR bytes (major type 2)
1959fn parse_cbor_bytes(data: &[u8]) -> Result<(Vec<u8>, usize), String> {
1960    if data.is_empty() {
1961        return Err("Empty CBOR data".to_string());
1962    }
1963
1964    let first = data[0];
1965    let major_type = first >> 5;
1966    let additional = first & 0x1f;
1967
1968    if major_type != 2 {
1969        return Err(format!(
1970            "Expected bytes (major type 2), got: {}",
1971            major_type
1972        ));
1973    }
1974
1975    let (len, header_len) = match additional {
1976        0..=23 => (additional as usize, 1),
1977        24 => {
1978            if data.len() < 2 {
1979                return Err("Truncated CBOR length".to_string());
1980            }
1981            (data[1] as usize, 2)
1982        }
1983        25 => {
1984            if data.len() < 3 {
1985                return Err("Truncated CBOR length".to_string());
1986            }
1987            (u16::from_be_bytes([data[1], data[2]]) as usize, 3)
1988        }
1989        26 => {
1990            if data.len() < 5 {
1991                return Err("Truncated CBOR length".to_string());
1992            }
1993            (
1994                u32::from_be_bytes([data[1], data[2], data[3], data[4]]) as usize,
1995                5,
1996            )
1997        }
1998        _ => return Err(format!("Unsupported CBOR additional info: {}", additional)),
1999    };
2000
2001    if data.len() < header_len + len {
2002        return Err("Truncated CBOR bytes".to_string());
2003    }
2004
2005    Ok((
2006        data[header_len..header_len + len].to_vec(),
2007        header_len + len,
2008    ))
2009}

2051/// Inspect a PCZT: extract spend/output details, cross-reference against verified notes.
2052/// Defense-in-depth caps on PCZT parsing. The pczt crate uses postcard for
2053/// deserialization, which doesn't bound declared collection lengths against
2054/// remaining slice length — a hostile PCZT can claim a Vec<Action> of 2^32
2055/// items and trigger allocation-driven OOM. We reject obviously-bogus inputs
2056/// upfront. Real Orchard bundles top out in the low tens of actions.
2057const MAX_PCZT_BYTES: usize = 1024 * 1024; // 1 MiB — far above any real PCZT
2058const MAX_ORCHARD_ACTIONS: usize = 4096; // far above any real bundle (~tens)
2059
2060/// Confirm that the recipient + amount the review screen is about to DISPLAY
2061/// are the ones the transaction actually pays.
2062///
2063/// A cold wallet's whole value is that the screen cannot lie. The `recipient`
2064/// and `value` fields of an orchard output are plaintext metadata the wallet
2065/// puts in the PCZT for exactly this display; nothing in the signing path
2066/// reads them, so on their own a hostile wallet could show any recipient and
2067/// any amount while the transaction pays somebody else. What the transaction
2068/// is actually bound to is the note commitment `cmx`, which is committed to by
2069/// the sighash we sign.
2070///
2071/// `Output::verify_note_commitment` recomputes `cmx` from
2072/// (recipient, value, rho = nf_old, rseed) and compares. zafu's
2073/// `redact_pczt_for_signer` deliberately KEEPS all of those on the output side
2074/// (only spend-side note plaintext and the fvk are stripped), so this check
2075/// runs on exactly the bytes that cross the airgap.
2076///
2077/// The rule is keyed on what is DISPLAYED, not on which field happens to be
2078/// present: **EVERY output must be proven against its note commitment. There
2079/// is no skip.**
2080///
2081/// The original version of this gate skipped whenever
2082/// `verify_note_commitment` reported `MissingRecipient` / `MissingValue` /
2083/// `MissingRandomSeed` — "a producer that withholds a field gets no display
2084/// guarantee, but also no false alarm". That made the control bypassable by an
2085/// ATTACKER rather than merely waived by an honest producer: the signing path
2086/// never reads the output `rseed` (`pczt`'s low-level signer sets
2087/// `rseed: None` itself, and the signature is over a sighash that binds the
2088/// real `cmx`), so a hostile wallet could strip the output `rseed`, set
2089/// `recipient` and `value` to whatever the victim expected to see, have the
2090/// device display those and sign, then reassemble with the true `rseed` and
2091/// broadcast. The review screen reads `value()`/`recipient()` directly
2092/// regardless of whether verification ran, so the skip was a silent no-op on
2093/// attacker-chosen input.
2094///
2095/// An intermediate fix keyed the skip on whether anything was DISPLAYED —
2096/// refuse an unprovable output that renders a recipient or a non-zero amount,
2097/// allow one that renders nothing. That escape hatch was itself reachable by a
2098/// hostile producer: strip `recipient`, `value` AND `rseed` from every output
2099/// and the review screen renders a blank recipient and 0 for each of them
2100/// while the transaction still pays a real amount bound by `cmx`, with a
2101/// plausible producer-supplied fee alongside. A device that displays "nothing,
2102/// 0" and then signs a real payment has failed at its only job.
2103///
2104/// So there is no skip. Any output this device cannot prove — for any reason,
2105/// including a withheld field — is a REFUSAL naming the action index and the
2106/// reason. zafu's `redact_pczt_for_signer` keeps `recipient`, `value` and
2107/// `rseed` on every output, dummy/padding actions included, and all of them
2108/// verify, so an honestly-redacted PCZT is unaffected (pinned by the
2109/// `zafu_redacted_*` fixtures). A third-party producer that redacts output
2110/// plaintext is refused — loudly, at review time, with the failing action
2111/// named. That is the correct failure direction: the alternative fails
2112/// silently and invisibly, which is how the original bug survived.
2113///
2114/// Still a silent no-op when the bundle cannot be reached at all (unparsable,
2115/// unknown consensus branch) — signing fails on those paths anyway.
2116///
2117/// The second half of this gate covers the FEE — see [`verify_value_balance`].
2118fn verify_displayed_summary(pczt: &pczt::Pczt) -> Result<(), ErrorDisplayed> {
2119    use pczt::roles::verifier::{OrchardError, Verifier};
2120
2121    fn check(bundle: &orchard::pczt::Bundle) -> Result<(), OrchardError<String>> {
2122        for (index, action) in bundle.actions().iter().enumerate() {
2123            match action.output().verify_note_commitment(action.spend()) {
2124                Ok(()) => {}
2125                // Not verifiable because the producer withheld a field the
2126                // check needs. Unconditional refusal — see above.
2127                Err(
2128                    e @ (orchard::pczt::VerifyError::MissingRecipient
2129                    | orchard::pczt::VerifyError::MissingValue
2130                    | orchard::pczt::VerifyError::MissingRandomSeed),
2131                ) => {
2132                    return Err(OrchardError::Custom(format!(
2133                        "output cannot be proven against its note commitment \
2134                         (action {index}: {e:?})"
2135                    )))
2136                }
2137                // A PROVEN lie.
2138                Err(e) => return Err(OrchardError::Verify(e)),
2139            }
2140        }
2141        verify_value_balance(bundle).map_err(OrchardError::Custom)
2142    }
2143
2144    let outcome = Verifier::new(pczt.clone())
2145        .with_orchard(check)
2146        .and_then(|v| v.with_ironwood(check));
2147
2148    match outcome {
2149        Ok(_) => Ok(()),
2150        // A PROVEN lie: the displayed recipient/amount do not produce the note
2151        // commitment this transaction pays out to. Refuse.
2152        Err(OrchardError::Verify(e)) => Err(ErrorDisplayed::Str {
2153            s: format!(
2154                "PCZT output does not match what it claims to pay ({e:?}) — \
2155                 refusing to display or sign it."
2156            ),
2157        }),
2158        // Something the review screen would show that this device cannot
2159        // prove: an output missing a field the commitment check needs, or a
2160        // declared value balance the commitments contradict.
2161        Err(OrchardError::Custom(reason)) => Err(ErrorDisplayed::Str {
2162            s: format!("PCZT {reason} — refusing to display or sign it."),
2163        }),
2164        // Not verifiable at all (structural parse, unrecognised consensus
2165        // branch). This was `Ok(())` - "signing fails later anyway" - but the
2166        // review screen would already be showing outputs nobody proved.
2167        // zcash_consensus_gate rejects the known cases first; this is the
2168        // backstop.
2169        Err(e) => Err(ErrorDisplayed::Str {
2170            s: format!("PCZT cannot be verified ({e:?}) — refusing to display or sign it."),
2171        }),
2172    }
2173}

2175/// Confirm that a bundle's declared `value_sum` — the number the displayed FEE
2176/// is computed from — matches the action value commitments.
2177///
2178/// `value_sum` is plaintext metadata the producer writes into the PCZT.
2179/// Nothing in the note-commitment check above constrains it, so on its own a
2180/// hostile wallet could pay the user's expected recipient the expected amount
2181/// (both now provably displayed) while declaring a `value_sum` that renders a
2182/// trivial fee, and route the rest of a large spent note to the miner. An
2183/// inflated fee is fund loss, not a cosmetic error.
2184///
2185/// It is provable without any spend-side note plaintext — which matters,
2186/// because zafu's redaction strips ALL of it (spend `value`, `recipient`,
2187/// `rho`, `rseed` and the `fvk`), so `Spend::verify_nullifier` and
2188/// `Action::verify_cv_net` are both unusable here and a
2189/// `fee = Σ spend − Σ output` derivation is impossible. What survives is the
2190/// relation the IO Finalizer established and consensus later re-checks via the
2191/// binding signature:
2192///
2193/// ```text
2194///     Σ cv_net − ValueCommit(value_sum, 0) == VerificationKey::from(bsk)
2195/// ```
2196///
2197/// `cv_net` is present on every action and is covered by the sighash this
2198/// device signs; `bsk` survives redaction (zafu clears spend note plaintext
2199/// and the fvk, not the binding key). Forging a `bsk'` for a different
2200/// `value_sum'` means finding the discrete log of `[value_sum − value_sum']V`
2201/// in base `R`, so a producer cannot move the declared balance and stay
2202/// consistent. This is a genuinely independent derivation, not the same
2203/// metadata re-read by another route.
2204///
2205/// A missing `bsk` is a REFUSAL, not a skip — otherwise withholding one field
2206/// would disable this check exactly the way withholding `rseed` disabled the
2207/// output check. An action-less bundle (the canonical-empty Ironwood bundle a
2208/// V5 PCZT carries) has no commitments to check against, so its `value_sum`
2209/// must be exactly zero; a non-zero one would shift the displayed fee with
2210/// nothing backing it, since `fee_zat` sums the ironwood term unconditionally.
2211///
2212/// SCOPE: this binds the orchard and ironwood terms of the fee, which for a
2213/// shielded-only transaction is the entire fee. The transparent and sapling
2214/// terms are NOT bound — see the note on `fee_zat` in `inspect_zcash_pczt`.
2215fn verify_value_balance(bundle: &orchard::pczt::Bundle) -> Result<(), String> {
2216    use orchard::value::{ValueCommitTrapdoor, ValueCommitment};
2217
2218    if bundle.actions().is_empty() {
2219        return if *bundle.value_sum() == orchard::value::ValueSum::default() {
2220            Ok(())
2221        } else {
2222            Err(format!(
2223                "bundle declares a value balance of {:?} but has no actions to back it",
2224                bundle.value_sum()
2225            ))
2226        };
2227    }
2228
2229    let bsk = bundle.bsk().as_ref().ok_or_else(|| {
2230        "declared value balance (the displayed fee) cannot be proven: binding key absent"
2231            .to_string()
2232    })?;
2233
2234    // ValueCommit(v, 0) — the zero trapdoor is not public API, but a 32-byte
2235    // zero scalar is the same thing.
2236    let zero_rcv = ValueCommitTrapdoor::from_bytes([0u8; 32])
2237        .into_option()
2238        .ok_or_else(|| "zero value-commitment trapdoor".to_string())?;
2239    let sum_cv: ValueCommitment = bundle.actions().iter().map(|a| a.cv_net()).sum();
2240    let bvk = (sum_cv - ValueCommitment::derive(*bundle.value_sum(), zero_rcv)).to_bytes();
2241    let expected: [u8; 32] = (&orchard::primitives::redpallas::VerificationKey::from(bsk)).into();
2242
2243    if bvk == expected {
2244        Ok(())
2245    } else {
2246        Err(format!(
2247            "declared value balance {:?} (which the displayed fee is derived from) \
2248             does not match the action value commitments",
2249            bundle.value_sum()
2250        ))
2251    }
2252}

2266    // The pczt parse + every subsequent access can panic on adversarial input
2267    // (declared collection lengths > remaining slice, integer overflow on
2268    // count fields). Catch the panic so the UDL surface returns a clean
2269    // ErrorDisplayed instead of unwinding into the Kotlin/Swift FFI boundary.
2270    let pczt = std::panic::catch_unwind(|| Pczt::parse(&pczt_bytes))
2271        .map_err(|_| ErrorDisplayed::Str {
2272            s: "Panic during PCZT parse — input is malformed".to_string(),
2273        })?
2274        .map_err(|e| ErrorDisplayed::Str {
2275            s: format!("Failed to parse PCZT: {:?}", e),
2276        })?;
2277
2278    let orchard = pczt.orchard();
2279    let action_count_raw = orchard.actions().len();
2280    if action_count_raw > MAX_ORCHARD_ACTIONS {
2281        return Err(ErrorDisplayed::Str {
2282            s: format!(
2283                "Orchard bundle has {action_count_raw} actions, exceeds cap {MAX_ORCHARD_ACTIONS}"
2284            ),
2285        });
2286    }
2287    let action_count = action_count_raw as u32;
2288
2289    // Ironwood (NU6.3 / V6) is a second, orchard-shaped bundle. A V5 PCZT
2290    // carries the canonical EMPTY ironwood bundle, so everything below is a
2291    // no-op there and every ironwood-derived value is 0 — the V5 inspection
2292    // is unchanged. The same allocation cap applies.
2293    let ironwood = pczt.ironwood();
2294    let ironwood_count_raw = ironwood.actions().len();
2295    if ironwood_count_raw > MAX_ORCHARD_ACTIONS {
2296        return Err(ErrorDisplayed::Str {
2297            s: format!(
2298                "Ironwood bundle has {ironwood_count_raw} actions, exceeds cap {MAX_ORCHARD_ACTIONS}"
2299            ),
2300        });
2301    }
2302    let ironwood_action_count = ironwood_count_raw as u32;
2303
2304    // Screen-honesty gate: everything below feeds the review screen, so refuse
2305    // before building a summary that provably misstates what is being paid.
2306    zcash_consensus_gate::check_supported(&pczt)?;
2307    verify_displayed_summary(&pczt)?;
2308
2309    // Load verified notes for cross-reference. Used by the known_spends warning
2310    // (zigner's distinguishing safety guard over a Keystone-equivalent signer —
2311    // we can warn when the PCZT spends notes we haven't verified) and to derive
2312    // the network for recipient-address encoding.
2313    //
2314    // The verified_anchor was previously used for byte-equality matching against
2315    // the PCZT's anchor; that check was rotko-invented and doesn't exist in the
2316    // canonical Zashi → Keystone flow, so it's gone. The DB still stores the
2317    // anchor for the note-sync UI on a separate screen; the PCZT path doesn't
2318    // consult it for signing decisions.
2319    let (verified_balance, verified_nullifiers, verified_notes_values, is_mainnet) = {
2320        let db_guard = DB.read().map_err(|_| ErrorDisplayed::MutexPoisoned)?;
2321        if let Some(database) = db_guard.as_ref() {
2322            let balance = db_handling::zcash::get_verified_balance(database).unwrap_or(0);
2323            // Anchor is read only to recover the network flag; the bytes are
2324            // discarded. Defaults to mainnet when no notes have been synced.
2325            let mainnet = db_handling::zcash::get_verified_anchor(database)
2326                .ok()
2327                .flatten()
2328                .map(|(_, _, m, _, _)| m)
2329                .unwrap_or(true);
2330            let notes = db_handling::zcash::get_verified_notes(database).unwrap_or_default();
2331            let nullifiers: std::collections::HashSet<String> = notes
2332                .iter()
2333                .map(|(_, nf_hex, _, _, _)| nf_hex.clone())
2334                .collect();
2335            // Map nullifier_hex → value for spend amount lookup
2336            let values: std::collections::HashMap<String, u64> = notes
2337                .iter()
2338                .map(|(val, nf_hex, _, _, _)| (nf_hex.clone(), *val))
2339                .collect();
2340            (balance, nullifiers, values, mainnet)
2341        } else {
2342            (
2343                0,
2344                std::collections::HashSet::new(),
2345                std::collections::HashMap::new(),
2346                true,
2347            )
2348        }
2349    };
2350
2351    // Extract spend details. Known nullifiers get their value from our
2352    // verified store; unknown ones report 0 here (the pczt crate's Spend
2353    // doesn't expose per-spend value). The UI tells dummies from real
2354    // unverified notes via the value balance instead — orchard_net_value +
2355    // outputs - verified spends.
2356    let mut spends = Vec::new();
2357    let mut known_spends = 0u32;
2358    for action in orchard.actions() {
2359        let nullifier_hex = hex::encode(action.spend().nullifier());
2360        let known = verified_nullifiers.contains(&nullifier_hex);
2361        let value = if known {
2362            known_spends += 1;
2363            verified_notes_values
2364                .get(&nullifier_hex)
2365                .copied()
2366                .unwrap_or(0)
2367        } else {
2368            0
2369        };
2370        spends.push(ZcashPcztSpend {
2371            value,
2372            nullifier_hex,
2373            known,
2374        });
2375    }
2376    // Ironwood spends are appended to the SAME list: they are orchard-shaped,
2377    // the verified-note store is keyed by nullifier regardless of pool, and the
2378    // review screen must show every note the transaction consumes. Empty for V5.
2379    for action in ironwood.actions() {
2380        let nullifier_hex = hex::encode(action.spend().nullifier());
2381        let known = verified_nullifiers.contains(&nullifier_hex);
2382        let value = if known {
2383            known_spends += 1;
2384            verified_notes_values
2385                .get(&nullifier_hex)
2386                .copied()
2387                .unwrap_or(0)
2388        } else {
2389            0
2390        };
2391        spends.push(ZcashPcztSpend {
2392            value,
2393            nullifier_hex,
2394            known,
2395        });
2396    }
2397
2398    // Extract output details with human-readable addresses
2399    let mut outputs = Vec::new();
2400    for action in orchard.actions().iter().chain(ironwood.actions().iter()) {
2401        let value = action.output().value().unwrap_or(0);
2402        let recipient_hex = match action.output().recipient() {
2403            Some(raw_bytes) => {
2404                // Decode raw 43-byte Orchard address to unified address string.
2405                // Ironwood receivers are the same 43-byte raw orchard address,
2406                // so the same encoding applies.
2407                encode_orchard_recipient(raw_bytes, is_mainnet)
2408                    .unwrap_or_else(|| hex::encode(raw_bytes))
2409            }
2410            None => String::new(),
2411        };
2412        outputs.push(ZcashPcztOutput {
2413            value,
2414            recipient: recipient_hex,
2415        });
2416    }
2417
2418    // Orchard's value_sum = (total spent − total output) in zatoshis.
2419    // Sign-extend into i64 for fee math.
2420    let &(value_sum_magnitude, value_sum_is_negative) = orchard.value_sum();
2421    let orchard_net_value: i64 = if value_sum_is_negative {
2422        -(value_sum_magnitude as i64)
2423    } else {
2424        value_sum_magnitude as i64
2425    };
2426    let &(iw_magnitude, iw_is_negative) = ironwood.value_sum();
2427    let ironwood_net_value: i64 = if iw_is_negative {
2428        -(iw_magnitude as i64)
2429    } else {
2430        iw_magnitude as i64
2431    };
2432
2433    // True fee = Σ over ALL bundles of (in − out).
2434    // - transparent: explicit input.value − output.value sum
2435    // - sapling: bundle exposes value_sum (i128) — same sign convention
2436    // - orchard: orchard_net_value above
2437    // - ironwood: ironwood_net_value above
2438    //
2439    // For shielded-only txs the transparent + sapling terms are zero and the
2440    // fee equals orchard_net_value, so the old code accidentally worked.
2441    // For shielding / deshielding it didn't.
2442    //
2443    // The ironwood term is load-bearing for a turnstile migration: the migrated
2444    // value LEAVES the orchard pool, so orchard_net_value alone is ~the whole
2445    // migrated amount. Omitting the ironwood value_sum would display that as
2446    // the fee. It is exactly 0 for a V5 transaction, so V5 fees are unchanged.
2447    //
2448    // HOW FAR THIS FEE IS TRUSTED (read before relying on it, and note that an
2449    // inflated fee is fund loss, not a cosmetic error):
2450    //
2451    // Every term below is plaintext metadata in the PCZT, so for each one the
2452    // question is whether anything binds it to what will actually be mined.
2453    //
2454    // * orchard + ironwood: BOUND. `verify_value_balance`, called via
2455    //   `verify_displayed_summary` above before any of this runs, proves each
2456    //   bundle's value_sum against Σ cv_net and bsk; a producer cannot move it
2457    //   without solving a discrete log, and an action-less bundle must declare
2458    //   exactly zero. For a shielded-only transaction — the release-critical
2459    //   flow — that is the WHOLE fee, so the number displayed is trustworthy.
2460    // * transparent: NOT bound. `input.value` is taken at face value here.
2461    //   Those amounts are covered by the per-input transparent sighash we sign,
2462    //   but nothing cross-checks them before the review screen renders.
2463    // * sapling: NOT bound. The sapling bundle carries its own bsk and the same
2464    //   relation would prove it, but there is no sapling display path or
2465    //   fixture here, and the i128→i64 clamp below silently yields 0 on an
2466    //   out-of-range value_sum, which a hostile producer can force.
2467    //
2468    // So a PCZT carrying transparent or sapling components can still show a
2469    // partly producer-supplied fee. Extending the same binding to those
2470    // bundles is the next step if either becomes a supported flow.
2471    let transparent = pczt.transparent();
2472    let transparent_in: u64 = transparent.inputs().iter().map(|i| *i.value()).sum();
2473    let transparent_out: u64 = transparent.outputs().iter().map(|o| *o.value()).sum();
2474    let transparent_balance: i64 = (transparent_in as i64).saturating_sub(transparent_out as i64);
2475    // value_sum is i128 because intermediate arithmetic during PCZT
2476    // construction can exceed i64. A complete PCZT's value_sum always fits
2477    // in i64; clamp on overflow for safety.
2478    let sapling_v: i128 = *pczt.sapling().value_sum();
2479    let sapling_balance: i64 = if (i64::MIN as i128..=i64::MAX as i128).contains(&sapling_v) {
2480        sapling_v as i64
2481    } else {
2482        0
2483    };
2484    let fee_zat: i64 = transparent_balance
2485        .saturating_add(sapling_balance)
2486        .saturating_add(orchard_net_value)
2487        .saturating_add(ironwood_net_value);
2488
2489    // Global tx metadata. Cold device displays these so the user knows
2490    // which consensus fork + expiry window they're authorizing.
2491    let global = pczt.global();
2492    let tx_version: u32 = *global.tx_version();
2493    let consensus_branch_id_hex: String = format!("{:08x}", global.consensus_branch_id());
2494    let expiry_height: u32 = *global.expiry_height();
2495
2496    Ok(ZcashPcztInspection {
2497        action_count,
2498        ironwood_action_count,
2499        spends,
2500        outputs,
2501        fee_zat,
2502        orchard_net_value,
2503        ironwood_net_value,
2504        verified_balance,
2505        known_spends,
2506        tx_version,
2507        consensus_branch_id_hex,
2508        expiry_height,
2509    })
2510}

2531/// `tx_modifiable` bits the PCZT spec clears once a shielded spend has been
2532/// authorized (the signature commits to every transaction effect). The `pczt`
2533/// crate keeps its own copies `pub(crate)`, so they are restated here; the
2534/// high-level `Signer` role clears exactly these three after each successful
2535/// `sign_orchard` / `sign_ironwood`, and the low-level role leaves it to us.
2536const PCZT_FLAG_TRANSPARENT_INPUTS_MODIFIABLE: u8 = 0b0000_0001;
2537const PCZT_FLAG_TRANSPARENT_OUTPUTS_MODIFIABLE: u8 = 0b0000_0010;
2538const PCZT_FLAG_SHIELDED_MODIFIABLE: u8 = 0b1000_0000;
2539
2540/// Spend-auth sign every action of one orchard-shaped bundle (orchard or
2541/// ironwood) that this seed controls, with the full viewing key supplied by
2542/// the caller rather than read from the PCZT.
2543///
2544/// Skips, rather than aborts on, actions that are not ours: dummy/padding
2545/// spends carry an ephemeral key the user does not hold (the wallet's
2546/// IoFinalizer already signed them), and a foreign spend in a multi-party
2547/// PCZT belongs to someone else. `signed` counts the actions we actually
2548/// authorized, so the caller can refuse a PCZT it contributed nothing to.
2549fn sign_orchard_shaped_bundle(
2550    bundle: &mut orchard::pczt::Bundle,
2551    fvk: &orchard::keys::FullViewingKey,
2552    ask: &orchard::keys::SpendAuthorizingKey,
2553    shielded_sighash: [u8; 32],
2554    tx_modifiable: &mut u8,
2555    signed: &mut usize,
2556) -> Result<(), LowLevelSignError> {
2557    for action in bundle.actions_mut().iter_mut() {
2558        // Consistency check against the caller-supplied fvk. The redaction
2559        // that crosses the airgap also strips the spent note's plaintext, so
2560        // recipient/value/rho/rseed surface as `Missing*`; tolerate exactly
2561        // the set the high-level `Signer` tolerates. Anything else (a real
2562        // `WrongFvkForNote` / `InvalidNullifier` on a note we do not own) is
2563        // a skip, not a hard failure.
2564        match action.spend().verify_nullifier(Some(fvk)) {
2565            Ok(())
2566            | Err(
2567                orchard::pczt::VerifyError::MissingRecipient
2568                | orchard::pczt::VerifyError::MissingValue
2569                | orchard::pczt::VerifyError::MissingRho
2570                | orchard::pczt::VerifyError::MissingRandomSeed,
2571            ) => {}
2572            Err(_) => continue,
2573        }
2574        // `Action::sign` re-derives `rk` from `ask` and `alpha` and refuses
2575        // unless it matches the PCZT's `rk`. That check — not the fvk — is
2576        // what binds the signature to a note this seed controls.
2577        if action
2578            .sign(
2579                shielded_sighash,
2580                ask,
2581                rand010::rand_core::UnwrapErr(rand010::rngs::SysRng),
2582            )
2583            .is_ok()
2584        {
2585            *signed += 1;
2586            *tx_modifiable &= !(PCZT_FLAG_TRANSPARENT_INPUTS_MODIFIABLE
2587                | PCZT_FLAG_TRANSPARENT_OUTPUTS_MODIFIABLE
2588                | PCZT_FLAG_SHIELDED_MODIFIABLE);
2589        }
2590    }
2591    Ok(())
2592}

2594/// This function:
2595/// 1. Parses the PCZT to extract orchard actions
2596/// 2. Signs each action with the spending key derived from seed
2597/// 3. Injects signatures back into the PCZT
2598fn sign_zcash_pczt(
2599    seed_phrase: &str,
2600    account_index: u32,
2601    pczt_bytes: Vec<u8>,
2602) -> Result<Vec<u8>, ErrorDisplayed> {
2603    use pczt::roles::low_level_signer::Signer as LowLevelSigner;
2604    use pczt::roles::signer::Signer;
2605    use pczt::Pczt;
2606    use transaction_signing::zcash::OrchardSpendingKey;
2607
2608    // Run inspection; signing itself is intentionally permissive.
2609    //
2610    // The previous anchor-equality and "no verified notes" gates were
2611    // rotko-invented and don't appear in the canonical Zashi → Keystone PCZT
2612    // flow — Keystone's firmware signs whatever passes user review of
2613    // recipient + amount + fee, without inspecting the PCZT anchor at all
2614    // (verified at /steam/rotko/keystone3-firmware/rust/apps/zcash/src/pczt/
2615    // {sign,check}.rs). We mirror that.
2616    //
2617    // zigner's distinguishing safety value (the verified-notes store) is
2618    // surfaced as a SOFT on-device warning at review time, not a hard refuse
2619    // here (rotkonetworks/zigner#16). A hard `known_spends == 0` / implied-
2620    // spend-vs-balance refuse is impractical: the cold device can't learn its
2621    // own change notes (their tree position isn't assigned until the tx is
2622    // mined), so a synced wallet's change-spends would be refused until a
2623    // re-sync. Instead the sign screen shows the `WARN_UNRECOGNIZED`
2624    // acknowledge page when the tx spends value from a note not in the verified
2625    // set (0-value dummies excluded), plus inline per-spend verified / dummy /
2626    // unknown labels (see ZcashPcztScreen).
2627    let inspection = inspect_zcash_pczt(pczt_bytes.clone())?;
2628
2629    // Ironwood (NU6.3 / V6) spends carry the SAME orchard spend-auth key: the
2630    // pool reuses the orchard key tree, so the one `ask` below authorizes both
2631    // bundles. 0 for a V5 transaction.
2632    let total_actions =
2633        inspection.action_count as usize + inspection.ironwood_action_count as usize;
2634
2635    // Parse the PCZT to get action count first
2636    let pczt = Pczt::parse(&pczt_bytes).map_err(|e| ErrorDisplayed::Str {
2637        s: format!("Failed to parse PCZT: {:?}", e),
2638    })?;
2639
2640    // Derive the orchard spending key from seed
2641    let spending_key =
2642        OrchardSpendingKey::from_seed_phrase(seed_phrase, account_index).map_err(|e| {
2643            ErrorDisplayed::Str {
2644                s: format!("Failed to derive spending key: {}", e),
2645            }
2646        })?;
2647
2648    // Get the actual orchard spending key for signing
2649    let orchard_sk = spending_key
2650        .to_spending_key()
2651        .map_err(|e| ErrorDisplayed::Str {
2652            s: format!("Failed to get orchard spending key: {}", e),
2653        })?;
2654
2655    // The signer needs the ask (spend authorizing key) derived from sk.
2656    let ask = orchard::keys::SpendAuthorizingKey::from(&orchard_sk);
2657    // ...and the matching full viewing key, RECONSTRUCTED FROM THE SEED. The
2658    // wallet-side redaction (zafu/zcli `redact_pczt_for_signer`) strips the
2659    // spend `fvk` from the PCZT before it crosses the airgap — it is the
2660    // account's 96-byte orchard FullViewingKey and shipping it over the QR
2661    // transport is a viewing-key leak. The high-level `pczt::roles::signer`
2662    // role cannot sign such a PCZT: `sign_orchard` / `sign_ironwood` hardcode
2663    // `Spend::verify_nullifier(None)`, which returns `MissingFullViewingKey`
2664    // and is NOT in the tolerated-Missing* set, so every action fails and the
2665    // device refuses a perfectly valid transaction. We hold the seed, so we
2666    // reconstruct the fvk and drive the LOW-LEVEL Signer role, passing it to
2667    // `verify_nullifier(Some(fvk))` ourselves (`fvk_for_validation` returns
2668    // the caller-supplied key when the PCZT's own field is absent). This is
2669    // the same construction the wasmi protocol module uses
2670    // (`pczt_signing::sign_redacted_pczt`), so both shipped signing paths
2671    // accept the identical redaction. `Action::sign` itself never reads the
2672    // fvk — the real ownership binding is `rk == alpha * ak`, which still
2673    // holds and still rejects notes this seed does not control.
2674    let orchard_fvk = orchard::keys::FullViewingKey::from(&orchard_sk);
2675
2676    // The shielded sighash is a function of transaction EFFECTS only (it does
2677    // not depend on any spend-auth signature), so compute it once from the
2678    // parsed PCZT and reuse it for both bundles.
2679    let shielded_sighash = Signer::new(pczt.clone())
2680        .map_err(|e| ErrorDisplayed::Str {
2681            s: format!("Failed to create PCZT signer: {:?}", e),
2682        })?
2683        .shielded_sighash();
2684
2685    // Try-and-skip: ignore per-action sign failures (dummy actions are spent
2686    // under an ephemeral key the user doesn't hold; the wallet's IoFinalizer
2687    // already signed those). Refuse only if we couldn't sign anything at all.
2688    let mut signed_count = 0usize;
2689    let low = LowLevelSigner::new(pczt);
2690    let low = low
2691        .sign_orchard_with(|_pczt, bundle, tx_modifiable| {
2692            sign_orchard_shaped_bundle(
2693                bundle,
2694                &orchard_fvk,
2695                &ask,
2696                shielded_sighash,
2697                tx_modifiable,
2698                &mut signed_count,
2699            )
2700        })
2701        .map_err(|e: LowLevelSignError| ErrorDisplayed::Str {
2702            s: format!("Failed to sign orchard bundle: {e}"),
2703        })?;
2704    // Same treatment for the ironwood bundle. Empty (zero iterations) for V5.
2705    let low = low
2706        .sign_ironwood_with(|_pczt, bundle, tx_modifiable| {
2707            sign_orchard_shaped_bundle(
2708                bundle,
2709                &orchard_fvk,
2710                &ask,
2711                shielded_sighash,
2712                tx_modifiable,
2713                &mut signed_count,
2714            )
2715        })
2716        .map_err(|e: LowLevelSignError| ErrorDisplayed::Str {
2717            s: format!("Failed to sign ironwood bundle: {e}"),
2718        })?;
2719    if signed_count == 0 && total_actions > 0 {
2720        return Err(ErrorDisplayed::Str {
2721            s: format!(
2722                "Could not sign any of the {total_actions} action(s) — none match the user's spending key."
2723            ),
2724        });
2725    }
2726
2727    // Finish signing and get the signed PCZT
2728    let signed_pczt = low.finish();
2729
2730    // Serialize back to bytes. Since pczt 0.9 `serialize` picks the MINIMAL
2731    // encoding that can carry the content: the v1 PCZT encoding whenever the
2732    // PCZT is v1-representable (V5 tx + canonical-empty ironwood bundle), the
2733    // v2 encoding otherwise. A V5 response therefore still goes back to
2734    // Zashi/zafu as v1 bytes, byte-identical in format to what shipped; only a
2735    // V6 / ironwood response needs v2.
2736    signed_pczt.serialize().map_err(|e| ErrorDisplayed::Str {
2737        s: format!("Failed to serialize signed PCZT: {:?}", e),
2738    })
2739}

2741/// Encode signed PCZT as UR string(s) for QR display
2742/// Returns a vector of UR parts for animated QR (or single part for small PCZTs)
2743fn encode_signed_pczt_ur(
2744    pczt_bytes: Vec<u8>,
2745    max_fragment_len: u32,
2746) -> Result<Vec<String>, ErrorDisplayed> {
2747    // Wrap PCZT bytes in CBOR: { 1: bytes }
2748    let cbor_data = encode_pczt_to_cbor(&pczt_bytes);
2749
2750    if max_fragment_len == 0 || cbor_data.len() <= max_fragment_len as usize {
2751        // Single part UR
2752        let ur_string = ur::ur::encode(&cbor_data, &ur::Type::Custom("zcash-pczt"));
2753        Ok(vec![ur_string])
2754    } else {
2755        // Multi-part (animated) UR using fountain codes
2756        let mut encoder = ur::ur::Encoder::new(&cbor_data, max_fragment_len as usize, "zcash-pczt")
2757            .map_err(|e| ErrorDisplayed::Str {
2758                s: format!("Failed to create UR encoder: {:?}", e),
2759            })?;
2760
2761        let mut parts = Vec::new();
2762        // 1.3× redundancy: with the slowed (~700ms/frame) display each frame
2763        // should land cleanly, so we only need a slim margin over the 1.05–1.20×
2764        // BC-UR convergence threshold.
2765        let total_parts = (encoder.fragment_count() * 13).div_ceil(10);

2770            })?;

2775    }

2778/// Encode PCZT bytes into CBOR wrapper
2779/// CBOR structure: { 1: bytes }
2780fn encode_pczt_to_cbor(pczt_bytes: &[u8]) -> Vec<u8> {
2781    let mut cbor = Vec::new();
2782
2783    // map(1)
2784    cbor.push(0xa1);
2785    // key 1
2786    cbor.push(0x01);

2788    // bytes with appropriate length encoding
2789    let len = pczt_bytes.len();

2791        cbor.push(0x40 | len as u8);
2792    } else if len <= 255 {
2793        cbor.push(0x58);
2794        cbor.push(len as u8);
2795    } else if len <= 65535 {
2796        cbor.push(0x59);
2797        cbor.push((len >> 8) as u8);
2798        cbor.push(len as u8);
2799    } else {
2800        cbor.push(0x5a);
2801        cbor.push((len >> 24) as u8);
2802        cbor.push((len >> 16) as u8);
2803        cbor.push((len >> 8) as u8);
2804        cbor.push(len as u8);
2805    }
2806
2807    cbor.extend_from_slice(pczt_bytes);
2808    cbor

2815/// Decode UR-encoded zcash-notes, verify merkle paths, store in sled.
2816///
2817/// Trust model:
2818/// - If attestation-required flag is set (sticky, set on first FROST DKG):
2819///   anchor attestation (CBOR key 5) is REQUIRED. Verified against stored
2820///   FROST group keys. Hard reject if missing, invalid, or no key matches.
2821/// - If flag is not set (never had a FROST wallet): accept without attestation.
2822///   The user trusts zcli in this mode — that's the model they chose.
2823fn decode_and_verify_zcash_notes(
2824    ur_parts: Vec<String>,
2825) -> Result<ZcashNoteSyncResult, ErrorDisplayed> {
2826    use transaction_signing::zcash::{decode_notes_bundle_from_cbor, verify_merkle_path};
2827
2828    if ur_parts.is_empty() {
2829        return Err(ErrorDisplayed::Str {
2830            s: "No QR parts provided".to_string(),
2831        });
2832    }
2833
2834    // auto-detect UR or zoda transport format
2835    let cbor_data = decode_qr_payload(&ur_parts, "zcash-notes")?;
2836
2837    let bundle = decode_notes_bundle_from_cbor(&cbor_data).map_err(|e| ErrorDisplayed::Str {
2838        s: format!("Failed to parse notes CBOR: {e}"),
2839    })?;
2840
2841    // Single DB lock for the entire operation
2842    let db_guard = DB.read().map_err(|_| ErrorDisplayed::MutexPoisoned)?;
2843    let database = db_guard.as_ref().ok_or(ErrorDisplayed::DbNotInitialized)?;
2844
2845    // Monotonic height check: reject anchors older than what we already have
2846    if let Ok(Some((_, stored_height, _, _, _))) = db_handling::zcash::get_verified_anchor(database)
2847    {
2848        if bundle.anchor_height < stored_height {
2849            return Err(ErrorDisplayed::Str {
2850                s: format!(
2851                    "Anchor height {} is older than stored height {}. \
2852                     Rejecting stale anchor (possible replay).",
2853                    bundle.anchor_height, stored_height
2854                ),
2855            });
2856        }
2857    }
2858
2859    // Verify anchor attestation against ANY enabled verifier in the
2860    // sled-backed registry. Bootstrapped from ROTKO_ZCASH_VERIFIER on
2861    // first run, but users running their own zidecar can scan in
2862    // additional verifier keys via the dedicated UR import path.
2863    let anchor_verified =
2864        if let Some(attestation) = &bundle.anchor_attestation {
2865            if attestation.len() != 64 {
2866                return Err(ErrorDisplayed::Str {
2867                    s: format!(
2868                        "attestation length {} bytes — expected exactly 64 (ed25519)",
2869                        attestation.len()
2870                    ),
2871                });
2872            }
2873            let signature = &attestation[..64];
2874            let verifier_pubkeys = db_handling::anchor_verifiers::enabled_pubkeys(database)
2875                .map_err(|e| ErrorDisplayed::Str {
2876                    s: format!("anchor verifier registry: {e}"),
2877                })?;
2878
2879            if verifier_pubkeys.is_empty() {
2880                // No enabled verifiers — refuse rather than accept silently. A
2881                // device with no trusted verifier MUST NOT accept attested
2882                // bundles; re-add at least the built-in verifier or scan in
2883                // your own.
2884                return Err(ErrorDisplayed::Str {
2885                    s: "No enabled anchor verifiers. Re-enable the built-in verifier \
2886                    or add your own zidecar verifier key in Settings before \
2887                    importing notes."
2888                        .to_string(),
2889                });
2890            }
2891
2892            let sig = sp_core::ed25519::Signature::from_raw({
2893                let mut s = [0u8; 64];
2894                s.copy_from_slice(signature);
2895                s
2896            });
2897
2898            // Try each enabled verifier — accept on the first match.
2899            // Per-verifier digest because the verifier's pubkey is bound
2900            // into the digest's domain separation.
2901            let mut matched = false;
2902            for verifier_key in &verifier_pubkeys {
2903                use sha2::{Digest, Sha256};
2904                let mut hasher = Sha256::new();
2905                hasher.update(b"zcash-anchor-v1");
2906                hasher.update(verifier_key);
2907                hasher.update(bundle.anchor);
2908                hasher.update(bundle.anchor_height.to_le_bytes());
2909                hasher.update([u8::from(bundle.mainnet)]);
2910                let digest: [u8; 32] = hasher.finalize().into();
2911                let pubkey = sp_core::ed25519::Public::from_raw(*verifier_key);
2912                if <sp_core::ed25519::Pair as sp_core::Pair>::verify(&sig, digest, &pubkey) {
2913                    matched = true;
2914                    break;
2915                }
2916            }
2917            if !matched {
2918                return Err(ErrorDisplayed::Str {
2919                    s: "Anchor attestation signature invalid — not signed by any \
2920                    trusted verifier. Check Settings → Anchor verifiers."
2921                        .to_string(),
2922                });
2923            }
2924            true
2925        } else {
2926            false // no attestation present, accept as unverified
2927        };
2928
2929    // Sticky attestation flag: once a device has participated in any FROST
2930    // wallet, attestation is required for all subsequent note imports. This
2931    // prevents a malicious zafu from feeding fabricated note bundles to an
2932    // ex-FROST device (which would otherwise be in single-signer "trust zcli"
2933    // mode after wallet deletion).
2934    if db_handling::zcash::is_attestation_required(database).unwrap_or(false) && !anchor_verified {
2935        return Err(ErrorDisplayed::Str {
2936            s: "This device requires FROST-attested anchors. Bundle has no \
2937                attestation or attestation failed verification. Refusing to \
2938                import unverified notes."
2939                .to_string(),
2940        });
2941    }
2942
2943    // Verify each note's merkle path against anchor
2944    let mut verified_notes = Vec::new();
2945    for (i, note) in bundle.notes.iter().enumerate() {
2946        let valid = verify_merkle_path(&note.cmx, note.position, &note.merkle_path, &bundle.anchor)
2947            .map_err(|e| ErrorDisplayed::Str {
2948                s: format!("Merkle verification error for note {i}: {e}"),
2949            })?;
2950
2951        if !valid {
2952            return Err(ErrorDisplayed::Str {
2953                s: format!(
2954                    "Note {i} failed merkle verification (cmx={}, pos={})",
2955                    hex::encode(note.cmx),
2956                    note.position
2957                ),
2958            });
2959        }
2960
2961        verified_notes.push((
2962            note.value,
2963            note.nullifier,
2964            note.cmx,
2965            note.position,
2966            note.block_height,
2967        ));
2968    }
2969
2970    let total_balance: u64 = verified_notes.iter().map(|(v, _, _, _, _)| v).sum();
2971    let notes_verified = verified_notes.len() as u32;
2972
2973    db_handling::zcash::store_verified_notes(
2974        database,
2975        &verified_notes,
2976        &bundle.anchor,
2977        bundle.anchor_height,
2978        bundle.mainnet,
2979        bundle.anchor_time.unwrap_or(0),
2980    )
2981    .map_err(|e| ErrorDisplayed::Str {
2982        s: format!("Failed to store notes: {e}"),
2983    })?;
2984
2985    // Surface when these notes were stored (device clock). Read it back from
2986    // the anchor we just wrote so the UI can answer "is this balance stale?"
2987    // with both a chain block height and a human scan time. A read failure
2988    // here must not fail the import -> degrade to 0 (unknown).
2989    let synced_at = db_handling::zcash::get_verified_anchor(database)
2990        .ok()
2991        .flatten()
2992        .map(|(_, _, _, ts, _)| ts)
2993        .unwrap_or(0);
2994
2995    Ok(ZcashNoteSyncResult {
2996        notes_verified,
2997        total_balance,
2998        anchor_hex: hex::encode(bundle.anchor),
2999        anchor_height: bundle.anchor_height,
3000        mainnet: bundle.mainnet,
3001        anchor_verified,
3002        synced_at,
3003        anchor_time: bundle.anchor_time.unwrap_or(0),
3004    })
3005}

3119/// Feed collected `zt:type/hex` frames into a zoda decoder, skipping stray /
3120/// foreign-session frames. Shared by the scan-progress probe and the full
3121/// decode so both see identical accumulation semantics.
3122///
3123/// Animated QR frames arrive in cycling order, so the metadata frame (index 0,
3124/// carrying k/n) is rarely captured first. zoda's decoder cannot recover the
3125/// metadata if a non-zero frame is seen first (it pins the session and then
3126/// treats the later frame 0 as an ordinary chunk), so we hand it the index-0
3127/// frame first. Per-frame errors are skipped rather than fatal — a single
3128/// garbled QR read shouldn't abort a scan that's still collecting frames.
3129fn zt_decoder_from_parts(
3130    parts: &[String],
3131    expected_type: &str,
3132) -> Result<zoda_vss::transport::Decoder, ErrorDisplayed> {
3133    use zoda_vss::transport::Decoder as ZtDecoder;
3134
3135    let zt_prefix = format!("zt:{}/", expected_type);
3136
3137    // hex-decode the matching frames, skipping non-matching / partial reads.
3138    let mut frames: Vec<Vec<u8>> = Vec::new();
3139    for part in parts {
3140        if !part.to_lowercase().starts_with(&zt_prefix) {
3141            continue;
3142        }
3143        if let Ok(bytes) = hex::decode(&part[zt_prefix.len()..]) {
3144            frames.push(bytes);
3145        }
3146    }
3147
3148    // frame layout: session_id(8) || index(1) || chunk.
3149    let mut decoder = ZtDecoder::new();
3150
3151    // Wait for the metadata (index-0) frame before feeding anything. zoda's
3152    // decoder pins the session on the first frame it sees and panics
3153    // (`self.n.unwrap()`) if a *second* non-zero frame arrives before
3154    // metadata. Holding off until frame 0 is present — and feeding it first —
3155    // guarantees k/n are established before any chunk frame, so that unwrap is
3156    // never reached. Until then we report an empty decoder (have=0, k unknown).
3157    if !frames.iter().any(|f| f.get(8) == Some(&0)) {
3158        return Ok(decoder);
3159    }
3160    // stable-sort the index-0 frame to the front; everything else keeps order.
3161    frames.sort_by_key(|f| u8::from(f.get(8) != Some(&0)));
3162
3163    for frame_bytes in &frames {
3164        // tolerate per-frame rejects (foreign session, dup).
3165        let _ = decoder.receive(frame_bytes);
3166        if decoder.complete() {
3167            break;
3168        }
3169    }
3170
3171    Ok(decoder)
3172}

3476                (crc >> 1) ^ 0xedb88320
3477            } else {
3478                crc >> 1
3479            };
3480        }
3481    }
3482    crc ^ 0xffffffff
3483}

3696    data: &[u8],
3697    ur_type: &str,
3698    max_fragment_len: u32,

3700    // Wrap in CBOR bstr

3717            let part = encoder.next_part().map_err(|e| ErrorDisplayed::Str {
3718                s: format!("UR part: {e:?}"),

3721        }

3729            s: "no UR parts".to_string(),

3731    }

3734            s: format!("UR decode: {e:?}"),

3737            s: format!("CBOR: {e}"),

3744            s: format!("UR receive: {e:?}"),

3746        if decoder.complete() {

3748                .message()
3749                .map_err(|e| ErrorDisplayed::Str {
3750                    s: format!("UR message: {e:?}"),
3751                })?
3752                .ok_or_else(|| ErrorDisplayed::Str {
3753                    s: "UR complete but no message".to_string(),
3754                })?;
3755            let (bytes, _) = parse_cbor_bytes(&cbor).map_err(|e| ErrorDisplayed::Str {
3756                s: format!("CBOR: {e}"),
3757            })?;
3758            return Ok(bytes);
3759        }
3760    }
3761    Err(ErrorDisplayed::Str {
3762        s: "incomplete UR".to_string(),
3763    })
3764}

3800fn parse_cbor_tstr(data: &[u8], offset: usize) -> Result<(String, usize), String> {
3801    if offset >= data.len() {
3802        return Err("truncated tstr".to_string());
3803    }
3804    let first = data[offset];
3805    let major = first >> 5;
3806    if major != 3 {
3807        return Err(format!("expected tstr, got major {major}"));
3808    }
3809    let additional = first & 0x1f;
3810    let (len, header_end) = match additional {
3811        0..=23 => (additional as usize, offset + 1),
3812        24 => {
3813            if offset + 2 > data.len() {
3814                return Err("truncated".to_string());
3815            }
3816            (data[offset + 1] as usize, offset + 2)
3817        }
3818        25 => {
3819            if offset + 3 > data.len() {
3820                return Err("truncated".to_string());
3821            }
3822            (
3823                u16::from_be_bytes([data[offset + 1], data[offset + 2]]) as usize,
3824                offset + 3,
3825            )
3826        }
3827        _ => return Err(format!("unsupported tstr additional {additional}")),
3828    };
3829    if header_end + len > data.len() {
3830        return Err("truncated tstr data".to_string());
3831    }
3832    let s = std::str::from_utf8(&data[header_end..header_end + len])
3833        .map_err(|e| format!("invalid UTF-8: {e}"))?;
3834    Ok((s.to_string(), header_end + len))
3835}

3837/// Parse a CBOR map header, returning (length, offset after header).
3838fn cbor_parse_map_header(data: &[u8]) -> Result<(usize, usize), String> {
3839    if data.is_empty() {
3840        return Err("empty CBOR".to_string());
3841    }
3842    let first = data[0];
3843    let major = first >> 5;
3844    if major != 5 {
3845        return Err(format!("expected map (major 5), got major {major}"));
3846    }
3847    let additional = first & 0x1f;
3848    cbor_parse_length(data, 0, additional).map(|(len, off)| (len as usize, off))
3849}

3864/// Decode CBOR additional info into a length/value.
3865fn cbor_parse_length(data: &[u8], offset: usize, additional: u8) -> Result<(u64, usize), String> {
3866    match additional {
3867        0..=23 => Ok((additional as u64, offset + 1)),
3868        24 => {
3869            if offset + 2 > data.len() {
3870                return Err("truncated length".to_string());
3871            }
3872            Ok((data[offset + 1] as u64, offset + 2))
3873        }
3874        25 => {
3875            if offset + 3 > data.len() {
3876                return Err("truncated length".to_string());
3877            }
3878            Ok((
3879                u16::from_be_bytes([data[offset + 1], data[offset + 2]]) as u64,
3880                offset + 3,
3881            ))
3882        }
3883        26 => {
3884            if offset + 5 > data.len() {
3885                return Err("truncated length".to_string());
3886            }
3887            Ok((
3888                u32::from_be_bytes([
3889                    data[offset + 1],
3890                    data[offset + 2],
3891                    data[offset + 3],
3892                    data[offset + 4],
3893                ]) as u64,
3894                offset + 5,
3895            ))
3896        }
3897        _ => Err(format!("unsupported CBOR additional {additional}")),
3898    }
3899}
3900
3901/// Skip a single CBOR value of any type at offset.
3902fn cbor_skip_any(data: &[u8], offset: usize) -> Result<usize, String> {
3903    if offset >= data.len() {
3904        return Err("truncated CBOR value".to_string());
3905    }
3906    let first = data[offset];
3907    let major = first >> 5;
3908    let additional = first & 0x1f;
3909    let (content_len, header_end) = cbor_parse_length(data, offset, additional)?;
3910    let content_len = content_len as usize;
3911    match major {
3912        0 | 1 => Ok(header_end),               // uint/negint
3913        2 | 3 => Ok(header_end + content_len), // bstr/tstr
3914        4 => {
3915            let mut pos = header_end;
3916            for _ in 0..content_len {
3917                pos = cbor_skip_any(data, pos)?;
3918            }
3919            Ok(pos)
3920        }
3921        5 => {
3922            let mut pos = header_end;
3923            for _ in 0..content_len {
3924                pos = cbor_skip_any(data, pos)?; // key
3925                pos = cbor_skip_any(data, pos)?; // value
3926            }
3927            Ok(pos)
3928        }
3929        7 => Ok(offset + 1), // simple (bool/null)
3930        _ => Err(format!("unsupported CBOR major {major}")),
3931    }
3932}

4279/// Parse the KeyPackage + PublicKeyPackage hex and assert they agree with
4280/// the envelope's claimed signing policy. Returns the cryptographically
4281/// authoritative `(min_signers, max_signers)` for storage so future code
4282/// reads canonical values, not envelope claims.
4283///
4284/// Threat: a hostile envelope can ship correct KeyPackage + PublicKeyPackage
4285/// material with tampered envelope-level threshold/max_signers fields. The
4286/// UI then displays a different m-of-n than the FROST polynomial encodes.
4287/// Cross-check pins the displayed policy to the cryptographic ground truth.
4288fn verify_envelope_threshold(
4289    public_key_package_hex: &str,
4290    key_package_hex: &str,
4291    claimed_threshold: u16,
4292    claimed_max_signers: u16,
4293) -> Result<(u16, u16), ErrorDisplayed> {
4294    use frost_spend::frost_keys::{KeyPackage, PublicKeyPackage};
4295    use frost_spend::orchestrate::from_hex;
4296    let pubkeys: PublicKeyPackage =
4297        from_hex(public_key_package_hex).map_err(|e| ErrorDisplayed::Str {
4298            s: format!("parse public_key_package: {e}"),
4299        })?;
4300    let key_pkg: KeyPackage = from_hex(key_package_hex).map_err(|e| ErrorDisplayed::Str {
4301        s: format!("parse key_package: {e}"),
4302    })?;
4303    let pkp_max: u16 =
4304        pubkeys
4305            .verifying_shares()
4306            .len()
4307            .try_into()
4308            .map_err(|_| ErrorDisplayed::Str {
4309                s: "PublicKeyPackage participant count exceeds u16".into(),
4310            })?;
4311    let kp_min: u16 = *key_pkg.min_signers();
4312    if kp_min != claimed_threshold {
4313        return Err(ErrorDisplayed::Str {
4314            s: format!(
4315                "envelope threshold {claimed_threshold} disagrees with KeyPackage min_signers {kp_min}"
4316            ),
4317        });
4318    }
4319    if pkp_max != claimed_max_signers {
4320        return Err(ErrorDisplayed::Str {
4321            s: format!(
4322                "envelope max_signers {claimed_max_signers} disagrees with PublicKeyPackage participant count {pkp_max}"
4323            ),
4324        });
4325    }
4326    Ok((kp_min, pkp_max))
4327}

4534/// Decode a `ur:zcash-verifier-key` UR (CBOR map: 1=pubkey, 2=label) into
4535/// a preview struct. Does NOT add the verifier — the native layer must
4536/// show a confirmation screen with the full hex pubkey before calling
4537/// add_anchor_verifier.
4538fn decode_verifier_key_qr(ur_parts: Vec<String>) -> Result<AnchorVerifierImport, ErrorDisplayed> {
4539    if ur_parts.is_empty() {
4540        return Err(ErrorDisplayed::Str {
4541            s: "no QR parts".into(),
4542        });
4543    }
4544    let cbor = decode_qr_payload(&ur_parts, "zcash-verifier-key")?;
4545
4546    // CBOR: a2 01 (bstr 32 bytes) 02 (tstr label)
4547    if cbor.len() < 3 || cbor[0] != 0xa2 {
4548        return Err(ErrorDisplayed::Str {
4549            s: "verifier-key CBOR must be a 2-entry map".into(),
4550        });
4551    }
4552    let mut pubkey: Option<[u8; 32]> = None;
4553    let mut label: Option<String> = None;
4554    let mut o = 1;
4555    for _ in 0..2 {
4556        let key = *cbor.get(o).ok_or_else(|| ErrorDisplayed::Str {
4557            s: "truncated key".into(),
4558        })?;
4559        o += 1;
4560        match key {
4561            0x01 => {
4562                // bstr 32: 0x58 0x20 || 32 bytes (or 0x40+len for short)
4563                let prefix = *cbor.get(o).ok_or_else(|| ErrorDisplayed::Str {
4564                    s: "truncated pubkey header".into(),
4565                })?;
4566                let body_start = if prefix == 0x58 {
4567                    let len = *cbor.get(o + 1).ok_or_else(|| ErrorDisplayed::Str {
4568                        s: "truncated pubkey length".into(),
4569                    })?;
4570                    if len != 32 {
4571                        return Err(ErrorDisplayed::Str {
4572                            s: format!("expected 32-byte pubkey, got {len}"),
4573                        });
4574                    }
4575                    o + 2
4576                } else if prefix == 0x40 + 32 {
4577                    o + 1
4578                } else {
4579                    return Err(ErrorDisplayed::Str {
4580                        s: format!("expected bstr(32), got prefix 0x{prefix:02x}"),
4581                    });
4582                };
4583                let end = body_start
4584                    .checked_add(32)
4585                    .ok_or_else(|| ErrorDisplayed::Str {
4586                        s: "pubkey overflow".into(),
4587                    })?;
4588                let bytes = cbor
4589                    .get(body_start..end)
4590                    .ok_or_else(|| ErrorDisplayed::Str {
4591                        s: "pubkey truncated".into(),
4592                    })?;
4593                let mut pk = [0u8; 32];
4594                pk.copy_from_slice(bytes);
4595                pubkey = Some(pk);
4596                o = end;
4597            }
4598            0x02 => {
4599                // tstr: prefix 0x60+len (short) or 0x78 len (medium)
4600                let prefix = *cbor.get(o).ok_or_else(|| ErrorDisplayed::Str {
4601                    s: "truncated label header".into(),
4602                })?;
4603                let (body_start, body_len) = if (0x60..0x78).contains(&prefix) {
4604                    (o + 1, (prefix - 0x60) as usize)
4605                } else if prefix == 0x78 {
4606                    let len = *cbor.get(o + 1).ok_or_else(|| ErrorDisplayed::Str {
4607                        s: "truncated label length".into(),
4608                    })?;
4609                    (o + 2, len as usize)
4610                } else {
4611                    return Err(ErrorDisplayed::Str {
4612                        s: format!("expected tstr label, got prefix 0x{prefix:02x}"),
4613                    });
4614                };
4615                if body_len > 128 {
4616                    return Err(ErrorDisplayed::Str {
4617                        s: format!("label too long ({body_len} bytes, max 128)"),
4618                    });
4619                }
4620                let end = body_start
4621                    .checked_add(body_len)
4622                    .ok_or_else(|| ErrorDisplayed::Str {
4623                        s: "label overflow".into(),
4624                    })?;
4625                let bytes = cbor
4626                    .get(body_start..end)
4627                    .ok_or_else(|| ErrorDisplayed::Str {
4628                        s: "label truncated".into(),
4629                    })?;
4630                label =
4631                    Some(
4632                        String::from_utf8(bytes.to_vec()).map_err(|e| ErrorDisplayed::Str {
4633                            s: format!("label utf-8: {e}"),
4634                        })?,
4635                    );
4636                o = end;
4637            }
4638            other => {
4639                return Err(ErrorDisplayed::Str {
4640                    s: format!("unexpected CBOR map key {other}"),
4641                });
4642            }
4643        }
4644    }
4645    let pubkey = pubkey.ok_or_else(|| ErrorDisplayed::Str {
4646        s: "missing pubkey field".into(),
4647    })?;
4648    let label = label.unwrap_or_default();
4649    Ok(AnchorVerifierImport {
4650        pubkey_hex: hex::encode(pubkey),
4651        label,
4652    })
4653}

4822pub fn module_summarize_request(
4823    module_wasm: &[u8],
4824    payload: &[u8],
4825) -> Result<Vec<ModulePcztSummary>, ErrorDisplayed> {
4826    let mut rt = module_runtime(module_wasm)?;
4827    let raw = rt
4828        .summarize_request(payload)
4829        .map_err(|e| ErrorDisplayed::Str {
4830            s: format!("{e:?}"),
4831        })?;
4832    // module ABI: records separated by 0x1e; head line
4833    // "actions=N ironwood_actions=I t_inputs=M fee=F" (fee is a zatoshi
4834    // integer or the literal "unknown"), following lines "label=value".
4835    // ironwood_actions/fee are absent from pre-ironwood modules; treat a
4836    // missing ironwood_actions as 0 and a missing/"unknown" fee as None so
4837    // old modules keep parsing.
4838    let mut out = Vec::new();
4839    for record in raw.split(|b| *b == 0x1e).filter(|r| !r.is_empty()) {
4840        let text = String::from_utf8_lossy(record);
4841        let mut lines = text.lines();
4842        let head = lines.next().unwrap_or_default();
4843        let mut actions = 0u32;
4844        let mut ironwood_actions = 0u32;
4845        let mut t_inputs = 0u32;
4846        let mut fee_zat = None;
4847        let mut is_delegation = false;
4848        for part in head.split_whitespace() {
4849            if let Some(v) = part.strip_prefix("ironwood_actions=") {
4850                ironwood_actions = v.parse().unwrap_or(0);
4851            } else if let Some(v) = part.strip_prefix("actions=") {
4852                actions = v.parse().unwrap_or(0);
4853            } else if let Some(v) = part.strip_prefix("t_inputs=") {
4854                t_inputs = v.parse().unwrap_or(0);
4855            } else if let Some(v) = part.strip_prefix("fee=") {
4856                fee_zat = v.parse::<u64>().ok();
4857            } else if let Some(v) = part.strip_prefix("kind=") {
4858                is_delegation = v == "delegation";
4859            }
4860        }
4861        out.push(ModulePcztSummary {
4862            orchard_actions: actions,
4863            ironwood_actions,
4864            transparent_inputs: t_inputs,
4865            fee_zat,
4866            output_lines: lines.map(str::to_owned).collect(),
4867            is_delegation,
4868        });
4869    }
4870    Ok(out)
4871}

4969/// Fountain-decode `ur:zigner-module` protocol-module REQUEST frames back to
4970/// the raw prelude envelope bytes (`[0x53][crypto][tx_type] || payload`).
4971///
4972/// The emitting wallet (zafu) animates the module request over the SAME BC-UR
4973/// fountain as `ur:zcash-pczt`, but wraps the RAW prelude envelope directly -
4974/// there is NO CBOR `{1: bytes}` wrap on the request side (the wallet feeds the
4975/// bare envelope to `ur_encode_frames`). So unlike `decode_ur_zcash_pczt` we
4976/// return the fountain message verbatim without a CBOR unwrap; the scan
4977/// dispatcher then routes on the 6-hex prelude exactly as for a raw-byte /
4978/// substrate multi-QR request.
4979fn decode_ur_module_request(ur_parts: Vec<String>) -> Result<Vec<u8>, ErrorDisplayed> {
4980    if ur_parts.is_empty() {
4981        return Err(ErrorDisplayed::Str {
4982            s: "No UR parts provided".to_string(),
4983        });
4984    }
4985
4986    // Type guard: every part must be ur:zigner-module/... so a stray QR mid
4987    // stream can never poison the fountain decoder.
4988    for part in &ur_parts {
4989        if !part.to_lowercase().starts_with("ur:zigner-module/") {
4990            return Err(ErrorDisplayed::Str {
4991                s: format!(
4992                    "Expected ur:zigner-module/... got: {}",
4993                    part.chars().take(30).collect::<String>()
4994                ),
4995            });
4996        }
4997    }
4998
4999    if ur_parts.len() == 1 {
5000        let (_kind, bytes) = ur::ur::decode(&ur_parts[0]).map_err(|e| ErrorDisplayed::Str {
5001            s: format!("Failed to decode UR: {e:?}"),
5002        })?;
5003        return Ok(bytes);
5004    }
5005
5006    let mut decoder = ur::ur::Decoder::default();
5007    for part in &ur_parts {
5008        decoder.receive(part).map_err(|e| ErrorDisplayed::Str {
5009            s: format!("Failed to receive UR part: {e:?}"),
5010        })?;
5011        if decoder.complete() {
5012            break;
5013        }
5014    }
5015    if !decoder.complete() {
5016        return Err(ErrorDisplayed::Str {
5017            s: format!(
5018                "Incomplete UR sequence: received {} parts but not complete",
5019                ur_parts.len()
5020            ),
5021        });
5022    }
5023    decoder
5024        .message()
5025        .map_err(|e| ErrorDisplayed::Str {
5026            s: format!("Failed to get UR message: {e:?}"),
5027        })?
5028        .ok_or_else(|| ErrorDisplayed::Str {
5029            s: "UR decoder returned None despite being complete".to_string(),
5030        })
5031}

5033/// CBOR unsigned integer (major type 0), shortest form.
5034fn cbor_uint(out: &mut Vec<u8>, v: u32) {
5035    match v {
5036        0..=0x17 => out.push(v as u8),
5037        0x18..=0xff => out.extend_from_slice(&[0x18, v as u8]),
5038        0x100..=0xffff => {
5039            out.push(0x19);
5040            out.extend_from_slice(&(v as u16).to_be_bytes());
5041        }
5042        _ => {
5043            out.push(0x1a);
5044            out.extend_from_slice(&v.to_be_bytes());
5045        }
5046    }
5047}

5049/// `{1: response, 2: {1: baked, 2: active, 3: host_abi}}`.
5050///
5051/// Key 2 is what the KERNEL knows about itself - the module cannot attest its
5052/// own version, so these values come from the APK, never from module output.
5053/// Wallets use them the way they use Keystone's firmware version: to tell the
5054/// user to update the device before a request it cannot handle, rather than
5055/// after they approved it. Wallets that predate key 2 reject map(2), so this
5056/// format requires a wallet release that accepts it (zafu
5057/// `unwrapSignerEnvelope`).
5058fn encode_module_response_cbor(response: &[u8], active_module_version: u32) -> Vec<u8> {
5059    let mut cbor = encode_pczt_to_cbor(response);
5060    debug_assert_eq!(cbor[0], 0xa1);
5061    cbor[0] = 0xa2; // map(2)
5062    cbor_uint(&mut cbor, 2);
5063    cbor.push(0xa3); // map(3)
5064    cbor_uint(&mut cbor, 1);
5065    cbor_uint(&mut cbor, module_host::BAKED_MODULE_VERSION);
5066    cbor_uint(&mut cbor, 2);
5067    cbor_uint(&mut cbor, active_module_version);
5068    cbor_uint(&mut cbor, 3);
5069    cbor_uint(&mut cbor, module_host::HOST_ABI_VERSION);
5070    cbor
5071}

5073/// Frame a module sign-response envelope (prelude || digests || signed
5074/// PCZTs) as UR strings for animated QR display, with the kernel's version
5075/// facts (see [`encode_module_response_cbor`]). Same fountain encoder as the
5076/// signed-PCZT path, but under a distinct UR type so wallets never mistake the
5077/// envelope for a bare PCZT.
5078///
5079/// `active_module_version` is the version of the module that produced
5080/// `response` (installed slot or baked asset), as the caller loaded it.
5081pub fn module_response_to_ur(
5082    response: &[u8],
5083    active_module_version: u32,
5084    max_fragment_len: u32,
5085) -> Result<Vec<String>, ErrorDisplayed> {
5086    const UR_TYPE: &str = "zigner-module";
5087    let cbor_data = encode_module_response_cbor(response, active_module_version);
5088
5089    if max_fragment_len == 0 || cbor_data.len() <= max_fragment_len as usize {
5090        return Ok(vec![ur::ur::encode(&cbor_data, &ur::Type::Custom(UR_TYPE))]);
5091    }
5092
5093    let mut encoder = ur::ur::Encoder::new(&cbor_data, max_fragment_len as usize, UR_TYPE)
5094        .map_err(|e| ErrorDisplayed::Str {
5095            s: format!("Failed to create UR encoder: {e:?}"),
5096        })?;
5097
5098    // 1.3x redundancy, matching encode_signed_pczt_ur (see comment there).
5099    let total_parts = (encoder.fragment_count() * 13).div_ceil(10);
5100    let mut parts = Vec::with_capacity(total_parts);
5101    for _ in 0..total_parts {
5102        parts.push(encoder.next_part().map_err(|e| ErrorDisplayed::Str {
5103            s: format!("Failed to encode UR part: {e:?}"),
5104        })?);
5105    }
5106    Ok(parts)
5107}

5751    /// LEB128, the varint postcard uses for `u64`.
5752    fn varint(mut v: u64) -> Vec<u8> {
5753        let mut out = Vec::new();
5754        loop {
5755            let b = (v & 0x7f) as u8;
5756            v >>= 7;
5757            if v == 0 {
5758                out.push(b);
5759                return out;
5760            }
5761            out.push(b | 0x80);
5762        }
5763    }

5765    /// Replace every occurrence of `needle` with `replacement` (same length).
5766    /// Returns the patched bytes and the number of substitutions.
5767    fn patch_all(bytes: &[u8], needle: &[u8], replacement: &[u8]) -> (Vec<u8>, usize) {
5768        assert_eq!(needle.len(), replacement.len());
5769        let mut out = bytes.to_vec();
5770        let mut n = 0;
5771        let mut i = 0;
5772        while i + needle.len() <= out.len() {
5773            if &out[i..i + needle.len()] == needle {
5774                out[i..i + needle.len()].copy_from_slice(replacement);
5775                n += 1;
5776                i += needle.len();
5777            } else {
5778                i += 1;
5779            }
5780        }
5781        (out, n)
5782    }

6053    // Patch the global branch id in place. postcard LEB128-encodes the u32 and
6054    // every id used here is >= 2^28 (5 bytes), so offsets stay intact.
6055    fn leb128(mut v: u32) -> Vec<u8> {
6056        let mut out = Vec::new();
6057        loop {
6058            let b = (v & 0x7f) as u8;
6059            v >>= 7;
6060            if v == 0 {
6061                out.push(b);
6062                return out;
6063            }
6064            out.push(b | 0x80);
6065        }
6066    }
6067
6068    fn with_branch(pczt: &[u8], from: u32, to: u32) -> Vec<u8> {
6069        let (needle, repl) = (leb128(from), leb128(to));
6070        let at = pczt
6071            .windows(needle.len())
6072            .position(|w| w == needle.as_slice())
6073            .expect("branch id present");
6074        let mut out = pczt.to_vec();
6075        out[at..at + repl.len()].copy_from_slice(&repl);
6076        assert_eq!(
6077            *Pczt::parse(&out).unwrap().global().consensus_branch_id(),
6078            to
6079        );
6080        out
6081    }

6133    /// V6 shielded sighash straight from `v6_signature_hash` over the PCZT's
6134    /// effects. The verifier's `Signer` picks the digest by tx version; this
6135    /// pins it to V6, so a match shows the verifier chose the V6 digest (the
6136    /// old one took V5). Same crate on both sides - not a cross-version check.
6137    fn v6_sighash(pczt: &[u8]) -> String {
6138        use zcash_primitives::transaction::sighash::SignableInput;
6139        use zcash_primitives::transaction::sighash_v6::v6_signature_hash;
6140        use zcash_primitives::transaction::txid::TxIdDigester;
6141        let tx = Pczt::parse(pczt).unwrap().into_effects().expect("effects");
6142        let parts = tx.digest(TxIdDigester);
6143        hex::encode(v6_signature_hash(&tx, &SignableInput::Shielded, &parts).as_bytes())
6144    }
