1// ssh.rs — SSH signing over the airgap.
2//
3// The device holds an ed25519 key derived under its own domain and signs SSH
4// payloads scanned in as QR. The counterpart on the host is an ssh-agent, not
5// a patched ssh client: OpenSSH already exposes the agent protocol, so `ssh`,
6// `scp`, `git` and `ssh-keygen -Y sign` all work unmodified against it.
7//
8// Two properties this file is responsible for:
9//
10//  1. DOMAIN SEPARATION. The SSH key is derived under SSH_DOMAIN, distinct
11//     from the release-signing and auth-identity domains. A signature made
12//     here can never be valid as a release manifest signature, because it is
13//     made with a different key entirely.
14//
15//  2. NO GENERIC SIGNING ORACLE. `classify_request` parses the payload into a
16//     known SSH framing and everything else is refused. Without this, an agent
17//     that will sign arbitrary bytes is a machine for producing signatures
18//     over structures we never inspected. Refusing unknown framing is what
19//     keeps "the device signs SSH things" from becoming "the device signs".
20

26/// Derivation domain for SSH keys. Distinct from the release domain so that
27/// compromise of one authority does not imply the other.
28pub const SSH_DOMAIN: &str = "zigner-ssh";
29
30/// The only public-key algorithm this surface speaks.
31const ALG: &str = "ssh-ed25519";
32
33/// SSHSIG blobs (`ssh-keygen -Y sign`, git commit signing) open with this
34/// literal, unprefixed. Its absence is what distinguishes a userauth blob.
35const SSHSIG_MAGIC: &[u8] = b"SSHSIG";
36
37/// RFC 4252 section 7.
38const SSH_MSG_USERAUTH_REQUEST: u8 = 50;
39
40/// A signing payload we understood well enough to show a human.
41#[derive(Debug, Clone, PartialEq, Eq)]
42pub enum SshSignRequest {
43    /// Interactive authentication. Note what is NOT here: the hostname. The
44    /// signed blob does not carry one - the session identifier binds the
45    /// server's host key, so a signature cannot be replayed to a different
46    /// server, but it cannot be inverted to display where the login is going.
47    /// The confirm screen must not imply otherwise.
48    UserAuth { username: String, service: String },
49    /// A detached signature over a document - `ssh-keygen -Y sign`, which is
50    /// also what git drives for signed commits and tags. The namespace ("git",
51    /// "file", ...) and the message hash are displayable.
52    SshSig {
53        namespace: String,
54        hash_algorithm: String,
55        message_hash_hex: String,
56    },
57}

61/// Encode a length-prefixed SSH string.
62fn ssh_string(bytes: &[u8]) -> Vec<u8> {
63    let mut out = Vec::with_capacity(4 + bytes.len());
64    out.extend_from_slice(&(bytes.len() as u32).to_be_bytes());
65    out.extend_from_slice(bytes);
66    out
67}
68
69/// Read a length-prefixed SSH string, returning it and the remainder.
70fn take_string(b: &[u8]) -> Result<(&[u8], &[u8]), String> {
71    if b.len() < 4 {
72        return Err("truncated: expected a length-prefixed string".into());
73    }
74    let len = u32::from_be_bytes(b[0..4].try_into().unwrap()) as usize;
75    // checked_add, not `4 + len`. armeabi-v7a is a shipped ABI and its usize
76    // is 32-bit, so a declared length near u32::MAX wraps: `4 + len` becomes
77    // a small number, the bounds check below passes, and the slice is then
78    // built with end < start - a panic on hostile input, on a real target.
79    // Release builds wrap silently, so nothing would have caught it earlier.
80    let end = 4usize
81        .checked_add(len)
82        .ok_or_else(|| format!("string length {len} overflows this platform's address size"))?;
83    if b.len() < end {
84        return Err(format!(
85            "truncated: string claims {} bytes, {} remain",
86            len,
87            b.len() - 4
88        ));
89    }
90    Ok((&b[4..end], &b[end..]))
91}
92
93fn take_utf8(b: &[u8]) -> Result<(String, &[u8]), String> {
94    let (s, rest) = take_string(b)?;
95    let s = std::str::from_utf8(s).map_err(|_| "non-utf8 where text expected".to_string())?;
96    Ok((s.to_string(), rest))
97}

99/// The `ssh-ed25519` public key blob: `string "ssh-ed25519" || string key`.
100pub fn public_key_blob(pubkey: &[u8; 32]) -> Vec<u8> {
101    let mut out = ssh_string(ALG.as_bytes());
102    out.extend_from_slice(&ssh_string(pubkey));
103    out
104}

108/// Derive the SSH public key as an `authorized_keys` line.
109pub fn public_key_line(seed_phrase: &str, index: u32, comment: &str) -> Result<String, String> {
110    use base64::{engine::general_purpose::STANDARD, Engine as _};
111    let pair = derive_domain_keypair(seed_phrase, SSH_DOMAIN, index)?;
112    let blob = STANDARD.encode(public_key_blob(&pair.public().0));
113    Ok(if comment.is_empty() {
114        format!("{ALG} {blob}")
115    } else {
116        format!("{ALG} {blob} {comment}")
117    })
118}

120/// Parse a signing payload into a framing we can display, or refuse it.
121pub fn classify_request(blob: &[u8]) -> Result<SshSignRequest, String> {
122    Ok(parse_request(blob)?.0)
123}
124
125/// As [`classify_request`], but also returns the public key blob a userauth
126/// request names, so the caller can check it is OURS.
127///
128/// The key blob has to come out of the parser rather than be searched for in
129/// the raw bytes: username, service and session id are all attacker-chosen, so
130/// "our key appears somewhere in the blob" is a property an attacker can
131/// arrange while the actual key field names a different identity.
132fn parse_request(blob: &[u8]) -> Result<(SshSignRequest, Option<Vec<u8>>), String> {
133    if blob.starts_with(SSHSIG_MAGIC) {
134        return Ok((classify_sshsig(&blob[SSHSIG_MAGIC.len()..])?, None));
135    }
136    let (req, key_blob) = classify_userauth(blob)?;
137    Ok((req, Some(key_blob)))
138}
139
140fn classify_sshsig(rest: &[u8]) -> Result<SshSignRequest, String> {
141    let (namespace, rest) = take_utf8(rest)?;
142    let (_reserved, rest) = take_string(rest)?;
143    let (hash_algorithm, rest) = take_utf8(rest)?;
144    let (message_hash, rest) = take_string(rest)?;
145    if !rest.is_empty() {
146        return Err("trailing bytes after SSHSIG structure".into());
147    }
148    if namespace.is_empty() {
149        return Err("SSHSIG namespace is empty".into());
150    }
151    Ok(SshSignRequest::SshSig {
152        namespace,
153        hash_algorithm,
154        message_hash_hex: hex::encode(message_hash),
155    })
156}
157
158fn classify_userauth(blob: &[u8]) -> Result<(SshSignRequest, Vec<u8>), String> {
159    let (session_id, rest) = take_string(blob)?;
160    if session_id.is_empty() {
161        return Err("userauth blob has an empty session identifier".into());
162    }
163    let (&msg_type, rest) = rest
164        .split_first()
165        .ok_or_else(|| "truncated: expected a message type".to_string())?;
166    if msg_type != SSH_MSG_USERAUTH_REQUEST {
167        return Err(format!(
168            "unrecognised payload: not an SSH userauth request (message type {msg_type})"
169        ));
170    }
171    let (username, rest) = take_utf8(rest)?;
172    let (service, rest) = take_utf8(rest)?;
173    let (method, rest) = take_utf8(rest)?;
174    if method != "publickey" {
175        return Err(format!("unsupported authentication method '{method}'"));
176    }
177    let (&has_sig, rest) = rest
178        .split_first()
179        .ok_or_else(|| "truncated: expected the signature-present flag".to_string())?;
180    if has_sig != 1 {
181        return Err("userauth blob is not a signing request".into());
182    }
183    let (alg, rest) = take_utf8(rest)?;
184    if alg != ALG {
185        return Err(format!("unsupported key algorithm '{alg}'"));
186    }
187    let (key_blob, rest) = take_string(rest)?;
188    if !rest.is_empty() {
189        return Err("trailing bytes after userauth request".into());
190    }
191    Ok((
192        SshSignRequest::UserAuth { username, service },
193        key_blob.to_vec(),
194    ))
195}

197/// Sign an SSH payload, refusing anything whose framing we did not recognise.
198///
199/// Returns `(authorized_keys_line, signature_hex)`. The caller is expected to
200/// have shown the user the result of [`classify_request`] and obtained
201/// approval - this function does not itself constitute consent.
202pub fn sign_request(
203    seed_phrase: &str,
204    index: u32,
205    blob: &[u8],
206) -> Result<(String, String), String> {
207    // Classify BEFORE signing: an unparseable payload must never be signed,
208    // whatever the UI did or did not display.
209    let (_request, named_key) = parse_request(blob)?;
210
211    let pair = derive_domain_keypair(seed_phrase, SSH_DOMAIN, index)?;
212
213    // A userauth blob names the key it expects to be signed with. Compare
214    // against the PARSED field, exactly - not a substring search over the
215    // whole blob, which an attacker satisfies by putting our key blob in the
216    // username and someone else's in the key field.
217    if let Some(named_key) = named_key {
218        if named_key != public_key_blob(&pair.public().0) {
219            return Err("this request is for a different SSH key than this device holds".into());
220        }
221    }
222
223    let signature = pair.sign(blob);
224    Ok((
225        public_key_line(seed_phrase, index, "")?,
226        hex::encode(signature.0),
227    ))
228}

236    fn pubkey(index: u32) -> [u8; 32] {
237        derive_domain_keypair(MNEMONIC, SSH_DOMAIN, index)

239            .public()

243    fn userauth_blob(pk: &[u8; 32], username: &str) -> Vec<u8> {
244        let mut b = ssh_string(b"session-id-bytes");
245        b.push(SSH_MSG_USERAUTH_REQUEST);
246        b.extend_from_slice(&ssh_string(username.as_bytes()));
247        b.extend_from_slice(&ssh_string(b"ssh-connection"));
248        b.extend_from_slice(&ssh_string(b"publickey"));
249        b.push(1);
250        b.extend_from_slice(&ssh_string(ALG.as_bytes()));
251        b.extend_from_slice(&ssh_string(&public_key_blob(pk)));
252        b
253    }
254
255    fn sshsig_blob(namespace: &str) -> Vec<u8> {
256        let mut b = SSHSIG_MAGIC.to_vec();
257        b.extend_from_slice(&ssh_string(namespace.as_bytes()));
258        b.extend_from_slice(&ssh_string(b""));
259        b.extend_from_slice(&ssh_string(b"sha512"));
260        b.extend_from_slice(&ssh_string(&[0xab; 64]));
261        b
262    }

264    #[test]
265    fn public_key_line_has_the_authorized_keys_shape() {
266        let line = public_key_line(MNEMONIC, 0, "tommi@zigner").unwrap();
267        let parts: Vec<&str> = line.split(' ').collect();
268        assert_eq!(parts.len(), 3);
269        assert_eq!(parts[0], ALG);
270        assert_eq!(parts[2], "tommi@zigner");
271
272        // The base64 body must decode to the wire blob, and the algorithm name
273        // is repeated INSIDE it - that is what makes the line valid to sshd,
274        // not just well-shaped.
275        use base64::{engine::general_purpose::STANDARD, Engine as _};
276        let decoded = STANDARD.decode(parts[1]).expect("base64 body");
277        assert_eq!(decoded, public_key_blob(&pubkey(0)));
278        let (alg, rest) = take_utf8(&decoded).unwrap();
279        assert_eq!(alg, ALG);
280        assert_eq!(take_string(rest).unwrap().0.len(), 32);
281    }

344    /// The key-ownership guard must look at the KEY FIELD, not anywhere in
345    /// the blob. The session identifier is attacker-influenced and carries
346    /// arbitrary bytes, so if our public key blob merely has to appear
347    /// somewhere, it can be embedded there while the actual key field names a
348    /// different identity.
349    #[test]
350    fn refuses_a_request_that_only_mentions_our_key_in_a_free_text_field() {
351        let ours = public_key_blob(&pubkey(0));
352        // The session identifier is a byte string with no UTF-8 constraint,
353        // so it is where raw key bytes can actually be smuggled - the username
354        // and service fields are text and would reject them.
355        let mut b = ssh_string(&ours);
356        b.push(SSH_MSG_USERAUTH_REQUEST);
357        b.extend_from_slice(&ssh_string(b"tommi"));
358        b.extend_from_slice(&ssh_string(b"ssh-connection"));
359        b.extend_from_slice(&ssh_string(b"publickey"));
360        b.push(1);
361        b.extend_from_slice(&ssh_string(ALG.as_bytes()));
362        b.extend_from_slice(&ssh_string(&public_key_blob(&pubkey(1))));
363
364        let err = sign_request(MNEMONIC, 0, &b)
365            .expect_err("must refuse: the key field names a different key");
366        assert!(
367            err.contains("different SSH key"),
368            "{}",
369            format!("unexpected: {err}")
370        );
371    }
