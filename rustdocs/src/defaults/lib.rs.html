1//! Default data for [Zigner](https://github.com/rotkonetworks/zigner)
2//! and Zigner-supporting ecosystem.
3//!
4//! This crate deals with data loaded into Zigner on build and retained on reset.
5//!
6//! Vault supports by default three networks: Polkadot, Kusama, and Westend,
7//! both on the cold and the hot side. More default networks could be considered
8//! in the future.
9//!
10//! # Defaults in the cold database
11//!
12//! Default build of the cold database for Vault includes:
13//!
14//! - general verifier `Verifier` set to `Some(VerifierValue)` with Parity
15//!   public key as a verifier value
16//! - network specs `OrderedNetworkSpecs` for default networks
17//! - verifiers for default networks, set to
18//!   `CurrentVerifier::Valid(ValidCurrentVerifier::General)`
19//! - two latest metadata versions for each of the default networks
20//! - default types information
21//!
22//! Latest metadata entries get updated as soon as they are published and could
23//! be fetched via RPC calls. For this, new metadata entry is added into
24//! `release_metadata` folder, and older one is removed.
25//!
26//! # Defaults in the hot database
27//!
28//! Default hot database contains:
29//!
30//! - network specs `NetworkSpecs` for default networks
31//! - address book containing the data for default networks
32//! - default types information
33//!
34//! # Features
35//! Feature `"signer"` corresponds to everything related to Vault air-gapped
36//! device. Currently the only Vault-related default is `Verifier` with Parity
37//! public key, that could be set up as a general verifier after Vault wipe by
38//! the user.
39//!
40//! Feature `"active"` corresponds to all Vault-related things happening
41//! **without** air-gap. This includes generating default:
42//! - network specs sets for cold and hot databases
43//! - verifiers for cold databases
44//! - metadata sets for cold databases
45//! - address book for hot database
46//! - types information for cold and hot databases
47//!
48//! Feature `"test"` includes both `"signer"` and `"active"` features, along
49//! with some testing, and is the default one.

85/// Generate default general verifier [`Verifier`].
86///
87/// The general verifier is the trust anchor for cold UPDATES (network specs,
88/// metadata, types) that reach the air-gapped device as QR payloads the user
89/// scans manually: the device accepts an update only if it is signed by this
90/// key. We anchor it to Rotko's ed25519 publisher key
91/// ([`constants::ROTKO_ZCASH_VERIFIER`]) - the SAME key already trusted for
92/// Zcash anchor attestation - instead of Parity's stock sr25519 key. This drops
93/// sr25519 from the update-trust path entirely.
94///
95/// NOTE: written only on FRESH cold-DB init (`cold_default`); already-
96/// provisioned devices keep whatever verifier is in their DB, so this takes
97/// effect on reflash / re-provision.
98///
99/// TODO(pq): migrate to an ed25519+ML-DSA hybrid `MultiSigner` variant (see the
100/// PQ-hybrid task) - format bump, not a redesign.
101/// TODO(domain-sep): a rigorous design would use a dedicated update-signing key
102/// distinct from the anchor-attestation key rather than reusing one ed25519 key
103/// for both purposes.
104pub fn default_general_verifier() -> Verifier {
105    Verifier {
106        v: Some(VerifierValue::Standard {
107            m: MultiSigner::Ed25519(sp_core::ed25519::Public::from_raw(
108                constants::ROTKO_ZCASH_VERIFIER,
109            )),
110        }),
111    }
112}

408/// Read metadata files from given directory. Is used to populate different
409/// variants of cold database (release and tests)
410#[cfg(feature = "active")]
411fn metadata(dir: &str) -> Result<Vec<MetaValues>> {
412    let mut out: Vec<MetaValues> = Vec::new();
413    let path_set = std::fs::read_dir(dir)?;
414    for x in path_set.flatten() {
415        if let Some(filename) = x.path().to_str() {
416            let meta_str = std::fs::read_to_string(x.path())?;
417            let new = MetaValues::from_str_metadata(meta_str.trim())?;
418            let mut found = false;
419            for a in default_network_info() {
420                if new.name == a.name {
421                    found = true;
422                    if let Some(prefix_from_meta) = new.optional_base58prefix {
423                        if prefix_from_meta != a.base58prefix {
424                            return Err(Error::Base58PrefixSpecsMismatch {
425                                specs: a.base58prefix,
426                                meta: prefix_from_meta,
427                            });
428                        }
429                    }
430                    break;
431                }
432            }
433            if !found {
434                return Err(Error::OrphanMetadata {
435                    name: new.name,
436                    filename: filename.to_string(),
437                });
438            }
439            out.push(new)
440        }
441    }
442    Ok(out)
443}

464#[cfg(feature = "active")]
465lazy_static! {
466    static ref REG_STRUCTS_WITH_NAMES: Regex = Regex::new(r"(pub )?struct (?P<name>.*?)( )?\{(?P<description>(\n +(pub )?\w+: .*(,)?)*\n)\}").expect("checked construction");
467    static ref REG_STRUCTS_NO_NAMES: Regex = Regex::new(r"(pub )?struct (?P<name>.*?)( )?\((pub )?(?P<description>.*)\)").expect("checked construction");
468    static ref REG_STRUCT_FIELDS: Regex = Regex::new(r"(?m)^ +(pub )?(?P<field_name>[\w]+): (?P<field_type>[^ ,]*\(.*\)[^ ,]*|[^ ,]*<.*>[^ ,]*|[^ ,]*\[.*\][^ ,]*|[^ ,]+)(,)?$").expect("checked construction");
469    static ref REG_ENUM: Regex = Regex::new(r"(pub )?enum (?P<name>.*?)( )?\{(?P<description>(\n +\w+( )?(\(.*\))?(\{[^}]*\})?(,)?)*)\n\}").expect("checked construction");
470    static ref REG_ENUM_VARIANTS: Regex = Regex::new(r"(?m)^ +(?P<variant_name>\w+)(( )?(?P<variant_type>(\(.*\))|(\{[^}]*\})))?(,)?$").expect("checked construction");
471    static ref REG_ENUM_SIMPLE: Regex = Regex::new(r"(?m)^(?P<simple_type>([^, ]*\[.*\][^, ]*)|([^, ]*<.*>[^, ]*)|([^, ]*\([^)]*\)[^, ]*)|([^, ]+))$").expect("checked construction");
472    static ref REG_ENUM_TUPLE: Regex = Regex::new(r"(?P<type_in_tuple>([^, ]*\[.*\][^, ]*)|([^, ]*<.*>[^, ]*)|([^, ]*\([^)]*\)[^, ]*)|([^, ]+))(, )?").expect("checked construction");
473    static ref REG_ENUM_STRUCT: Regex = Regex::new(r"(\n)?( +)?(?P<struct_field_name>\w+): (?P<struct_field_type>)(?P<fieldtype>[^ ,]*\(.*\)[^ ,]*|[^ ,]*<.*>[^ ,]*|[^ ,]*\[.*\][^ ,]*|[^ ,]+)(,)?").expect("checked construction");
474    static ref REG_TYPES: Regex = Regex::new(r#"(?m)(pub )?type (?P<name>.*) = (?P<description>.*);$"#).expect("checked construction");
475}

477/// Generate default types as [`TypeEntry`] set
478///
479/// Type definitions stored in `../defaults/default_types/full_types_information`
480/// are recorded with standard Rust syntax, processed with regular expressions
481/// to categorize, and collected as `TypeEntry` set.
482///
483/// Types information is necessary only to parse transactions produced with
484/// `RuntimeVersion` of the network metadata below `V14`. Therefore, it is
485/// obsolete for default networks.
486///
487/// Types information currently on file was collected for older metadata
488/// versions of Westend, Polkadot, Kusama, and Rococo, when they were still
489/// using metadata runtime version `V12` and `V13`. Type definitions were gathered
490/// (mostly) from substrate crates and in some cases from js client code, when
491/// it was not possible to find explicit descriptions in substrate crates.
492///
493/// This types collection is totally not expected to be exhaustive for any
494/// network a user may encounter, although the most common types are likely
495/// described. If user gets an error that certain type is unknown, this type
496/// description could be added into types file by user, and types update could
497/// be generated by user.
498#[cfg(feature = "active")]
499pub fn default_types_vec() -> Result<Vec<TypeEntry>> {
500    let filename = "../defaults/default_types/full_types_information";
501    let type_info = fs::read_to_string(filename)?;
502
503    let mut types_prep: Vec<TypeEntry> = Vec::new();
504
505    for caps1 in REG_STRUCTS_WITH_NAMES.captures_iter(&type_info) {
506        let struct_name = (caps1["name"]).to_string();
507        let struct_description = (caps1["description"]).to_string();
508        let mut struct_fields: Vec<StructField> = Vec::new();
509        for caps2 in REG_STRUCT_FIELDS.captures_iter(&struct_description) {
510            let new = StructField {
511                field_name: Some((caps2["field_name"]).to_string()),
512                field_type: (caps2["field_type"]).to_string(),
513            };
514            struct_fields.push(new);
515        }
516        let new_entry = TypeEntry {
517            name: struct_name,
518            description: Description::Struct(struct_fields),
519        };
520        types_prep.push(new_entry);
521    }
522    for caps in REG_STRUCTS_NO_NAMES.captures_iter(&type_info) {
523        let only_field = StructField {
524            field_name: None,
525            field_type: (caps["description"]).to_string(),
526        };
527        let new_entry = TypeEntry {
528            name: (caps["name"]).to_string(),
529            description: Description::Struct(vec![only_field]),
530        };
531        types_prep.push(new_entry);
532    }
533    for caps1 in REG_ENUM.captures_iter(&type_info) {
534        let enum_name = (caps1["name"]).to_string();
535        let enum_description = (caps1["description"]).to_string();
536        let enum_variants = enum_description
537            .lines()
538            .filter(|line| REG_ENUM_VARIANTS.is_match(line))
539            .map(|line| {
540                let caps2 = REG_ENUM_VARIANTS
541                    .captures(line)
542                    .expect("just checked it is match");
543                let variant_name = (caps2["variant_name"]).to_string();
544                let variant_type = match caps2.name("variant_type") {
545                    None => EnumVariantType::None,
546                    Some(a) => {
547                        let x = a.as_str().to_string();
548                        if x.starts_with('(') {
549                            // either a single type or a tuple
550                            match REG_ENUM_SIMPLE.captures(&x[1..x.len() - 1]) {
551                                // single type
552                                Some(b) => EnumVariantType::Type((b["simple_type"]).to_string()),
553                                // tuple
554                                None => EnumVariantType::Type(x),
555                            }
556                        } else {
557                            // struct with struct-like fields
558                            let mut type_is_struct: Vec<StructField> = Vec::new();
559                            for caps3 in REG_ENUM_STRUCT.captures_iter(&x) {
560                                let new = StructField {
561                                    field_name: Some((caps3["field_name"]).to_string()),
562                                    field_type: (caps3["field_type"]).to_string(),
563                                };
564                                type_is_struct.push(new);
565                            }
566                            EnumVariantType::Struct(type_is_struct)
567                        }
568                    }
569                };
570                EnumVariant {
571                    variant_name,
572                    variant_type,
573                }
574            })
575            .collect();
576        let new_entry = TypeEntry {
577            name: enum_name,
578            description: Description::Enum(enum_variants),
579        };
580        types_prep.push(new_entry);
581    }
582    for caps in REG_TYPES.captures_iter(&type_info) {
583        let new_entry = TypeEntry {
584            name: (caps["name"]).to_string(),
585            description: Description::Type((caps["description"]).to_string()),
586        };
587        types_prep.push(new_entry);
588    }
589
590    Ok(types_prep)
591}

601#[cfg(feature = "active")]
602pub fn substrate_chainspecs_with_ethereum() -> Vec<OrderedNetworkSpecs> {
603    let mut out: Vec<OrderedNetworkSpecs> = default_chainspecs();
604
605    let max_order = out.iter().max_by_key(|network| network.order);
606
607    // Add Mythos network
608    let mythos = OrderedNetworkSpecs {
609        specs: NetworkSpecs {
610            base58prefix: 29972,
611            color: String::from("#262528"),
612            decimals: 18,
613            encryption: Encryption::Ethereum,
614            genesis_hash: H256::from_str(
615                "f6ee56e9c5277df5b4ce6ae9983ee88f3cbed27d31beeb98f9f84f997a1ab0b9",
616            )
617            .expect("known value"),
618            logo: String::from("mythos"),
619            name: String::from("mythos"),
620            path_id: String::from("//mythos"),
621            secondary_color: String::from("#262626"),
622            title: String::from("Mythos"),
623            unit: String::from("MYTH"),
624        },
625        order: max_order.map(|v| v.order + 1).unwrap_or(0),
626    };
627    out.push(mythos);
628    out
629}
