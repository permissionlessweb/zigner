13/// function to take data as `Vec<u8>`, apply `raptorq` to get `Vec<EncodingPacket>`
14/// and serialize it to get `Vec<u8>` output
15pub fn make_data_packs(input: &[u8], chunk_size: u16) -> Result<Vec<QrData>, &'static str> {
16    // checking that data is not too long, set limit for now at 2^31 bit
17    if input.len() >= 0x80000000 {
18        return Err("Input data is too long, processing not possible");
19    }
20    // added at the beginning to each vector before transforming into qr code: contains input length info, also has first bit always 1 indicating it is new fountain qr - possibly need to change this later
21    let data_size_info = (input.len() as u32 + 0x80000000).to_be_bytes();
22
23    // number of additional packets; currently roughly equal to number of core packets
24    let repair_packets_per_block: u32 = {
25        if input.len() as u32 <= chunk_size as u32 {
26            0
27        } else {
28            input.len() as u32 / chunk_size as u32
29        }
30    };
31    // making `raptorq` Encoder, with defaults
32    let raptor_encoder = raptorq::Encoder::with_defaults(input, chunk_size);
33    // making EncodingPacket and deserializing each into `Vec<u8>`
34    let out: Vec<QrData> = raptor_encoder
35        .get_encoded_packets(repair_packets_per_block)
36        .iter()
37        .map(|x| QrData::Regular {
38            data: [data_size_info.to_vec(), x.serialize()].concat(),
39        })
40        .collect();
41    let len_check = out[0].len();
42    for x in out.iter() {
43        if x.len() != len_check {
44            return Err("Encoded chunks have different length");
45        }
46    }
47    if len_check > 2953 {
48        // 2953 is bytes limit for qr codes having 8-bit binary data
49        return Err("Encoded chunks too large to be turned into QR codes");
50    }
51    Ok(out)
52}

54/// function to take data as `Vec<Vec<u8>>` with all stuff added and make `Vec<QrCode>`
55fn make_qr_codes(data: Vec<QrData>) -> Result<Vec<QrCode>, Box<dyn std::error::Error>> {
56    let mut out: Vec<QrCode> = Vec::new();
57    for x in data.iter() {
58        let new = QrCode::encode_binary(x.data(), QrCodeEcc::Low)?;
59        out.push(new);
60    }
61    Ok(out)
62}

64fn make_apng<P>(
65    data: Vec<QrCode>,
66    output_name: P,
67    fps: u16,
68) -> Result<(), Box<dyn std::error::Error>>
69where
70    P: AsRef<Path>,
71{
72    let mut output_file = fs::File::create(output_name)?;
73    let frames_count: u32 = data.len() as u32;
74    let border_size = BORDER * SCALING;
75
76    // size is always positive and small
77    let size: u32 = (data[0].size() as u32) * (SCALING as u32) + 2 * border_size as u32;
78    let mut encoder = png::Encoder::new(&mut output_file, size, size);
79
80    encoder.set_color(png::ColorType::Indexed);
81    encoder.set_palette(qr_palette());
82    encoder.set_animated(frames_count, 0)?;
83    encoder.set_frame_delay(FPS_NOM, fps)?;
84    encoder.set_depth(png::BitDepth::One);
85
86    let mut writer = encoder.write_header()?;
87    // making actual apng
88    // qr.get_module(x,y) = false corresponds to back color (white by default)
89    // qr.get_module(x,y) = true corresponds to main color (black by default)
90    for qr in data.iter() {
91        let mut buffer: Vec<u8> = Vec::new();
92        for y in 0..size {
93            let mut pixels: BitVec<u8, Msb0> = BitVec::with_capacity(size as usize);
94            for x in 0..size {
95                pixels
96                    .push(!qr.get_module(x as i32 / SCALING - BORDER, y as i32 / SCALING - BORDER))
97            }
98            buffer.extend_from_slice(&pixels.into_vec());
99        }
100        writer.write_image_data(&buffer)?;
101    }
102    writer.finish()?;
103
104    Ok(())
105}

107/// Function to transform input `Vec<u8>` into fountain qr-code
108pub fn transform_into_qr_apng<P>(
109    input: &[u8],
110    chunk_size: u16,
111    fps: u16,
112    output_name: P,
113) -> Result<(), Box<dyn std::error::Error>>
114where
115    P: AsRef<Path>,
116{
117    let data_packs = make_data_packs(input, chunk_size)?;
118    make_apng(make_qr_codes(data_packs)?, output_name, fps)?;
119    Ok(())
120}

122/// Function to make appropriately sized qr code, apng or static
123pub fn make_pretty_qr<P>(input: &[u8], output_name: P) -> Result<(), Box<dyn std::error::Error>>
124where
125    P: AsRef<Path>,
126{
127    if input.len() <= 2953 {
128        let qr = png_qr(input, DataType::Regular)?;
129        match std::fs::write(output_name, qr) {
130            Ok(_) => Ok(()),
131            Err(e) => Err(Box::from(format!("Output error {e}"))),
132        }
133    } else {
134        transform_into_qr_apng(input, CHUNK_SIZE, FPS_DEN, output_name)
135    }
136}
