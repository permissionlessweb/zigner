1//! cosmos transaction signing
2//!
3//! parses amino JSON sign requests from QR codes and signs with secp256k1 ECDSA.
4//! cosmos amino signing: sign SHA256(sign_doc_bytes) with secp256k1.
5//!
6//! NOTE: we cannot use sp_core::ecdsa::Pair::sign() because it uses blake2b-256
7//! as prehash (Substrate convention). Cosmos requires SHA256.
8
9use crate::{Error, Result};
10
11/// QR prelude bytes for cosmos sign request: [0x53, 0x05, 0x10]
12const COSMOS_PRELUDE: [u8; 3] = [0x53, 0x05, 0x10];
13
14/// maximum sign doc size (64 KiB). amino JSON for any reasonable transaction
15/// is well under this. defense in depth against oversized payloads.
16const MAX_SIGN_DOC_LEN: usize = 65536;
17
18/// parsed cosmos sign request from QR
19pub struct CosmosSignRequest {
20    /// BIP44 account index (hardened `account'` in m/44'/coin'/account'/0/index)
21    pub account_index: u32,
22    /// BIP44 address index (the non-hardened `.../0/index`). Lets the wallet
23    /// spend from fresh, never-reused receive addresses derived off one account.
24    pub address_index: u32,
25    /// chain name (e.g. "noble", "osmosis")
26    pub chain_name: String,
27    /// amino JSON sign doc bytes (UTF-8)
28    pub sign_doc_bytes: Vec<u8>,
29}
30
31/// a single parsed message from the amino sign doc
32pub struct CosmosMsg {
33    /// message type (e.g. "Send", "IBC Transfer", "Swap", "Contract Call")
34    pub msg_type: String,
35    /// target address (recipient, validator, contract, etc.)
36    pub recipient: String,
37    /// amount string (e.g. "1000000 uusdc")
38    pub amount: String,
39    /// extra detail the user must see (min output, pool id, raw contract msg, etc.)
40    pub detail: String,
41    /// true if this message cannot be fully verified by the wallet.
42    /// contract calls and unknown types are inherently blind — the UI
43    /// must warn the user prominently before signing.
44    pub blind: bool,
45}
46
47/// display info extracted from amino JSON sign doc
48pub struct CosmosSignDocDisplay {
49    /// chain_id from sign doc
50    pub chain_id: String,
51    /// ALL messages in the sign doc — a cold wallet must never hide messages
52    pub msgs: Vec<CosmosMsg>,
53    /// fee string
54    pub fee: String,
55    /// memo
56    pub memo: String,
57}

59impl CosmosSignRequest {
60    /// parse a cosmos sign request from QR hex data
61    ///
62    /// format:
63    /// ```text
64    /// [0x53][0x05][0x10]       3B  prelude
65    /// [account_index: 4 LE]    4B
66    /// [chain_name_len: 1]      1B
67    /// [chain_name: N]          NB
68    /// [sign_doc_len: 4 LE]     4B
69    /// [sign_doc_bytes: N]      NB  canonical amino JSON
70    /// [address_index: 4 LE]    4B  OPTIONAL - absent means 0 (back-compat)
71    /// ```
72    pub fn from_qr_hex(hex: &str) -> Result<Self> {
73        let bytes =
74            hex::decode(hex).map_err(|e| Error::Other(anyhow::anyhow!("invalid hex: {e}")))?;
75
76        if bytes.len() < 12 {
77            return Err(Error::Other(anyhow::anyhow!("QR data too short")));
78        }
79
80        // check prelude
81        if bytes[0..3] != COSMOS_PRELUDE {
82            return Err(Error::Other(anyhow::anyhow!(
83                "invalid cosmos QR prelude: expected 530510, got {:02x}{:02x}{:02x}",
84                bytes[0],
85                bytes[1],
86                bytes[2]
87            )));
88        }
89
90        let mut offset = 3;
91
92        // account index (4 bytes LE)
93        let account_index = u32::from_le_bytes(
94            bytes[offset..offset + 4]
95                .try_into()
96                .map_err(|_| Error::Other(anyhow::anyhow!("failed to read account index")))?,
97        );
98        offset += 4;
99
100        // chain name (length-prefixed, 1 byte)
101        let chain_name_len = bytes[offset] as usize;
102        offset += 1;
103
104        if offset + chain_name_len > bytes.len() {
105            return Err(Error::Other(anyhow::anyhow!(
106                "chain name extends past end of data"
107            )));
108        }
109
110        let chain_name = String::from_utf8(bytes[offset..offset + chain_name_len].to_vec())
111            .map_err(|e| Error::Other(anyhow::anyhow!("invalid chain name UTF-8: {e}")))?;
112        offset += chain_name_len;
113
114        // sign doc bytes (length-prefixed, 4 bytes LE)
115        if offset + 4 > bytes.len() {
116            return Err(Error::Other(anyhow::anyhow!("missing sign doc length")));
117        }
118        let sign_doc_len = u32::from_le_bytes(
119            bytes[offset..offset + 4]
120                .try_into()
121                .map_err(|_| Error::Other(anyhow::anyhow!("failed to read sign doc length")))?,
122        ) as usize;
123        offset += 4;
124
125        if sign_doc_len > MAX_SIGN_DOC_LEN {
126            return Err(Error::Other(anyhow::anyhow!(
127                "sign doc too large ({sign_doc_len} bytes, max {MAX_SIGN_DOC_LEN})"
128            )));
129        }
130
131        if offset + sign_doc_len > bytes.len() {
132            return Err(Error::Other(anyhow::anyhow!(
133                "sign doc extends past end of data (need {}, have {})",
134                offset + sign_doc_len,
135                bytes.len()
136            )));
137        }
138
139        let sign_doc_bytes = bytes[offset..offset + sign_doc_len].to_vec();
140        offset += sign_doc_len;
141
142        // Optional trailing address_index (4 bytes LE). Absent = 0, keeping QRs
143        // that predate fresh-address derivation valid. Any other trailing length
144        // is a malformed/ambiguous payload and is rejected.
145        let address_index = match bytes.len() - offset {
146            0 => 0,
147            4 => u32::from_le_bytes(
148                bytes[offset..offset + 4]
149                    .try_into()
150                    .map_err(|_| Error::Other(anyhow::anyhow!("failed to read address index")))?,
151            ),
152            n => {
153                return Err(Error::Other(anyhow::anyhow!(
154                    "trailing data after sign doc ({n} extra bytes)"
155                )))
156            }
157        };
158
159        Ok(CosmosSignRequest {
160            account_index,
161            address_index,
162            chain_name,
163            sign_doc_bytes,
164        })
165    }
166}

172/// format a single coin from {"amount": "X", "denom": "Y"}
173fn format_coin(c: &serde_json::Value) -> String {
174    let a = c["amount"].as_str().unwrap_or("0");
175    let d = c["denom"].as_str().unwrap_or("");
176    format!("{a} {d}")
177}
178
179/// format the first coin from an array of coins
180fn format_first_coin(arr: &serde_json::Value) -> String {
181    arr.as_array()
182        .and_then(|a| a.first())
183        .map(format_coin)
184        .unwrap_or_else(|| "unknown".to_string())
185}
186
187/// format coin from an object (not array) — used by delegate/undelegate/single-token fields
188fn format_coin_obj(obj: &serde_json::Value) -> String {
189    obj.as_object()
190        .map(|o| {
191            let a = o.get("amount").and_then(|v| v.as_str()).unwrap_or("0");
192            let d = o.get("denom").and_then(|v| v.as_str()).unwrap_or("");
193            format!("{a} {d}")
194        })
195        .unwrap_or_else(|| "unknown".to_string())
196}
197
198/// extract a JSON number or string as a display string
199fn json_num_or_str(v: &serde_json::Value) -> String {
200    if let Some(s) = v.as_str() {
201        s.to_string()
202    } else if let Some(n) = v.as_u64() {
203        n.to_string()
204    } else if let Some(n) = v.as_i64() {
205        n.to_string()
206    } else {
207        "?".to_string()
208    }
209}
210
211/// format swap routes: "pool 1 → uatom, pool 7 → uosmo"
212fn format_routes(routes: &[serde_json::Value]) -> String {
213    if routes.is_empty() {
214        return String::new();
215    }
216    let parts: Vec<String> = routes
217        .iter()
218        .map(|r| {
219            let id = json_num_or_str(&r["pool_id"]);
220            let denom = r["token_out_denom"].as_str().unwrap_or("?");
221            format!("pool {id} → {denom}")
222        })
223        .collect();
224    parts.join(", ")
225}

446                    } else {

448                    },
449                    detail: raw_msg,
450                    blind: true,
451                }
452            }
453
454            // ================================================================
455            // unknown msg type — BLIND SIGNING REQUIRED
456            //
457            // we don't recognize this message type. show the raw amino type
458            // string so the user can at least see what it claims to be.
459            // ================================================================
460            _ => CosmosMsg {
461                msg_type: if typ.is_empty() {
462                    "(empty type)".to_string()
463                } else {
464                    typ.to_string()
465                },
466                recipient: String::new(),
467                amount: String::new(),
468                detail: serde_json::to_string(v).unwrap_or_default(),
469                blind: true,
470            },
471        }
472    }

475impl CosmosSignDocDisplay {
476    /// parse display info from amino JSON sign doc bytes.
477    /// parses ALL messages — a cold wallet must never silently hide messages
478    /// from the user, as that enables multi-message blind signing attacks.
479    pub fn from_json(bytes: &[u8]) -> Result<Self> {
480        let json: serde_json::Value = serde_json::from_slice(bytes)
481            .map_err(|e| Error::Other(anyhow::anyhow!("invalid sign doc JSON: {e}")))?;
482
483        let chain_id = json["chain_id"].as_str().unwrap_or("").to_string();
484        let memo = json["memo"].as_str().unwrap_or("").to_string();
485
486        // parse ALL fee coins — never hide fee components from user
487        let fee_amount = json["fee"]["amount"]
488            .as_array()
489            .map(|coins| {
490                let parts: Vec<String> = coins.iter().map(format_coin).collect();
491                if parts.is_empty() {
492                    "none".to_string()
493                } else {
494                    parts.join(" + ")
495                }
496            })
497            .unwrap_or_else(|| "unknown".to_string());
498
499        // parse ALL messages
500        let msgs = match json["msgs"].as_array() {
501            Some(arr) => arr.iter().map(CosmosMsg::from_json).collect(),
502            None => vec![],
503        };
504
505        Ok(CosmosSignDocDisplay {
506            chain_id,
507            msgs,
508            fee: fee_amount,
509            memo,
510        })
511    }
512}

514/// Extract the signer (source) address from every message in an amino sign doc.
515///
516/// Cosmos messages name their signer differently by type; we check the standard
517/// field names in each message's `value`. Messages with no recognizable signer
518/// field contribute nothing - they are already surfaced as `blind` in the
519/// display and cannot be source-bound here (the chain still rejects a signature
520/// from the wrong key).
521///
522/// The device uses this to refuse signing with a key that is not the message's
523/// signer, so a compromised host cannot silently choose which of the user's
524/// addresses pays by varying `address_index`.
525pub fn extract_signers(sign_doc_bytes: &[u8]) -> Result<Vec<String>> {
526    const SIGNER_FIELDS: &[&str] = &[
527        "from_address",      // MsgSend
528        "sender",            // MsgTransfer (IBC), swaps
529        "delegator_address", // staking
530        "signer",            // generic
531        "voter",             // gov vote
532        "depositor",         // gov deposit
533        "granter",           // authz / feegrant
534        "admin",             // wasm
535        "proposer",          // gov submit
536    ];
537
538    let doc: serde_json::Value = serde_json::from_slice(sign_doc_bytes)
539        .map_err(|e| Error::Other(anyhow::anyhow!("sign doc is not JSON: {e}")))?;
540
541    let mut signers = Vec::new();
542    if let Some(msgs) = doc["msgs"].as_array() {
543        for m in msgs {
544            let v = &m["value"];
545            for field in SIGNER_FIELDS {
546                if let Some(addr) = v[*field].as_str() {
547                    if !addr.is_empty() {
548                        signers.push(addr.to_string());
549                        break;
550                    }
551                }
552            }
553        }
554    }
555    Ok(signers)
556}

558/// Sign Cosmos Amino sign doc bytes with `secp256k1` ECDSA.
559///
560/// Computes `SHA256(sign_doc_bytes)` then signs with the secret key.
561/// Returns 64-byte compact signature (r || s).
562///
563/// IMPORTANT: this does NOT use `sp_core::ecdsa` because that uses `blake2b-256`
564/// as pre-hash. Cosmos Amino requires `SHA256`.
565#[cfg(feature = "cosmos")]
566pub fn sign_cosmos_amino(secret_key: &[u8; 32], sign_doc_bytes: &[u8]) -> Result<[u8; 64]> {
567    use secp256k1::{Message, Secp256k1, SecretKey};
568    use sha2::{Digest, Sha256};
569
570    let secp = Secp256k1::new();
571
572    // SHA256 prehash (cosmos amino convention)
573    let hash = Sha256::digest(sign_doc_bytes);
574
575    let sk = SecretKey::from_slice(secret_key)
576        .map_err(|e| Error::Other(anyhow::anyhow!("invalid secret key: {e}")))?;
577
578    let message = Message::from_digest_slice(&hash)
579        .map_err(|e| Error::Other(anyhow::anyhow!("invalid message hash: {e}")))?;
580
581    let signature = secp.sign_ecdsa(&message, &sk);
582
583    // compact serialization = 64 bytes (r || s)
584    Ok(signature.serialize_compact())
585}

591    #[test]
592    fn test_parse_cosmos_sign_request() {
593        // build a test QR payload
594        let account_index: u32 = 0;
595        let chain_name = b"noble";
596        let sign_doc = br#"{"chain_id":"noble-1","memo":"","msgs":[],"fee":{"amount":[],"gas":"0"},"account_number":"0","sequence":"0"}"#;
597
598        let mut payload = Vec::new();
599        payload.extend_from_slice(&COSMOS_PRELUDE);
600        payload.extend_from_slice(&account_index.to_le_bytes());
601        payload.push(chain_name.len() as u8);
602        payload.extend_from_slice(chain_name);
603        payload.extend_from_slice(&(sign_doc.len() as u32).to_le_bytes());

622            payload.extend_from_slice(&0u32.to_le_bytes()); // account_index
623            payload.push(b"noble".len() as u8);
624            payload.extend_from_slice(b"noble");
625            payload.extend_from_slice(&(sign_doc.len() as u32).to_le_bytes());
626            payload.extend_from_slice(sign_doc);
627            payload.extend_from_slice(trailing);
628            hex::encode(&payload)
629        };
630
631        // present: 4-byte LE address_index is parsed
632        let req = CosmosSignRequest::from_qr_hex(&build(&7u32.to_le_bytes())).unwrap();
633        assert_eq!(req.address_index, 7);
634
635        // absent: defaults to 0
636        assert_eq!(
637            CosmosSignRequest::from_qr_hex(&build(&[]))
638                .unwrap()
639                .address_index,
640            0
641        );
642
643        // malformed: any trailing length other than 0 or 4 is rejected
644        assert!(CosmosSignRequest::from_qr_hex(&build(&[1, 2, 3])).is_err());
645    }
