1//! Signed module packages: the trust layer of the update architecture.
2//!
3//! package = manifest || module_bytes
4//! manifest (fixed order, little-endian):
5//!   magic "ZIGM" | manifest_version:u8=1 | module_version:u32 |
6//!   min_kernel_version:u32 | module_hash:[32] | payload_kind:u8 |
7//!   base_hash:[32] | payload_len:u32 | desc_len:u16 | desc |
8//!   sig_count:u8 | sig_count * sig:[64]   (untagged; verifier matches keys)
9//!
10//! `module_hash` is always the sha256 of the RESULTING module. For
11//! payload_kind FULL the payload is that module verbatim. For a patch the
12//! payload is a delta and `base_hash` names the module it applies to - full
13//! modules are ~2.7 MB and take ~36 minutes over QR at the default playback
14//! speed, where a measured delta is tens of KB and takes seconds.
15//!
16//! `payload_len` pins the payload byte-exactly. Without it a package had no
17//! canonical encoding: zstd ignores bytes after its frame and bsdiff ignores
18//! bytes after its control stream, so arbitrary junk could be appended to a
19//! signed delta and it still verified. It also lets a wrong-sized payload be
20//! rejected BEFORE decompression rather than after.
21//!
22//! Signatures cover every manifest byte before sig_count over the
23//! domain-separated message "zigner-module-v1" || fields. 2-of-3 of the
24//! kernel-baked release keys must verify, each from a DISTINCT key.
25//! module_hash must equal sha256(module_bytes). The kernel additionally
26//! enforces module_version > last_installed (anti-rollback) - persisted
27//! counter, outside this parser.
28
29use ed25519_dalek::{Signature, VerifyingKey};
30use sha2::{Digest, Sha256};
31
32pub const MAGIC: &[u8; 4] = b"ZIGM";
33pub const DOMAIN: &[u8] = b"zigner-module-v1";
34pub const REQUIRED_SIGS: usize = 2;
35
36/// Wire format version. v1 never shipped - no package could ever verify,
37/// because the release keys are still placeholders - so v2 replaces it
38/// outright rather than being accepted alongside it.
39pub const MANIFEST_VERSION: u8 = 2;
40
41/// Payload is the complete module.
42pub const PAYLOAD_FULL: u8 = 0;
43/// Payload is a bsdiff delta against `base_hash`, zstd-compressed.
44///
45/// bsdiff+zstd rather than `zstd --patch-from`: measured on a real module
46/// pair it is smaller (33,723 vs 42,839 bytes) AND applies with pure-Rust
47/// crates, where zstd's prefix-dictionary path needs the C library.
48pub const PAYLOAD_BSDIFF_ZSTD: u8 = 1;

50#[derive(Debug)]
51pub enum ManifestError {
52    Truncated,
53    BadMagic,
54    UnsupportedVersion(u8),
55    HashMismatch,
56    DuplicateKey(u8),
57    UnknownKey(u8),
58    BadSignature(u8),
59    NotEnoughSignatures(usize),
60    Rollback {
61        offered: u32,
62        installed: u32,
63    },
64    KernelTooOld {
65        required: u32,
66        running: u32,
67    },
68    UnknownPayloadKind(u8),
69    /// A patch package was offered but no base module was supplied.
70    MissingBaseModule,
71    /// The supplied base is not the one this patch was built against.
72    BaseHashMismatch,
73    /// The delta did not apply, decompress, or stayed within its size bound.
74    PatchFailed,
75    /// The payload is not the length the signed manifest committed to.
76    PayloadLengthMismatch,
77}

79pub struct VerifiedModule<'a> {
80    pub module_version: u32,
81    pub min_kernel_version: u32,
82    pub description: String,
83    /// Borrowed for a full payload, owned when produced by applying a patch.
84    pub module_bytes: std::borrow::Cow<'a, [u8]>,
85}

87/// The fields of a manifest's signed region.
88#[derive(Debug, Clone, PartialEq, Eq)]
89pub struct ManifestFields {
90    pub module_version: u32,
91    pub min_kernel_version: u32,
92    /// sha256 of the resulting module, whether shipped whole or patched.
93    pub module_hash: [u8; 32],
94    pub payload_kind: u8,
95    /// The module a patch applies to; all-zero for a full payload.
96    pub base_hash: [u8; 32],
97    /// Exact payload length. Signed, so the package has one encoding.
98    pub payload_len: u32,
99    pub description: String,
100}

102/// Parse the signed region of a manifest, returning its fields and its exact
103/// length in bytes - the region the release signatures cover.
104///
105/// Shared by the verifier and by the offline signing flow, deliberately: a
106/// signer that parsed this format separately from the verifier would be one
107/// refactor away from displaying different fields than the ones it signs.
108pub fn parse_signing_prefix(data: &[u8]) -> Result<(ManifestFields, usize), ManifestError> {
109    let take = |b: &[u8], n: usize| -> Result<(usize, usize), ManifestError> {
110        if b.len() < n {
111            Err(ManifestError::Truncated)
112        } else {
113            Ok((n, b.len() - n))
114        }
115    };
116    let mut at = 0usize;
117    let mut need = |n: usize| -> Result<std::ops::Range<usize>, ManifestError> {
118        let _ = take(&data[at.min(data.len())..], n)?;
119        // checked_add for the same reason as ssh::take_string: usize is 32-bit
120        // on armeabi-v7a, and a wrapped offset would produce a reversed range
121        // rather than an error. Lengths here are currently bounded by a u16, so
122        // this is defence in depth - but the bound is a property of today's
123        // format, not of the arithmetic.
124        let end = at.checked_add(n).ok_or(ManifestError::Truncated)?;
125        let r = at..end;
126        at = end;
127        Ok(r)
128    };
129
130    if data[need(4)?] != *MAGIC {
131        return Err(ManifestError::BadMagic);
132    }
133    let v = data[need(1)?][0];
134    if v != MANIFEST_VERSION {
135        return Err(ManifestError::UnsupportedVersion(v));
136    }
137    let module_version = u32::from_le_bytes(data[need(4)?].try_into().unwrap());
138    let min_kernel_version = u32::from_le_bytes(data[need(4)?].try_into().unwrap());
139    let module_hash: [u8; 32] = data[need(32)?].try_into().unwrap();
140    let payload_kind = data[need(1)?][0];
141    if !matches!(payload_kind, PAYLOAD_FULL | PAYLOAD_BSDIFF_ZSTD) {
142        return Err(ManifestError::UnknownPayloadKind(payload_kind));
143    }
144    let base_hash: [u8; 32] = data[need(32)?].try_into().unwrap();
145    let payload_len = u32::from_le_bytes(data[need(4)?].try_into().unwrap());
146    let desc_len = u16::from_le_bytes(data[need(2)?].try_into().unwrap()) as usize;
147    let desc = data[need(desc_len)?].to_vec();
148
149    Ok((
150        ManifestFields {
151            module_version,
152            min_kernel_version,
153            module_hash,
154            payload_kind,
155            base_hash,
156            payload_len,
157            description: String::from_utf8_lossy(&desc).into_owned(),
158        },
159        at,
160    ))
161}

163/// Verify a full-payload package. Patch packages need [`verify_package_with_base`].
164pub fn verify_package<'a>(
165    package: &'a [u8],
166    release_keys: &[VerifyingKey; 3],
167    kernel_version: u32,
168    last_installed_version: u32,
169) -> Result<VerifiedModule<'a>, ManifestError> {
170    verify_package_with_base(
171        package,
172        release_keys,
173        kernel_version,
174        last_installed_version,
175        None,
176    )
177}

179/// Parse + fully verify a module package against the kernel's trust
180/// anchors and state. Returns the module only if EVERY check passes.
181///
182/// `base` is the currently-installed module, required only when the package
183/// carries a patch. The patch itself needs no trust: the signatures cover the
184/// RESULT hash, so a hostile or corrupt delta fails the final hash check.
185pub fn verify_package_with_base<'a>(
186    package: &'a [u8],
187    release_keys: &[VerifyingKey; 3],
188    kernel_version: u32,
189    last_installed_version: u32,
190    base: Option<&[u8]>,
191) -> Result<VerifiedModule<'a>, ManifestError> {
192    let take = |b: &'a [u8], n: usize| -> Result<(&'a [u8], &'a [u8]), ManifestError> {
193        if b.len() < n {
194            Err(ManifestError::Truncated)
195        } else {
196            Ok(b.split_at(n))
197        }
198    };
199
200    let (fields, signed_len) = parse_signing_prefix(package)?;
201    let module_version = fields.module_version;
202    let min_kernel_version = fields.min_kernel_version;
203    let hash = fields.module_hash;
204    let rest = &package[signed_len..];
205
206    let mut msg = Vec::with_capacity(DOMAIN.len() + signed_len);
207    msg.extend_from_slice(DOMAIN);
208    msg.extend_from_slice(&package[..signed_len]);
209
210    let (sc, mut rest) = take(rest, 1)?;
211    // No key index on the wire: a signature identifies its own key. For each
212    // signature, find which pinned key it verifies under, and count DISTINCT
213    // keys. This makes the threshold about *how many different keys signed*,
214    // the property we actually want, instead of a self-declared index the
215    // signer could set wrong. One key cannot fill two slots: once a key is
216    // matched it is skipped, so a second signature from it advances nothing.
217    let mut matched = [false; 3];
218    let mut valid = 0usize;
219    for i in 0..sc[0] {
220        let (sig, r) = take(rest, 64)?;
221        rest = r;
222        let signature = Signature::from_bytes(sig.try_into().unwrap());
223        // verify_strict, not verify: the permissive form accepts small-order
224        // public keys and non-canonical R, both of which admit forgeries. This
225        // is a trust anchor, so take the strict equation. A given (msg, sig)
226        // verifies under at most one key, so the first match is the only match.
227        let mut advanced = false;
228        for k in 0..3 {
229            if matched[k] {
230                continue;
231            }
232            if release_keys[k].verify_strict(&msg, &signature).is_ok() {
233                matched[k] = true;
234                valid += 1;
235                advanced = true;
236                break;
237            }
238        }
239        // A signature that matches no unmatched pinned key is either forged, from
240        // a key we do not pin, or a duplicate of one already counted. Fail closed.
241        if !advanced {
242            return Err(ManifestError::BadSignature(i));
243        }
244    }
245    if valid < REQUIRED_SIGS {
246        return Err(ManifestError::NotEnoughSignatures(valid));
247    }
248
249    // Length first: this is the cheap reject. A payload of the wrong size
250    // cannot be the one that was signed, and checking here means a hostile
251    // delta never reaches the decompressor at all.
252    if rest.len() as u64 != fields.payload_len as u64 {
253        return Err(ManifestError::PayloadLengthMismatch);
254    }
255
256    // Resolve the payload into the actual module before any hash check.
257    let module_bytes: std::borrow::Cow<'a, [u8]> = match fields.payload_kind {
258        PAYLOAD_FULL => {
259            // An unused base_hash must be zero, so there is exactly one
260            // encoding of a full package and no room for a second reading.
261            if fields.base_hash != [0u8; 32] {
262                return Err(ManifestError::BaseHashMismatch);
263            }
264            std::borrow::Cow::Borrowed(rest)
265        }
266        PAYLOAD_BSDIFF_ZSTD => {
267            let base = base.ok_or(ManifestError::MissingBaseModule)?;
268            if <[u8; 32]>::from(Sha256::digest(base)) != fields.base_hash {
269                return Err(ManifestError::BaseHashMismatch);
270            }
271            std::borrow::Cow::Owned(apply_patch(base, rest)?)
272        }
273        other => return Err(ManifestError::UnknownPayloadKind(other)),
274    };
275    if <[u8; 32]>::from(Sha256::digest(module_bytes.as_ref())) != hash {
276        return Err(ManifestError::HashMismatch);
277    }
278    if module_version <= last_installed_version {
279        return Err(ManifestError::Rollback {
280            offered: module_version,
281            installed: last_installed_version,
282        });
283    }
284    if min_kernel_version > kernel_version {
285        return Err(ManifestError::KernelTooOld {
286            required: min_kernel_version,
287            running: kernel_version,
288        });
289    }
290
291    Ok(VerifiedModule {
292        module_version,
293        min_kernel_version,
294        description: fields.description,
295        module_bytes,
296    })
297}

299/// Largest module we will reconstruct from a delta.
300///
301/// This bound is load-bearing, not hygiene. The signatures cover the manifest
302/// (and therefore the RESULT hash) but NOT the payload bytes. So a valid
303/// signed manifest can be paired with a swapped payload, and the mismatch is
304/// only caught by the hash check at the end. Without a cap, a decompression
305/// bomb would be expanded in full before we ever got there.
306pub const MAX_MODULE_BYTES: usize = 16 * 1024 * 1024;
307
308/// Reconstruct a module from `base` plus a zstd-compressed bsdiff delta.
309///
310/// Measured on a real module pair: 33,723 bytes against 2.0 MB for the whole
311/// module - seconds of QR scanning instead of half an hour.
312///
313/// Public so a fuzz harness can reach it directly. It is the most exposed
314/// parser we have: the payload it consumes is NOT covered by the release
315/// signatures - the manifest commits only to the RESULT hash - so bsdiff and
316/// ruzstd are handed attacker-controlled bytes by design.
317///
318/// The `take` is the bomb defence: decompression stops at the cap rather than
319/// running to completion, so a swapped payload costs bounded work and then
320/// fails, instead of expanding without limit before the hash check catches it.
321pub fn apply_patch(base: &[u8], patch: &[u8]) -> Result<Vec<u8>, ManifestError> {
322    use std::io::Read;
323    let decoder =
324        ruzstd::decoding::StreamingDecoder::new(patch).map_err(|_| ManifestError::PatchFailed)?;
325    let mut raw = Vec::new();
326    decoder
327        .take(MAX_MODULE_BYTES as u64)
328        .read_to_end(&mut raw)
329        .map_err(|_| ManifestError::PatchFailed)?;
330    let mut out = Vec::new();
331    bsdiff::patch(base, &mut raw.as_slice(), &mut out).map_err(|_| ManifestError::PatchFailed)?;
332    // Belt to the take's braces: bsdiff output should be bounded by the patch
333    // it was read from, but that is a property of the format rather than
334    // something enforced above, so check it rather than assume it.
335    if out.len() > MAX_MODULE_BYTES {
336        return Err(ManifestError::PatchFailed);
337    }
338    Ok(out)
339}

341/// Build the signed region of a manifest - the exact bytes the release keys
342/// sign, and the exact bytes a signing device is handed.
343///
344/// Split from assembly because the two halves happen on different machines:
345/// this runs on the build host, signing happens on separate offline devices,
346/// and [`assemble_package`] joins the results.
347pub fn build_signing_prefix(
348    module_hash: [u8; 32],
349    payload_kind: u8,
350    base_hash: [u8; 32],
351    payload_len: u32,
352    module_version: u32,
353    min_kernel_version: u32,
354    description: &str,
355) -> Vec<u8> {
356    let mut out = Vec::new();
357    out.extend_from_slice(MAGIC);
358    out.push(MANIFEST_VERSION);
359    out.extend_from_slice(&module_version.to_le_bytes());
360    out.extend_from_slice(&min_kernel_version.to_le_bytes());
361    out.extend_from_slice(&module_hash);
362    out.push(payload_kind);
363    out.extend_from_slice(&base_hash);
364    out.extend_from_slice(&payload_len.to_le_bytes());
365    // u16 length: the description carries the human-readable changelog the
366    // device shows on the confirm screen, and it is INSIDE the signed bytes -
367    // so what the user reads is what the release keys attested to. A u8 would
368    // cap that at 255 bytes (one sentence) and, worse, `as u8` silently wrapped
369    // for anything longer, producing a package that misparses.
370    assert!(
371        description.len() <= u16::MAX as usize,
372        "module description too long: {} bytes (max {})",
373        description.len(),
374        u16::MAX
375    );
376    out.extend_from_slice(&(description.len() as u16).to_le_bytes());
377    out.extend_from_slice(description.as_bytes());
378    out
379}

381/// The message the release keys actually sign: domain tag, then the prefix.
382pub fn signing_message(prefix: &[u8]) -> Vec<u8> {
383    let mut msg = Vec::with_capacity(DOMAIN.len() + prefix.len());
384    msg.extend_from_slice(DOMAIN);
385    msg.extend_from_slice(prefix);
386    msg
387}

389/// Join a prefix with signatures collected from offline devices, plus payload.
390///
391/// Signatures are untagged - the verifier matches each against the pinned keys -
392/// so the order they are joined in does not matter.
393pub fn assemble_package(prefix: &[u8], sigs: &[[u8; 64]], payload: &[u8]) -> Vec<u8> {
394    let mut out = prefix.to_vec();
395    out.push(sigs.len() as u8);
396    for sig in sigs {
397        out.extend_from_slice(sig);
398    }
399    out.extend_from_slice(payload);
400    out
401}

403/// Packaging helper for TESTS ONLY: signs with keys held right here.
404///
405/// The real release path never does this - it would need every signing key on
406/// one machine, exactly what the 2-of-3 custody model forbids. That path is
407/// [`build_signing_prefix`], offline devices, then [`assemble_package`].
408#[allow(clippy::too_many_arguments)] // test helper: mirrors the full package field set
409pub fn build_package(
410    payload: &[u8],
411    module_hash: [u8; 32],
412    payload_kind: u8,
413    base_hash: [u8; 32],
414    module_version: u32,
415    min_kernel_version: u32,
416    description: &str,
417    signers: &[ed25519_dalek::SigningKey],
418) -> Vec<u8> {
419    use ed25519_dalek::Signer as _;
420    let prefix = build_signing_prefix(
421        module_hash,
422        payload_kind,
423        base_hash,
424        payload.len() as u32,
425        module_version,
426        min_kernel_version,
427        description,
428    );
429    let msg = signing_message(&prefix);
430    let sigs: Vec<[u8; 64]> = signers
431        .iter()
432        .map(|key| key.sign(&msg).to_bytes())
433        .collect();
434    assemble_package(&prefix, &sigs, payload)
435}

437/// Convenience for the common case: a complete module, no patch.
438pub fn build_full_package(
439    module_bytes: &[u8],
440    module_version: u32,
441    min_kernel_version: u32,
442    description: &str,
443    signers: &[ed25519_dalek::SigningKey],
444) -> Vec<u8> {
445    build_package(
446        module_bytes,
447        Sha256::digest(module_bytes).into(),
448        PAYLOAD_FULL,
449        [0u8; 32],
450        module_version,
451        min_kernel_version,
452        description,
453        signers,
454    )
