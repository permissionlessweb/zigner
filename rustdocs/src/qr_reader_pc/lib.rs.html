37/// Main cycle of video capture.
38/// Returns a string with decoded QR message in HEX format or error.
39///
40/// # Arguments
41///
42/// * `camera_settings` - `CameraSettings` struct that holds the camera parameters
43pub fn run_with_camera(camera_settings: CameraSettings) -> anyhow::Result<String> {
44    let camera_index = match camera_settings.index {
45        Some(index) => index,
46        None => return Err(anyhow!("There is no camera index.")),
47    };
48
49    let window = "video capture";
50    highgui::named_window(window, 1)?;
51
52    let mut camera = create_camera(camera_index, DEFAULT_WIDTH, DEFAULT_HEIGHT)?;
53    skip_frames(&mut camera); // clearing old frames if they are in the camera buffer
54
55    let mut out = Ready::NotYet(InProgress::None);
56    let mut line = String::new();
57
58    let pb = ProgressBar::new(1);
59    loop {
60        match out {
61            Ready::NotYet(decoding) => {
62                if let InProgress::Fountain(f) = &decoding {
63                    pb.set_length(f.total as u64);
64                    pb.set_position(f.collected() as u64)
65                }
66                out = match camera_capture(&mut camera, window) {
67                    Ok(img) => process_qr_image(&img, decoding)?,
68                    Err(_) => Ready::NotYet(decoding),
69                };
70            }
71            Ready::Yes(a) => {
72                line.push_str(&hex::encode(a));
73                break;
74            }
75            _ => todo!(),
76        }
77
78        if highgui::wait_key(10)? > 0 {
79            println!("Exit");
80            break;
81        };
82    }
83    highgui::destroy_window(window)?;
84    Ok(line)
85}

87fn create_camera(
88    camera_index: i32,
89    width: u32,
90    height: u32,
91) -> anyhow::Result<videoio::VideoCapture> {
92    #[cfg(ocvrs_opencv_branch_32)]
93    let mut camera = videoio::VideoCapture::new_default(camera_index)?;
94    #[cfg(not(ocvrs_opencv_branch_32))]
95    let mut camera = videoio::VideoCapture::new(camera_index, videoio::CAP_ANY)?;
96
97    match videoio::VideoCapture::is_opened(&camera) {
98        Ok(opened) if opened => {
99            camera.set(CAP_PROP_FRAME_WIDTH, width.into())?;
100            camera.set(CAP_PROP_FRAME_HEIGHT, height.into())?;
101        }
102        Ok(_) => return Err(anyhow!("Camera already opened.")),
103        Err(e) => return Err(anyhow!("Can`t open camera. {}", e)),
104    };
105
106    let mut frame = Mat::default();
107
108    match camera.read(&mut frame) {
109        Ok(_) if frame.size()?.width > 0 => Ok(camera),
110        Ok(_) => Err(anyhow!("Zero frame size.")),
111        Err(e) => Err(anyhow!("Can`t read camera. {}", e)),
112    }
113}

115fn camera_capture(camera: &mut videoio::VideoCapture, window: &str) -> Result<GrayImage> {
116    let mut frame = Mat::default();
117    camera.read(&mut frame)?;
118
119    if frame.size()?.width > 0 {
120        highgui::imshow(window, &frame)?;
121    };
122
123    let mut image: GrayImage = ImageBuffer::new(DEFAULT_WIDTH, DEFAULT_HEIGHT);
124    let mut ocv_gray_image = Mat::default();
125
126    cvt_color_def(&frame, &mut ocv_gray_image, COLOR_BGR2GRAY)?;
127
128    for y in 0..ocv_gray_image.rows() {
129        for x in 0..ocv_gray_image.cols() {
130            let pixel: Luma<u8> = Luma([*ocv_gray_image.at_2d(y, x)?]);
131            image.put_pixel(x as u32, y as u32, pixel);
132        }
133    }
134
135    Ok(image)
136}

138/// Function for decoding QR grayscale image.
139/// Returns a string with decoded QR message in HEX format or error.
140///
141/// # Arguments
142///
143/// * `image` - Grayscale image containing QR and background
144/// * `decoding` - Stores accumulated payload data for animated QR.
145pub fn process_qr_image(image: &GrayImage, decoding: InProgress) -> anyhow::Result<Ready> {
146    let mut qr_decoder = quircs::Quirc::new();
147    let codes = qr_decoder.identify(image.width() as usize, image.height() as usize, image);
148
149    match codes.last() {
150        Some(Ok(code)) => match code.decode() {
151            Ok(decoded) => process_decoded_payload(decoded.payload, &None, decoding)
152                .map_err(|e| anyhow!(e.to_string())),
153            Err(_) => Ok(Ready::NotYet(decoding)),
154        },
155        Some(_) => Ok(Ready::NotYet(decoding)),
156        None => Ok(Ready::NotYet(decoding)),
157    }
158}

160fn print_list_of_cameras() {
161    let mut indexes: Vec<i32> = vec![];
162    for dev_port in 0..=MAX_CAMERA_INDEX {
163        if create_camera(dev_port, DEFAULT_WIDTH, DEFAULT_HEIGHT).is_ok() {
164            indexes.push(dev_port);
165        };
166    }
167    println!("\nList of available devices:");
168    for index in indexes {
169        println!("Camera index: {index}");
170    }
171}

173fn skip_frames(camera: &mut videoio::VideoCapture) {
174    for _x in 0..SKIPPED_FRAMES_QTY {
175        if let Ok(false) | Err(_) = camera.grab() {
176            break;
177        }
178    }
179}

181/// The program's argument parser.
182/// The parser initializes the `CameraSettings` structure with program's arguments
183/// (described in the `readme.md` file).
184pub fn arg_parser(arguments: Vec<String>) -> anyhow::Result<CameraSettings> {
185    let mut args = arguments.into_iter();
186    args.next(); // skip program name
187
188    let mut settings = CameraSettings { index: None };
189
190    while let Some(arg) = args.next() {
191        let par = args.next().unwrap_or_default();
192
193        match &arg[..] {
194            "d" | "-d" | "--device" => match par.trim().parse() {
195                Ok(index) => settings.index = Some(index),
196                Err(e) => return Err(anyhow!("Camera index parsing error: {}", e)),
197            },
198            "h" | "-h" | "--help" => println!("Please read readme.md file."),
199            "l" | "-l" | "--list" => print_list_of_cameras(),
200            _ => return Err(anyhow!("Argument parsing error.")),
201        };
202    }
203
204    match settings.index {
205        Some(_) => Ok(settings),
206        None => Err(anyhow!(
207            "Need to provide camera index. Please read readme.md file."
208        )),
209    }
210}
