1//! Address key associated non-secret information stored in Vault database
2//!
3//! Vault database has a tree `ADDRTREE` with [`AddressKey`] in key form
4//! as a key and encoded [`AddressDetails`] as a value.
5//! [`AddressDetails`] contains non-secret information associated with address key.
6//!
7//! `ADDRTREE` is operated mainly from within the Vault.
8//!
9//! Release and test versions of the cold database are generated on the Active side.
10
11use parity_scale_codec::{Decode, Encode};
12use sled::IVec;
13use sp_runtime::MultiSigner;
14
15use crate::helpers::IdenticonStyle;
16use crate::{
17    crypto::Encryption,
18    error::{Error, Result},
19    helpers::multisigner_to_encryption,
20    keyring::{AddressKey, NetworkSpecsKey},
21};
22
23/// Address key associated non-secret information stored in Vault database
24///
25/// Info that should be available for any address key.
26/// No secrets are stored there.
27#[derive(Decode, PartialEq, Eq, Encode, Debug, Clone)]
28pub struct AddressDetails {
29    /// seed name (as it is known to the Vault device)
30    pub seed_name: String,
31
32    /// derivation path, only with soft (`/`) and hard (`//`) junctions (i.e. no password)
33    pub path: String,
34
35    /// whether the address key has an associated password
36    pub has_pwd: bool,
37
38    /// set of networks, identified through [`NetworkSpecsKey`], that are available
39    /// to work with this address key
40    pub network_id: Option<NetworkSpecsKey>,
41
42    /// encryption algorithm associated with the address key and all its associated networks
43    pub encryption: Encryption,
44
45    /// address, or its parent address, had or could have secret exposed
46    pub secret_exposed: bool,
47}
48
49impl AddressDetails {
50    /// Gets ([`MultiSigner`](https://docs.rs/sp-runtime/6.0.0/sp_runtime/enum.MultiSigner.html),
51    /// [`AddressDetails`]) tuple from [`AddressKey`] and associated value from
52    /// database tree `ADDRTREE`.
53    ///
54    /// Checks that there is no encryption mismatch.
55    pub fn process_entry_with_key_checked(
56        address_key: &AddressKey,
57        address_details_encoded: IVec,
58    ) -> Result<(MultiSigner, Self)> {
59        let multisigner = address_key.multi_signer().clone();
60        let address_details = AddressDetails::decode(&mut &address_details_encoded[..])?;
61        // Allow Ed25519 MultiSigner with non-substrate encryption types that store 32-byte keys
62        // Similar to ECDSA/Ethereum special case
63        let ms_encryption = multisigner_to_encryption(&multisigner);
64        let is_ecdsa_ethereum = ms_encryption == Encryption::Ecdsa
65            && address_details.encryption == Encryption::Ethereum;
66        let is_ecdsa_cosmos =
67            ms_encryption == Encryption::Ecdsa && address_details.encryption == Encryption::Cosmos;
68        let is_ed25519_non_substrate = ms_encryption == Encryption::Ed25519
69            && matches!(
70                address_details.encryption,
71                Encryption::Penumbra | Encryption::Zcash | Encryption::LedgerEd25519
72            );
73        if ms_encryption != address_details.encryption
74            && !is_ecdsa_ethereum
75            && !is_ecdsa_cosmos
76            && !is_ed25519_non_substrate
77        {
78            return Err(Error::EncryptionMismatch {
79                address_key: address_key.to_owned(),
80                encryption: address_details.encryption,
81            });
82        }
83        let network_specs_key = &address_details.network_id;
84        if let Some(network_specs_key) = network_specs_key {
85            let (_, network_specs_key_encryption) = network_specs_key.genesis_hash_encryption()?;
86            if network_specs_key_encryption != address_details.encryption {
87                return Err(Error::EncryptionMismatch {
88                    address_key: address_key.to_owned(),
89                    encryption: address_details.encryption,
90                });
91            }
92        }
93        Ok((multisigner, address_details))
94    }
95
96    /// Gets ([`MultiSigner`](https://docs.rs/sp-runtime/6.0.0/sp_runtime/enum.MultiSigner.html),
97    /// [`AddressDetails`]) tuple from database tree `ADDRTREE` (key, value) entry.
98    ///
99    /// Checks that there is no encryption mismatch.
100    pub fn process_entry_checked(
101        (address_key_vec, address_details_encoded): (IVec, IVec),
102    ) -> Result<(MultiSigner, Self)> {
103        let address_key = AddressKey::from_ivec(&address_key_vec)?;
104        AddressDetails::process_entry_with_key_checked(&address_key, address_details_encoded)
105    }
106
107    /// Gets [`AddressDetails`] from [`AddressKey`] and associated value from
108    /// database tree `ADDRTREE`.
109    ///
110    /// Checks that there is no encryption mismatch.
111    pub fn from_entry_with_key_checked(
112        address_key: &AddressKey,
113        address_details_encoded: IVec,
114    ) -> Result<Self> {
115        let (_, address_details) =
116            AddressDetails::process_entry_with_key_checked(address_key, address_details_encoded)?;
117        Ok(address_details)
118    }
119
120    /// Checks if the [`AddressDetails`] have empty derivation path (i.e.
121    /// derivation path is empty and there is no password).
122    ///
123    /// Address key in this case is called root key or seed key.
124    pub fn is_root(&self) -> bool {
125        self.path.is_empty() && !self.has_pwd && self.network_id.is_none()
126    }
127
128    /// The style to use for identicons.
129    ///
130    /// Jdenticon for root key.
131    /// Otherwise, the style is taken from the encryption algorithm.
132    pub fn identicon_style(&self) -> IdenticonStyle {
133        if self.is_root() {
134            return IdenticonStyle::Jdenticon;
135        }
136        self.encryption.identicon_style()
137    }
138}
