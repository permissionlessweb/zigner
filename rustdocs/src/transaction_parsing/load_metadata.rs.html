22enum FirstCard {
23    WarningCard(TransactionCard),
24    VerifierCard(TransactionCard),
25}

27pub fn load_metadata(database: &sled::Db, data_hex: &str) -> Result<TransactionAction> {
28    let checked_info = pass_crypto(data_hex, TransferContent::LoadMeta)?;
29    let (meta, genesis_hash) = ContentLoadMeta::from_slice(&checked_info.message).meta_genhash()?;
30    let meta_values = MetaValues::from_slice_metadata(&meta)?;
31    let general_verifier = get_general_verifier(database)?;
32    let verifier_key = VerifierKey::from_parts(genesis_hash);
33    let valid_current_verifier = try_get_valid_current_verifier(database, &verifier_key)?.ok_or(
34        Error::LoadMetaUnknownNetwork {
35            name: meta_values.name.clone(),
36        },
37    )?;
38    let specs_invariants =
39        genesis_hash_in_specs(database, genesis_hash)?.ok_or(Error::LoadMetaNoSpecs {
40            name: meta_values.name.clone(),
41            valid_current_verifier: valid_current_verifier.clone(),
42            general_verifier: general_verifier.clone(),
43        })?;
44    if meta_values.name != specs_invariants.name {
45        return Err(Error::LoadMetaWrongGenesisHash {
46            name_metadata: meta_values.name,
47            name_specs: specs_invariants.name,
48            genesis_hash,
49        });
50    }
51    if let Some(prefix_from_meta) = meta_values.optional_base58prefix {
52        if prefix_from_meta != specs_invariants.base58prefix {
53            return Err(
54                definitions::error::MetadataError::Base58PrefixSpecsMismatch {
55                    specs: specs_invariants.base58prefix,
56                    meta: prefix_from_meta,
57                }
58                .into(),
59            );
60        }
61    }
62    let mut stub = TrDbColdStub::new();
63    let mut index = 0;
64    let optional_ext_warning = {
65        if meta_values.warn_incomplete_extensions {
66            stub = stub.new_history_entry(Event::Warning {
67                warning: Warning::MetadataExtensionsIncomplete.show(),
68            });
69            Some(Card::Warning(Warning::MetadataExtensionsIncomplete).card(&mut index, 0))
70        } else {
71            None
72        }
73    };
74
75    let first_card = match checked_info.verifier {
76        Verifier { v: None } => {
77            stub = stub.new_history_entry(Event::Warning {
78                warning: Warning::NotVerified.show(),
79            });
80            match valid_current_verifier {
81                ValidCurrentVerifier::Custom {
82                    v: Verifier { v: None },
83                } => (),
84                ValidCurrentVerifier::Custom {
85                    v:
86                        Verifier {
87                            v: Some(verifier_value),
88                        },
89                } => {
90                    return Err(Error::NeedVerifier {
91                        name: meta_values.name,
92                        verifier_value,
93                    })
94                }
95                ValidCurrentVerifier::General => match general_verifier {
96                    Verifier { v: None } => (),
97                    Verifier {
98                        v: Some(verifier_value),
99                    } => {
100                        return Err(Error::NeedGeneralVerifier {
101                            content: GeneralVerifierForContent::Network {
102                                name: meta_values.name,
103                            },
104                            verifier_value,
105                        })
106                    }
107                },
108            }
109            FirstCard::WarningCard(Card::Warning(Warning::NotVerified).card(&mut index, 0))
110        }
111        Verifier {
112            v: Some(ref new_verifier_value),
113        } => {
114            match valid_current_verifier {
115                ValidCurrentVerifier::Custom { v: a } => {
116                    if checked_info.verifier != a {
117                        match a {
118                            Verifier { v: None } => {
119                                return Err(Error::LoadMetaSetVerifier {
120                                    name: meta_values.name,
121                                    new_verifier_value: new_verifier_value.to_owned(),
122                                })
123                            }
124                            Verifier {
125                                v: Some(old_verifier_value),
126                            } => {
127                                return Err(Error::LoadMetaVerifierChanged {
128                                    name: meta_values.name,
129                                    old_verifier_value,
130                                    new_verifier_value: new_verifier_value.to_owned(),
131                                })
132                            }
133                        }
134                    }
135                }
136                ValidCurrentVerifier::General => {
137                    if checked_info.verifier != general_verifier {
138                        match general_verifier {
139                            Verifier { v: None } => {
140                                return Err(Error::LoadMetaSetGeneralVerifier {
141                                    name: meta_values.name,
142                                    new_general_verifier_value: new_verifier_value.to_owned(),
143                                })
144                            }
145                            Verifier {
146                                v: Some(old_general_verifier_value),
147                            } => {
148                                return Err(Error::LoadMetaGeneralVerifierChanged {
149                                    name: meta_values.name,
150                                    old_general_verifier_value,
151                                    new_general_verifier_value: new_verifier_value.to_owned(),
152                                })
153                            }
154                        }
155                    }
156                }
157            }
158            FirstCard::VerifierCard(Card::Verifier(new_verifier_value).card(&mut index, 0))
159        }
160    };
161    if accept_meta_values(database, &meta_values)? {
162        stub = stub.add_metadata(&meta_values);
163        let checksum = stub.store_and_get_checksum(database)?;
164        let meta_display = MetaValuesDisplay::get(&meta_values);
165        let meta_card = Card::Meta(meta_display).card(&mut index, 0);
166        match first_card {
167            FirstCard::WarningCard(warning_card) => match optional_ext_warning {
168                Some(ext_warning) => Ok(TransactionAction::Stub {
169                    s: Box::new(TransactionCardSet {
170                        warning: Some(vec![ext_warning, warning_card]),
171                        meta: Some(vec![meta_card]),
172                        ..Default::default()
173                    }),
174                    u: checksum,
175                    stub: StubNav::LoadMeta {
176                        l: specs_invariants.first_network_specs_key,
177                    },
178                }),
179                None => Ok(TransactionAction::Stub {
180                    s: Box::new(TransactionCardSet {
181                        warning: Some(vec![warning_card]),
182                        meta: Some(vec![meta_card]),
183                        ..Default::default()
184                    }),
185                    u: checksum,
186                    stub: StubNav::LoadMeta {
187                        l: specs_invariants.first_network_specs_key,
188                    },
189                }),
190            },
191            FirstCard::VerifierCard(verifier_card) => match optional_ext_warning {
192                Some(ext_warning) => Ok(TransactionAction::Stub {
193                    s: Box::new(TransactionCardSet {
194                        warning: Some(vec![ext_warning]),
195                        verifier: Some(vec![verifier_card]),
196                        meta: Some(vec![meta_card]),
197                        ..Default::default()
198                    }),
199                    u: checksum,
200                    stub: StubNav::LoadMeta {
201                        l: specs_invariants.first_network_specs_key,
202                    },
203                }),
204                None => Ok(TransactionAction::Stub {
205                    s: Box::new(TransactionCardSet {
206                        verifier: Some(vec![verifier_card]),
207                        meta: Some(vec![meta_card]),
208                        ..Default::default()
209                    }),
210                    u: checksum,
211                    stub: StubNav::LoadMeta {
212                        l: specs_invariants.first_network_specs_key,
213                    },
214                }),
215            },
216        }
217    } else {
218        Err(Error::MetadataKnown {
219            name: meta_values.name,
220            version: meta_values.version,
221        })
222    }
223}
